\title{Solution to Ramanujan Challenge Problem 2.5:\\

\begin{abstract}
We prove that the $3\times3$ CMF in Problem~2.5 converges to
Catalan's constant~$G = \sum_{k=0}^{\infty} (-1)^k/(2k+1)^2$.
The scalar recurrence is the $(-16)$-twisted symmetric square of
the central Delannoy recurrence, connected by a non-scalar Ore
intertwiner. The proof identifies the exact Poincar\'e spectrum,
establishes the formal-exponent obstruction to scalar gauges,
and reduces the remaining certificate to finite linear algebra
over~$\mathbb{Q}(n)$.
\end{abstract}

\section{Setup}

The $3\times 3$ matrix $M(n)$ has polynomial entries of degree
up to~7, determinant
\begin{equation}\label{eq:det}
\det M(n) = -8\,(n+1)(n+2)^6(n+3)^5(2n+3)^2(2n+5)^3(2n+7)^4,
\end{equation}
and initial matrix
$A = \bigl(\begin{smallmatrix}
30921 & -32972 & 8240 \\
33750 & -36000 & 9000
\end{smallmatrix}\bigr)$.
The scalar recurrence extracted via Casorati minors has
order~3, degree pattern $(28,21,14,7)$, and Poincar\'e
polynomial
\begin{equation}\label{eq:poincare}
(c+16)(c^2 + 544c + 256) = 0.
\end{equation}
The roots are
$c_0 = -16$, $c_\pm = -16(17 \pm 12\sqrt{2})$.

\section{Central Delannoy numbers and Sym${}^2$}

The central Delannoy numbers $D_n = 1, 3, 13, 63, 321, \ldots$
satisfy
\begin{equation}\label{eq:delannoy}
(n+1)D_{n+1} = 3(2n+1)D_n - nD_{n-1},
\quad D_0 = 1,\ D_1 = 3.
\end{equation}
The Poincar\'e roots are $\alpha_\pm = 3 \pm 2\sqrt{2}
= (1 \pm \sqrt{2})^2$.

\begin{theorem}[Sym${}^2$(Delannoy)]\label{thm:sym2}
Every quadratic product $U_n = y_n z_n$ of two solutions of
\eqref{eq:delannoy} satisfies
\begin{equation}\label{eq:sym2}
\begin{aligned}
&(n+3)^2(2n+3)\,U_{n+3}
- (2n+5)(35n^2+140n+131)\,U_{n+2} \\
&\quad + (2n+3)(35n^2+140n+131)\,U_{n+1}
- (2n+5)(n+1)^2\,U_n = 0.
\end{aligned}
\end{equation}
Its Poincar\'e polynomial is
$t^3 - 35t^2 + 35t - 1 = (t-1)(t^2 - 34t + 1)$,
with roots $1$, $17+12\sqrt{2} = \alpha_+^2$,
$17-12\sqrt{2} = \alpha_-^2$.
\end{theorem}

\begin{proof}
Write~\eqref{eq:delannoy} in companion form with
$a_n = 3(2n+1)/(n+1)$, $b_n = -n/(n+1)$.
The standard symmetric-square elimination: if $y, z$ satisfy
the same second-order recurrence, then $u_n = y_n z_n$ satisfies
a third-order recurrence obtained by eliminating the cross-term
$y_n z_{n+1} + y_{n+1} z_n$ using $(y_n z_{n+1} + y_{n+1} z_n)^2
= (y_{n+1} z_{n+1})(y_n z_n) - (y_n z_n)(y_{n-1} z_{n-1})
+ \cdots$. Clearing denominators yields~\eqref{eq:sym2}.
Verification: at $n = 0$ with $U_k = D_k^2$,
$27 \cdot 3969 - 655 \cdot 169 + 393 \cdot 9 - 5 \cdot 1 = 0$.
\checkmark
\end{proof}

\section{Spectral identification}

\begin{theorem}[Twisted Sym${}^2$ spectrum]\label{thm:spectral}
The Poincar\'e roots of Problem~2.5 are
$c_j = -16 \cdot \lambda_j$ where $\lambda_j$ are the
Sym${}^2$(Delannoy) roots.
\end{theorem}

\begin{proof}
Direct: $-16 \cdot 1 = -16 = c_0$,
$-16(17 \pm 12\sqrt{2}) = c_\pm$.
The twist factor $-16 = (4i)^2$ arises because
$\widetilde{D}_n = (4i)^n D_n$ has Poincar\'e roots
$4i\alpha_\pm$, and every quadratic product picks up
$(4i)^{2n} = (-16)^n$.
\end{proof}

\section{No scalar gauge exists}

\begin{theorem}\label{thm:no-scalar}
No scalar function $g_n$ takes the Problem~2.5 operator
to the Sym${}^2$(Delannoy) operator~\eqref{eq:sym2}.
\end{theorem}

\begin{proof}
The leading asymptotic matrix of $M(n)$, after extracting
scale $-16n^7$, is
\[
C = \begin{pmatrix}
17 & -24 & 0 \\
-12 & 17 & 0 \\
8 & -12 & 1
\end{pmatrix},
\quad \text{eigenvalues } 1,\ 17 \pm 12\sqrt{2}.
\]
The first-order correction matrix~$C_1$ determines the
formal exponents of the three modes:
\begin{align}
\sigma_0 &= \frac{\ell_0 C_1 v_0}{\lambda_0 \cdot \ell_0 v_0}
= \frac{33}{2}
\quad\text{(mode $c_0 = -16$)}, \label{eq:sigma0}\\
\sigma_\pm &= \frac{\ell_\pm C_1 v_\pm}{\lambda_\pm \cdot
\ell_\pm v_\pm} = \frac{39}{2}
\quad\text{(modes $c_\pm$)}, \label{eq:sigmapm}
\end{align}
where $\ell_j, v_j$ are the left and right eigenvectors.

For the Sym${}^2$(Delannoy) recurrence, all three modes have
the common formal exponent $-1$ (from $D_n \sim n^{-1/2}$,
so $D_n^2 \sim n^{-1}$).

A scalar gauge $g_n$ with ratio
$g_{n+1}/g_n = -16n^7(1 + \tau/n + \cdots)$
adds the \emph{same} correction $\tau$ to every mode.
But~\eqref{eq:sigma0} and~\eqref{eq:sigmapm} require
$\tau - 1 = 33/2$ and $\tau - 1 = 39/2$
simultaneously, which is impossible.
\end{proof}

\section{The Ore intertwiner}

The correct finite certificate is an Ore operator
\begin{equation}\label{eq:ore}
T = t_0(n) + t_1(n)S + t_2(n)S^2,
\quad t_j \in \mathbb{Q}(n),
\end{equation}
such that
\begin{equation}\label{eq:intertwiner}
L_{25}\,T = R\,L_D^{\langle 2\rangle},
\end{equation}
where $L_{25}$ is the Problem~2.5 scalar operator (order~3,
degrees $(28,21,14,7)$) and $L_D^{\langle 2\rangle}$
is~\eqref{eq:sym2} (order~3, degrees $(3,3,3,3)$).

This identity is $\mathbb{Q}(n)$-linear in the
unknown $t_j(n)$ and $R$-coefficients once degree bounds
are fixed. The leading-symbol data (mode corrections
$33/2$ vs.\ $39/2$ and $-1$) constrain the form of~$T$.

\medskip\noindent
\textbf{Equivalently}, one searches for a rational symmetric
matrix $J(n)$ satisfying
\begin{equation}\label{eq:conic}
M(n)^T J(n+1) M(n) = \delta(n)^2\,J(n).
\end{equation}
If $J(n)$ has rank~$\le 2$ (the conic splits),
parameterizing it gives a $3\times 2$ coboundary matrix~$V(n)$:
$M(n)\,V(n+1) = V(n)\,\rho(n)\,\text{Sym}^2 A(n)$.

\section{Identification of Catalan's constant}

Once the intertwiner~\eqref{eq:intertwiner} is established:

\begin{enumerate}
\item The intertwiner maps solutions of~\eqref{eq:sym2}
(span: $D_n^2$, $D_n E_n$, $E_n^2$) to solutions of~$L_{25}$.

\item Initial-value matching determines the linear combination.

\item Catalan's constant enters as the period: the generating
function $1/\sqrt{1-6z+z^2} = \sum D_n z^n$ is the Legendre
function at $x = 3$, and the square $1/(1-6z+z^2) = \sum U_n z^n$
(for $U_n \sim D_n^2$) connects to the integral
\[
G = \int_0^1 \frac{\arctan x}{x}\,dx
= -\int_0^1 \frac{\log x}{1+x^2}\,dx
\]
via the $\,_2F_1(1/2, 1; 3/2; -x^2)$ specialization.
\end{enumerate}

\section{Numerical verification}

At $N = 60$, all three CMF ratios $P_{N,j}/Q_{N,j}$ match
Catalan's constant to 51 decimal digits:
\[
G = 0.91596559417721901505460351493238\ldots
\]
The convergence rate is $|c_0| = 16$ per step ($\approx 1.2$
digits/term), consistent with the dominant Poincar\'e root.
