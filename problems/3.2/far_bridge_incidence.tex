\begin{proposition}[Far-bridge incidence majorants]
\label{prop:far-bridge-incidence}
Fix an odd prime~$p$.  Let $\pi_p$ be the projective Ap\'ery orbit on
$\{0,\ldots,p-1\}$, and let $\Gamma_p(q)$ be a reflection-invariant
collection of filtered four-consecutive-occurrence windows as in
Proposition~\ref{prop:projective-variance-reduction}.  Assume that every
selected window has span at most
\[
 H_p=\lfloor\sqrt p\rfloor.
\]
For each ordered separated pair of distinct reflection orbits, make the
deterministic choice of oriented representatives used in
Proposition~\ref{prop:near-bridge-energy}, and write its bridge as~$G$.
Let $E_p^{\mathrm{far}}(K)$ count the pairs for which $G>K$, where
$1\le K<p$ is an integer.

Define the conditioned incidence
\[
 B_p(K)=
 \sum_{\substack{q\in\mathbf P^1(\mathbf F_p)\\
                   \gamma=(x_0,x_1,x_2,x_3)\in\Gamma_p(q)}}
 \#\{z:x_3+K<z<p,\ \pi_p(z)=\pi_p(x_3)\},
\]
and put
\[
 \operatorname{Supp}(\overline\gamma)
 =\{x_0,x_1,x_2,x_3,
     p-1-x_3,p-1-x_2,p-1-x_1,p-1-x_0\}.
\]
The external conditioned incidence is
\[
 B_p^\circ(K)=
 \sum_{\substack{q\in\mathbf P^1(\mathbf F_p)\\
                   \gamma=(x_0,x_1,x_2,x_3)\in\Gamma_p(q)}}
 \#\left\{z:
 \begin{array}{c}
 x_3+K<z<p,\quad \pi_p(z)=\pi_p(x_3),\\
 z\notin\operatorname{Supp}(\overline\gamma)
 \end{array}
 \right\}.
\]
For an occurrence~$z$ of a projective state, let
$z=z_0<z_1<z_2<z_3$ denote that occurrence and the next three occurrences
of the same state, when they exist.  Define $B_p^{[4]}(K)$ by retaining in
$B_p^\circ(K)$ only those~$z$ for which these successors exist and
$z_3-z_0\le H_p$.  Finally, define the nonwrapping triple incidence
\[
 A_p^{\mathrm{nw}}(K)=
 \#\left\{(x,y,z):
 \begin{array}{c}
 0\le x<y<z<p,\quad 2\le y-x\le H_p,\\
 z-y>K,\quad \pi_p(x)=\pi_p(y)=\pi_p(z)
 \end{array}
 \right\}.
\]
Then
\begin{equation}
 E_p^{\mathrm{far}}(K)\le B_p^{[4]}(K)
 \le B_p^\circ(K)
 \le B_p(K)\le A_p^{\mathrm{nw}}(K).
\label{eq:far-incidence-hierarchy}
\end{equation}
If $M_p=\sum_q|\Gamma_p(q)|$ is the selected raw mass, then also
\begin{equation}
 0\le B_p(K)-B_p^\circ(K)\le4M_p.
\label{eq:far-reflection-baseline}
\end{equation}
Moreover, if $N_h$ denotes the positional-gap continuant, then
\begin{equation}
\begin{split}
 A_p^{\mathrm{nw}}(K)
 ={}&\sum_{s=2}^{H_p}\ \sum_{G=K+1}^{p-1-s}
 \#\bigl\{x\in\mathbf Z:
 0\le x\le p-1-s-G,\\[-2pt]
 &\hspace{105pt}
 N_s(x)\equiv N_G(x+s)\equiv0\pmod p\bigr\}.
\end{split}
\label{eq:far-incidence-continuants}
\end{equation}
In particular, it is bounded above by the unrestricted affine-root mass
\begin{equation}
 \widetilde A_p(K)=
 \sum_{s=2}^{H_p}\ \sum_{G=K+1}^{p-1-s}
 \#\{u\in\mathbf F_p:
       N_s(u)=N_G(u+s)=0\}.
\label{eq:far-incidence-affine}
\end{equation}
The summands in~\eqref{eq:far-incidence-affine} count actual
$\mathbf F_p$-roots; they are not degrees of polynomial greatest common
divisors.
\end{proposition}

\begin{proof}
Take an ordered separated pair counted by
$E_p^{\mathrm{far}}(K)$.  Its deterministic representatives have the form
\[
 \gamma=(x_0,x_1,x_2,x_3),\qquad
 \delta=(y_0,y_1,y_2,y_3),
 \qquad x_3<y_0,\qquad y_0-x_3>K.
\]
Both windows belong to the same projective fiber, so
$\pi_p(y_0)=\pi_p(x_3)$.  Map the ordered quotient pair to
$(\gamma,y_0)$.

This map is injective.  Indeed, $\gamma$ determines the first reflection
orbit.  Given $y_0$, the next three occurrences of
$\pi_p(y_0)$ are uniquely determined by the ordered fiber.  Hence there is
at most one four-consecutive-occurrence window beginning at~$y_0$, and, when
the image comes from a selected pair, it is precisely~$\delta$.  Thus the
second reflection orbit is recovered as well.  Separation implies that
$y_0$ is not in the reflection-invariant support of the first orbit.
Moreover, $\delta$ supplies the next three occurrences after~$y_0$, and its
span is at most~$H_p$.  The image is therefore counted by~$B_p^{[4]}(K)$.
This proves the first inequality in
\eqref{eq:far-incidence-hierarchy}.  No factor of two is lost here: the
canonical choice assigns one oriented pair to each ordered pair of quotient
orbits, whereas $B_p^{[4]}(K)$ already sums over all selected orientations.
The inequalities $B_p^{[4]}(K)\le B_p^\circ(K)\le B_p(K)$ are immediate.
For a fixed raw window, the two incidences differ only at later indices among
the four reflected indices
$p-1-x_3,p-1-x_2,p-1-x_1,p-1-x_0$.  Thus their difference is at most four
per selected raw window, proving~\eqref{eq:far-reflection-baseline}.

For an incidence $(\gamma,z)$ counted by~$B_p(K)$, set
\[
 x=x_0,\qquad y=x_3.
\]
The selected-window hypotheses give
$2\le y-x\le H_p$, and
\[
 \pi_p(x)=\pi_p(y)=\pi_p(z),\qquad z-y>K.
\]
The map $(\gamma,z)\mapsto(x,y,z)$ is injective, because the two internal
indices of a four-consecutive-occurrence window are the first two
occurrences of $\pi_p(x)$ strictly between its endpoints.  This proves
$B_p(K)\le A_p^{\mathrm{nw}}(K)$.

It remains only to reparametrize the latter set.  Put
\[
 s=y-x,\qquad G=z-y.
\]
Then
\[
 2\le s\le H_p,\qquad K+1\le G\le p-1-s,\qquad
 0\le x\le p-1-s-G.
\]
All two-time intervals are nonwrapping.  The exact gap criterion therefore
gives
\[
 \pi_p(x)=\pi_p(x+s)
 \iff N_s(x)\equiv0\pmod p
\]
and
\[
 \pi_p(x+s)=\pi_p(x+s+G)
 \iff N_G(x+s)\equiv0\pmod p.
\]
This proves~\eqref{eq:far-incidence-continuants}.  Dropping the interval
condition on the standard representative of~$x$ gives
$A_p^{\mathrm{nw}}(K)\le\widetilde A_p(K)$.

Finally, a degree such as
$\deg\gcd_{\mathbf F_p[x]}(N_s(x),N_G(x+s))$ also counts common irreducible
factors having no root in~$\mathbf F_p$.  It is therefore only an upper
bound for the corresponding summand in
\eqref{eq:far-incidence-affine}, not the exact observable in this
proposition.
\end{proof}

\begin{lemma}[Uniform projective-fiber cap]
\label{lem:far-projective-fiber-cap}
For every prime $p\ge5$ and every
$q\in\mathbf P^1(\mathbf F_p)$,
\[
 m_p(q):=\#\{0\le n<p:\pi_p(n)=q\}\le4p^{2/3}.
\]
Consequently, uniformly in~$K$,
\begin{equation}
 A_p^{\mathrm{nw}}(K)\le6p^{5/3}.
\label{eq:far-fiber-cap-incidence}
\end{equation}
\end{lemma}

\begin{proof}
For $p=5$, the first conclusion follows from $m_p(q)\le p$.
The second follows from
$A_p^{\mathrm{nw}}(K)\le\binom p3$.
Hence assume $p\ge7$.

Write the occurrence list of~$q$ as
$r_1<\cdots<r_m$, and let $e_h$ be the number of its consecutive gaps
$r_{i+1}-r_i$ that equal~$h$.  The exact gap criterion and the
nonvanishing of~$N_h$ modulo~$p$ give
\[
 e_h\le\#\{0\le x<p-h:N_h(x)=0\}\le3(h-1).
\]
Moreover,
\[
 \sum_{h=1}^{p-1}e_h=m-1,
 \qquad
 \sum_{h=1}^{p-1}he_h=r_m-r_1\le p-1.
\]
Put $T=\lfloor p^{1/3}\rfloor$.  Splitting the gaps at~$T$ gives
\[
 \begin{split}
 m-1
 &\le \sum_{h\le T}3(h-1)
      +\frac1{T+1}\sum_{h>T}he_h\\
 &\le\frac32T(T-1)+\frac{p-1}{T+1}
 <\frac52p^{2/3}.
 \end{split}
\]
The stated bound follows.

The number of pairs $x<y$ in a common projective fiber with
$y-x\le H_p$ is at most
\[
 \sum_{h=2}^{H_p}\#\{0\le x<p-h:N_h(x)=0\}
 \le3\sum_{h=2}^{H_p}(h-1)
 \le\frac32p.
\]
After such a pair is fixed, there are at most
$\max_q m_p(q)\le4p^{2/3}$ possible later points in its fiber.  Dropping
the lower bound on the second gap can only enlarge
$A_p^{\mathrm{nw}}(K)$, and proves
\eqref{eq:far-fiber-cap-incidence}.
\end{proof}

\begin{lemma}[Reciprocal-gap triple bound]
\label{lem:far-reciprocal-gap-triples}
For every prime $p\ge7$, uniformly in~$K$,
\begin{equation}
 A_p^{\mathrm{nw}}(K)\ll p^{3/2}\log(2p).
\label{eq:far-reciprocal-gap-triples}
\end{equation}
Consequently,
\begin{equation}
 \sum_{X<p\le2X}A_p^{\mathrm{nw}}(K)\ll X^{5/2},
\label{eq:far-reciprocal-gap-dyadic}
\end{equation}
uniformly for any prime-dependent choice of~$K$.
\end{lemma}

\begin{proof}
For a projective state~$q$, write its full occurrence list as
\[
 r_{q,1}<\cdots<r_{q,m_q}
\]
and put
\[
 g_{q,i}=r_{q,i+1}-r_{q,i},\qquad
 T_q=\sum_{i=1}^{m_q-1}g_{q,i}^{-2},\qquad
 L_p=\sum_qT_q.
\]
Empty sums are understood as zero.  The orbit-energy theorem
\ref{thm:oe-energy-three-halves} is stated on
$\{1,\ldots,p-2\}$.  If $m_q^\circ$ denotes the corresponding fiber
sizes and $a_q=m_q-m_q^\circ$, then
$\sum_qa_q=2$ and $\sum_qa_q^2\le4$.  The triangle inequality in
$\ell^2$ therefore gives
\begin{equation}
 \mathcal F_p^{\mathrm{full}}
 :=\sum_qm_q^2
 \le\left(\left(\sum_q(m_q^\circ)^2\right)^{1/2}+2\right)^2
 \ll p^{3/2}.
\label{eq:far-full-fiber-energy}
\end{equation}

Let $e_q(h)$ count consecutive gaps of length~$h$ in the $q$-fiber.
The exact gap criterion and $\deg N_h=3(h-1)$ give
\[
 \sum_qe_q(h)\le3(h-1).
\]
It follows that
\begin{equation}
 L_p
 =\sum_{h=2}^{p-1}\frac{\sum_qe_q(h)}{h^2}
 \le3\sum_{h=2}^{p-1}\frac{h-1}{h^2}
 \le3\log p.
\label{eq:far-reciprocal-gap-budget}
\end{equation}
For a fiber of size at least two, H\"older's inequality gives
\begin{equation}
 m_q-1
 =\sum_i g_{q,i}^{2/3}g_{q,i}^{-2/3}
 \le\left(\sum_i g_{q,i}\right)^{2/3}T_q^{1/3}
 \le p^{2/3}T_q^{1/3}.
\label{eq:far-fiber-size-reciprocal}
\end{equation}
Consequently,
$m_q\le1+p^{2/3}T_q^{1/3}$ for every fiber, including those of size at
most one.

Put $H=H_p$ and let $P_q(H)$ be the number of pairs of occurrences in
the $q$-fiber whose positive difference is at most~$H$.  Partition
$\{0,\ldots,p-1\}$ into
$J=\lceil p/H\rceil$ consecutive blocks of cardinality at most~$H$, and
let $t_{q,j}$ count the $q$-occurrences in the $j$-th block.  Every pair
counted by $P_q(H)$ lies in one block or in two adjacent blocks.  Hence
\begin{equation}
 P_q(H)
 \le\sum_j\binom{t_{q,j}}2+\sum_jt_{q,j}t_{q,j+1}
 \le\frac32\sum_jt_{q,j}^2.
\label{eq:far-near-pair-blocks}
\end{equation}
For a nonempty block, let $T_{q,j}$ be the sum of $g_{q,i}^{-2}$ over
the consecutive fiber gaps having both endpoints in that block.  Its
$t_{q,j}$ points have total internal span at most~$H$, so the argument
of~\eqref{eq:far-fiber-size-reciprocal} inside this block gives
\[
 t_{q,j}-1\le H^{2/3}T_{q,j}^{1/3},\qquad
 t_{q,j}^2\le2+2H^{4/3}T_{q,j}^{2/3}.
\]
There are at most $m_q$ nonempty blocks, the sets of internal gaps are
disjoint, and $\sum_jT_{q,j}\le T_q$.  By concavity,
\begin{equation}
 P_q(H)
 \le3m_q+3H^{4/3}J^{1/3}T_q^{2/3}.
\label{eq:far-near-pair-reciprocal}
\end{equation}

After a near pair in a fiber is fixed, dropping the requirements that the
third occurrence be later and that its gap exceed~$K$ leaves at most
$m_q$ choices.  Thus
\begin{equation}
 A_p^{\mathrm{nw}}(K)
 \le\sum_qm_qP_q(H)
 \le3\mathcal F_p^{\mathrm{full}}
 +3H^{4/3}J^{1/3}\sum_qm_qT_q^{2/3}.
\label{eq:far-triple-before-budget}
\end{equation}
By~\eqref{eq:far-fiber-size-reciprocal}, and using the fact that at most
$p$ fibers are nonempty,
\begin{align}
 \sum_qm_qT_q^{2/3}
 &\le\sum_qT_q^{2/3}+p^{2/3}\sum_qT_q \notag\\
 &\le p^{1/3}L_p^{2/3}+p^{2/3}L_p.
\label{eq:far-weighted-reciprocal-budget}
\end{align}
Finally, $J\le2p/H$, so
$H^{4/3}J^{1/3}\le2^{1/3}Hp^{1/3}$.  Combining
\eqref{eq:far-full-fiber-energy},
\eqref{eq:far-reciprocal-gap-budget},
\eqref{eq:far-triple-before-budget}, and
\eqref{eq:far-weighted-reciprocal-budget}, and using
$H\le\sqrt p$, proves~\eqref{eq:far-reciprocal-gap-triples}.
Summing this estimate over $X<p\le2X$ and using
$\pi(2X)-\pi(X)\ll X/\log X$ proves
\eqref{eq:far-reciprocal-gap-dyadic}.
\end{proof}

\begin{remark}[Scale of the remaining arithmetic input]
By Lemma~\ref{lem:nonvanish}, for $s,G<p$ neither gap polynomial vanishes
identically modulo~$p$.  The actual common-root count is at most the degree
of the greatest common divisor, and hence at most $3(s-1)$.  Consequently
the elementary degree estimate gives only
\[
 \widetilde A_p(K)
 \le\sum_{s\le H_p}\sum_{G<p}3(s-1)\ll pH_p^2\ll p^2.
\]
After summing over $X<p\le2X$, this is
$O(X^3/\log X)$, one full power of~$X$ above the required separated-energy
budget.  Lemma~\ref{lem:far-reciprocal-gap-triples} improves this elementary
estimate to
\[
 \sum_{X<p\le2X}A_p^{\mathrm{nw}}(K)
 \ll X^{5/2}.
\]
This remains a factor $X^{1/2}\log X$ above the target.  Thus
\eqref{eq:far-incidence-hierarchy} still requires an average arithmetic
estimate for the actual affine roots, or the sharper conditioning retained
in~$B_p^{[4]}(K)$.
\end{remark}
