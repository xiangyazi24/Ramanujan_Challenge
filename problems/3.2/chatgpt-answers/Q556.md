# 15. Exact verification code

```python
from math import comb, gcd


def L(n, k):
    if k < 0 or k > n:
        return 0
    return comb(n, k) * comb(n + k, k)


def franel(m):
    return sum(comb(m, a)**3 for a in range(m + 1))


def apery(m):
    return sum(L(m, k)**2 for k in range(m + 1))


def coeffs_T(n, q):
    J = (n - q) // (2*q + 1)
    C = [0] * (n + 1)
    C[0] = sum(L(n, i) * franel(i) for i in range(J + 1))
    for d in range(1, n + 1):
        lo = max(0, J - d + 1)
        hi = min(J, n - d)
        C[d] = sum(
            (-1)**(J-i)
            * L(n, i+d)
            * comb(i+d, i)
            * comb(d-1, J-i)
            * franel(i)
            for i in range(lo, hi + 1)
        )
    return C


def primes_upto(N):
    sieve = [True] * (N + 1)
    sieve[:2] = [False, False]
    for a in range(2, int(N**0.5) + 1):
        if sieve[a]:
            for b in range(a*a, N + 1, a):
                sieve[b] = False
    return [a for a in range(2, N + 1) if sieve[a]]


q = 3
for n in range(4, 151):
    C = coeffs_T(n, q)
    for p in primes_upto(n):
        if p <= q or n // p != q:
            continue
        r = n - q*p
        j = min(r, p-1-r)
        Aj = apery(j) % p
        for d, value in enumerate(C):
            expected = 0
            if d % p == 0 and d // p <= q:
                expected = Aj * L(q, d // p)
            assert (value - expected) % p == 0

        content = 0
        for value in C:
            content = gcd(content, abs(value))
        assert (content % p == 0) == (Aj == 0)

print("all q=3 checks passed")
```
