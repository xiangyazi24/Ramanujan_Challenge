# SageMath
from sage.all import *
from functools import reduce


def Pcoef(m):
    return 34*m^3 + 51*m^2 + 27*m + 5


def apery_pair(N):
    b = [ZZ(1), ZZ(5)]
    a = [QQ(0), QQ(6)]
    for m in range(1, N):
        den = (m + 1)^3
        b.append((Pcoef(m)*b[m] - m^3*b[m-1]) // den)
        a.append((Pcoef(m)*a[m] - m^3*a[m-1]) / den)
    return a, b


def gcd_list(v):
    return reduce(gcd, [abs(ZZ(x)) for x in v], ZZ(0))


def q1_hits_at_indices(n, js, b):
    out = []
    for j in js:
        cand = [n-j]
        if (n+1+j) % 2 == 0:
            cand.append((n+1+j)//2)
        good = [p for p in cand if is_prime(p) and b[j] % p == 0]
        out.append((j, good))
    return out


def companion_image_data(n, hit_pairs):
    # hit_pairs = [(j_1,p_1),...]
    L = max(j for j,p in hit_pairs)
    a, b = apery_pair(L)
    D = lcm(range(1, L+1))^3
    PP = prod(p for j,p in hit_pairs)

    c = []
    B = []
    for j,p in hit_pairs:
        assert b[j] % p == 0
        Aj = ZZ(D*a[j])
        c.append((PP//p)*Aj)
        B.append(b[j]//p)

    g1 = gcd_list(c)
    minors = []
    for i in range(len(c)):
        for k in range(i+1, len(c)):
            minors.append(c[i]*B[k] - c[k]*B[i])

    Delta2 = gcd_list(minors)
    d = Delta2 // g1

    ans = {
        'P': PP,
        'D': D,
        'c': vector(ZZ,c),
        'B': vector(ZZ,B),
        'g1': g1,
        'Delta2': Delta2,
        'image_step': d,
    }

    if len(c) == 3 and d != 0:
        mp = [m//g1 for m in minors]
        h = vector(ZZ,[mp[2], -mp[1], mp[0]]) // d
        assert vector(ZZ,c)*h == 0
        assert vector(ZZ,B)*h == 0
        ans['common_kernel_generator'] = h

    return ans
