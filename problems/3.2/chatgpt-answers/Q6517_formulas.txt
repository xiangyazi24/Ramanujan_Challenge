P(x) = 34*x^3 + 51*x^2 + 27*x + 5.

(n+1)^3*b[n+1] = P(n)*b[n] - n^3*b[n-1].
(n+1)^3*c[n+1] = P(n)*c[n] - n^3*c[n-1].

b[0] = 1,  b[1] = 5.
c[0] = 0,  c[1] = 1.

W[n] = (n!)^3*c[n].
U[n] = (n!)^3*b[n].

p^3*c[a*p+v] == c[a]*b[v]  (mod p).

W[a*p+v] / p^(3*(a-1))
    == (-1)^a * W[a] * (v!)^3 * b[v]  (mod p).

if p does not divide W[a]*b[v], then
    ord_p(W[a*p+v]) = 3*(a-1).

if p divides b[v], then
    ord_p(W[a*p+v]) >= 3*(a-1) + 1.

K_0(T) = 1.
K_1(T) = P(T)/(T+1)^3.

(T+v+1)^3*K_(v+1)(T)
    = P(T+v)*K_v(T) - (T+v)^3*K_(v-1)(T).

C_v^(p) = p^3*c[p+v].

(p+v+1)^3*C_(v+1)^(p)
    = P(p+v)*C_v^(p) - (p+v)^3*C_(v-1)^(p).

p^3*c[p+1] - (p^3*c[p])*K_1(p)
    = -p^6*c[p-1]/(p+1)^3.

c[p-1] == 0  (mod p).

therefore
p^3*c[p+1] == (p^3*c[p])*K_1(p)  (mod p^7).

p^3*c[p+v] == (p^3*c[p])*K_v(p)  (mod p^7).

A_ent(z)
    = sum_{k=0}^{infinity}
        binom(z,k)^2 * binom(z+k,k)^2

    = 4F3(-z,-z,z+1,z+1; 1,1,1; 1).

A_ent(v) = b[v].

H_0 = 0.
H_m = sum_{j=1}^{m} 1/j.

D[v]
    = sum_{k=0}^{v}
        binom(v,k)^2 * binom(v+k,k)^2
        * (H_(v+k) - H_(v-k)).

A_ent'(v) = 2*D[v].
K_v'(0)   = 2*D[v].

E[v] = coefficient_of(T^2, K_v(T)) = K_v''(0)/2.
F[v] = coefficient_of(T^3, K_v(T)) = K_v'''(0)/6.

K_v(T)
    = b[v] + 2*D[v]*T + E[v]*T^2 + F[v]*T^3 + O(T^4).

K_0(T) = 1.

K_1(T)
    = 5 + 12*T + 0*T^2 - 7*T^3 + O(T^4).

K_2(T)
    = 73 + 210*T + 72*T^2 - (1011/8)*T^3 + O(T^4).

D[0] = 0,   E[0] = 0,  F[0] = 0.
D[1] = 6,   E[1] = 0,  F[1] = -7.
D[2] = 105, E[2] = 72, F[2] = -1011/8.

p^3*c[p] == 1  (mod p^3).

p^3*c[p+v]
    == b[v] + 2*p*D[v] + p^2*E[v]  (mod p^3).

kappa_p
    = ((p^3*c[p] - 1)/p^3)  (mod p).

p^3*c[p]
    = W[p]/((p-1)!)^3.

t/(exp(t)-1) = sum_{m=0}^{infinity} B_m*t^m/m!.

kappa_p == -(2/3)*B_(p-3)  (mod p).

p^3*c[p+v]
    == b[v]
       + 2*p*D[v]
       + p^2*E[v]
       + p^3*(F[v] + kappa_p*b[v])
       (mod p^4).

zeta_p(3) == B_(p-3)/3  (mod p).

kappa_p == -2*zeta_p(3)  (mod p).

p^3*c[p+1]
    == 5 + 12*p + p^3*(-7 + 5*kappa_p)
       (mod p^4).

H_n^(3) = sum_{j=1}^{n} 1/j^3.

T(n,k) = binom(n,k)^2 * binom(n+k,k)^2.

q(n,k)
    = H_n^(3)
      + (1/2)*sum_{m=1}^{k}
          (-1)^(m-1)
          /
          (m^3 * binom(n,m) * binom(n+m,m)).

A_num[n]
    = 6*c[n]
    = sum_{k=0}^{n} T(n,k)*q(n,k).

T(a*p+v, q*p+s)
    == T(a,q)*T(v,s)  (mod p).

if s > v, then
    T(v,s) = 0.

if v+s >= p, then
    binom(v+s,s) == 0  (mod p),
    hence T(v,s) == 0  (mod p).

p^3*q(a*p+v, q*p+s)
    == q(a,q)  (mod p).

p^3*H_(a*p+v)^(3) == H_a^(3)  (mod p).

binom(a*p+v, ell*p) == binom(a,ell)  (mod p).

binom(a*p+v+ell*p, ell*p)
    == binom(a+ell,ell)  (mod p).

p^3*A_num[a*p+v]
    == A_num[a]*b[v]  (mod p).

p^3*c[a*p+v]
    == c[a]*b[v]  (mod p).

b[p+v] == 5*b[v]  (mod p).

p^3*c[p+v] == b[v]  (mod p).

(b[p+v], p^3*c[p+v])
    == b[v]*(5,1)  (mod p).

delta[v]
    = (p^3*c[p+v] - b[v])/p  (mod p).

delta[v] == 2*D[v]  (mod p).

J[v] = (b[v], 2*D[v], E[v], F[v], ...).

row[v] = (1, b[v], D[v], E[v], F[v], c[v]).

collision(r,s)
    := b[r]*c[s] - b[s]*c[r]
    == 0  (mod p).

jet_D(r,s) = b[r]*D[s] - b[s]*D[r].
jet_E(r,s) = b[r]*E[s] - b[s]*E[r].
jet_F(r,s) = b[r]*F[s] - b[s]*F[r].

rate_D
    = #{(r,s): collision(r,s)=0 and jet_D(r,s)=0}
      /
      #{(r,s): collision(r,s)=0}.

rate_E
    = #{(r,s): collision(r,s)=0 and jet_E(r,s)=0}
      /
      #{(r,s): collision(r,s)=0}.

rate_F
    = #{(r,s): collision(r,s)=0 and jet_F(r,s)=0}
      /
      #{(r,s): collision(r,s)=0}.

theta = t*d/dt.

L
    = theta^3
      - t*(2*theta+1)*(17*theta^2+17*theta+5)
      + t^2*(theta+1)^3.

L(B) = 0,  where B(t) = sum_{n=0}^{infinity} b[n]*t^n.
L(C) = t,  where C(t) = sum_{n=0}^{infinity} c[n]*t^n.

FrobOp(y_i(t^p))
    = p^i * sum_{j=0}^{i} alpha_j*y_(i-j)(t).

q(t) = 1 - 34*t + t^2.

Y(t) = (C(t), theta*C(t), theta^2*C(t), t)^T.

theta*Y(t) = G(t)*Y(t).

G(t) =
[
  [ 0,                    1,                     0,                  0 ],
  [ 0,                    0,                     1,                  0 ],
  [ t*(5-t)/q(t),         3*t*(9-t)/q(t),        3*t*(17-t)/q(t),   1/q(t) ],
  [ 0,                    0,                     0,                  1 ]
].

theta(Phi(t))
    = G(t)*Phi(t) - p*Phi(t)*G(t^p).
