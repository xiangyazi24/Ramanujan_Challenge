## 10. Complete reproducible core

The following Python/SymPy code computes the primitive square pair, verifies the interpolation, evaluates P(n), and computes the Smith quotient data. It uses exact integers throughout.

```python
from math import comb, gcd
from functools import reduce
from sympy import Matrix, ilcm


def apery(N):
    A = [0] * (N + 1)
    A[0] = 1
    if N >= 1:
        A[1] = 5
    for m in range(1, N):
        num = (34*m**3 + 51*m**2 + 27*m + 5) * A[m] \
              - m**3 * A[m-1]
        den = (m + 1)**3
        assert num % den == 0
        A[m+1] = num // den
    return A


def difference_table(A, N):
    tab = [A[:N+1]]
    for _ in range(N):
        tab.append([
            tab[-1][i+1] - tab[-1][i]
            for i in range(len(tab[-1]) - 1)
        ])
    return tab


def primitive_vector(v):
    den = 1
    for z in v:
        den = ilcm(den, int(z.q))
    w = [int(z * den) for z in v]
    g = reduce(gcd, (abs(x) for x in w if x), 0)
    w = [x // g for x in w]
    for x in w:
        if x:
            if x < 0:
                w = [-y for y in w]
            break
    return w


def square_pair(n, a, A, DT):
    H = (n - 1) // 3
    b = H - a

    M = Matrix([
        [comb(k, ell) * DT[k-ell][ell]
         for ell in range(b + 1)]
        for k in range(a + 1, H + 1)
    ])

    ns = M.nullspace()
    assert len(ns) == 1
    q = primitive_vector(ns[0])

    # Newton coefficients of P=AQ.
    p = []
    for k in range(a + 1):
        p.append(sum(
            q[ell] * comb(k, ell) * DT[k-ell][ell]
            for ell in range(min(k, b) + 1)
        ))

    # High coefficients vanish exactly.
    for k in range(a + 1, H + 1):
        assert sum(
            q[ell] * comb(k, ell) * DT[k-ell][ell]
            for ell in range(b + 1)
        ) == 0

    Pn = sum(p[k] * comb(n, k) for k in range(a + 1))
    Qn = sum(q[ell] * comb(n, ell) for ell in range(b + 1))

    # Maximal-minor gcd = Smith saturation.
    minors = []
    for ell in range(b + 1):
        cols = [j for j in range(b + 1) if j != ell]
        minors.append(abs(int(M[:, cols].det())))
    delta = reduce(gcd, minors, 0)

    # Appended determinant.
    s = []
    for ell in range(b + 1):
        s.append(sum(
            comb(n, k) * comb(k, ell) * DT[k-ell][ell]
            for k in range(ell, a + 1)
        ))
    Mhat = M.col_join(Matrix([s]))
    det_hat = abs(int(Mhat.det()))
    assert det_hat == delta * abs(Pn)

    # Node verification.
    for j in range(H + 1):
        Pj = sum(p[k] * comb(j, k) for k in range(a + 1))
        Qj = sum(q[ell] * comb(j, ell) for ell in range(b + 1))
        assert Pj == A[j] * Qj

    return p, q, Pn, Qn, delta, det_hat


N = 120
A = apery(N)
DT = difference_table(A, N)
for n in range(4, N + 1):
    H = (n - 1) // 3
    for a in range(H + 1):
        p, q, Pn, Qn, delta, det_hat = square_pair(n, a, A, DT)
        assert Pn != 0
```
