from math import gcd

LIMIT = 5000


def primes_upto(n):
    s = bytearray(b"\x01") * (n + 1)
    s[:2] = b"\x00\x00"
    for q in range(2, int(n**0.5) + 1):
        if s[q]:
            s[q*q:n+1:q] = b"\x00" * (((n-q*q)//q)+1)
    return [q for q in range(3, n+1, 2) if s[q]]


def P(n):
    return 34*n**3 + 51*n**2 + 27*n + 5


def dP(n):
    return 102*n**2 + 102*n + 27


def apery_and_derivative(p):
    p2 = p*p

    # A_m modulo p^2.
    A = [0]*p
    A[0], A[1] = 1, 5
    for m in range(1, p-1):
        num = (P(m)*A[m] - m**3*A[m-1]) % p2
        den = pow(m+1, 3, p2)
        A[m+1] = num * pow(den, -1, p2) % p2

    # D_m = parameter derivative modulo p.
    D = [0]*p
    D[0], D[1] = 0, 12 % p
    for m in range(1, p-1):
        rhs = (
            dP(m)*(A[m] % p)
            + P(m)*D[m]
            - 3*m*m*(A[m-1] % p)
            - m**3*D[m-1]
            - 3*(m+1)**2*(A[m+1] % p)
        ) % p
        D[m+1] = rhs * pow(pow(m+1, 3, p), -1, p) % p

    return A, D


zero_pairs = []
reflection_failures = []
multiple_roots = []
p2_reflection = []
left_p2 = []
right_p2 = []

for p in primes_upto(LIMIT):
    A, D = apery_and_derivative(p)
    p2 = p*p

    for j in range((p-3)//2 + 1):
        jr = p-1-j

        # Exact theorem A_ref = A_j - p D_j mod p^2.
        if A[jr] % p2 != (A[j] - p*D[j]) % p2:
            reflection_failures.append((p,j))

        if A[j] % p == 0:
            qj = (A[j]//p) % p
            qr = (A[jr]//p) % p
            assert qr == (qj-D[j]) % p
            zero_pairs.append((p,j,jr,qj,D[j],qr))

            if D[j] == 0:
                multiple_roots.append((p,j))
            if (A[jr]-A[j]) % p2 == 0:
                p2_reflection.append((p,j))
            if A[j] % p2 == 0:
                left_p2.append((p,j))
            if A[jr] % p2 == 0:
                right_p2.append((p,j))

print("reflection failures:", reflection_failures)
print("zero pairs:", len(zero_pairs))
print("multiple roots:", multiple_roots)
print("p^2 reflection coincidences:", p2_reflection)
print("left p^2 members:", left_p2)
print("right p^2 members:", right_p2)
print("first twenty zero pairs:")
for row in zero_pairs[:20]:
    print(row)
