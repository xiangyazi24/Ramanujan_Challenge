\begin{theorem}[Certificate chain]\label{thm:gh-cert}
For every $2\leq h\leq 30$, the mirror quotient $W_h$ is squarefree and satisfies $W_h(0)\neq 0$. For $h\leq 12$ this follows from exact integer resultants. For $13\leq h\leq 30$ it follows from degree- and content-preserving modular certificates at
\[
\ell\in\{1009,65537,1000003\},
\]
with bad reductions excluded.

Consequently $s_h=4h-4$ (full Morse). Hence
\[
\operatorname{Mon}(f_h)=S_{3h},
\]
and the collision curve $G_h$ is absolutely irreducible over $\mathbb Q$ for all $2\leq h\leq 30$.
\end{theorem}

\begin{corollary}[Frequency-layer collision theorem]\label{cor:l2freq-30}
The [L2-FREQ] frequency-layer estimate holds unconditionally for $h\leq 30$:
\[
\operatorname{Collision}(h)=2p+O(h^2\sqrt p).
\]
\end{corollary}

\begin{proof}[Proof sketch]
The certificate chain is
\[
W_h\ \text{squarefree}
\Longrightarrow
\text{full Morse}
\Longrightarrow
\mathrm{CRIT\text{-}2H}
\Longrightarrow
\text{indecomposability + transposition}
\Longrightarrow
S_{3h}
\Longrightarrow
\text{Pakovich irreducibility criterion}.
\]
This gives absolute irreducibility of the collision curve and the stated frequency estimate.
\end{proof}

\begin{remark}
The quotient $W_h$ has additionally been verified irreducible over $\mathbb Q$ for $h=2,\ldots,7$. The conjecture [W-IRRED] that $W_h$ is irreducible for all $h$ would extend the full chain to every $h$. This is a characteristic-zero statement and contains no quenched quantifier.
\end{remark}

\begin{conjecture}[Midpoint anti-concentration prime form]\label{conj:midpoint-ac-prime}
Let $X=\sqrt N L$ and
\[
G_0=\frac{cN}{X(1+\log X)}.
\]
Then the midpoint sum
\[
\sum_{u}\sum_{\substack{h,d>G_0\\h+d\leq X}}
\mathbf 1[\Delta_p(u-h,h)=0]\mathbf 1[\Delta_p(u,d)=0]
\ll N
\]
holds for nonwrapping configurations, with the mirror/fixed-column skeleton separated.

Together with the near-wall estimate
\[
K_p(H,D)\ll HD(1+\log H),
\]
this closes [MESO-PAIR]. Any power saving
\[
X^{2-\rho}(1+\log X)
\]
with fixed $\rho>0$ is sufficient for $L=N^{o(1)}$.
\end{conjecture}

\begin{remark}[The two faces of the mesoscopic wall]\label{rem:meso-two-faces}
The mesoscopic wall has exactly two faces. The hinge form is [MIDPOINT-AC'] above. The census form is [MESO-S1-2/3]:
\[
S_1(D)\ll N^{2/3}
\]
implies [MESO-PAIR] through the spectral triangle bound
\[
Q_D\leq \frac{\sqrt2}{3}S_1(D)^{3/2}.
\]
Both are strictly weaker than [WALL-zero].
\end{remark}
