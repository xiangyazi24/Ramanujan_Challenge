from sage.all import ZZ, QQ, PolynomialRing, binomial, prod


def C(n, k):
    if k < 0 or k > n:
        return ZZ(0)
    return ZZ(binomial(n, k))


w = {-1: ZZ(1), 0: ZZ(2), 1: ZZ(1)}


def lambda_coeff(v):
    a, b, c = v
    z = ZZ(0)
    if a in (-1, 0) and b in (-1, 0, 1) and c in (-1, 0, 1):
        z += w[b] * w[c]
    if a in (0, 1) and b in (0, 1) and c in (0, 1):
        z += 1
    return z


S = [
    (a, b, c)
    for a in (-1, 0, 1)
    for b in (-1, 0, 1)
    for c in (-1, 0, 1)
    if lambda_coeff((a, b, c)) != 0
]
lam = {v: lambda_coeff(v) for v in S}

assert len(S) == 22
assert sum(lam.values()) == 40
assert lam[(0, 0, 0)] == 5


def coeff(N, A, B, Cc, modulus=None):
    out = ZZ(0)
    for k in range(N + 1):
        z = (
            C(N, k)
            * C(N, A + N - k)
            * C(2 * N - k, B + N - k)
            * C(2 * N - k, Cc + N - k)
        )
        out += z
        if modulus is not None:
            out %= modulus
    return out if modulus is None else out % modulus


def shell(N, m, weighted=False, modulus=None):
    out = ZZ(0)

        z = coeff(N, m * a, m * b, m * c, modulus)
        if weighted:
            z *= lam[v]
        out += z
        if modulus is not None:
            out %= modulus
    return out if modulus is None else out % modulus

    return sum(C(r, k)**2 * C(r + k, k)**2 for k in range(r + 1))

def normalized_pair_residue(n, p, q):
    assert p > q
    N = n - 1
    m = p - 1
    h = p - q
    assert q == m - h + 1
    mod = (p * q)**2

    X = shell(N, m, modulus=mod)
    Y = shell(N, m - h, modulus=mod)
    assert X % p == 0
    assert Y % q == 0

    cross = (p * Y - q * X) % mod
    assert cross % (p * q) == 0

    # This is C_h/(pq) modulo pq.
    return (cross // (p * q)) % (p * q)

R = PolynomialRing(QQ, names=('N', 'm', 'k'))
N, m, k = R.gens()


def X_pair(s):
    if s == 1:
        return k - m, N + m - k + 1
    if s == 0:
        return R(1), R(1)
    if s == -1:
        return N - m - k, k + m + 1
    raise ValueError(s)


def Y_pair(s):
    if s == 1:
        return N - m, N + m - k + 1
    if s == 0:
        return R(1), R(1)
    if s == -1:
        return N - m - k, N + m + 1
    raise ValueError(s)


def ore_polynomials(v):
    a, b, c = v

    Pk = (N-k) * (N+a*m-k) * (N+b*m-k) * (N+c*m-k)
    Qk = (k+1) * (k-a*m+1) * (2*N-k)**2

    pxa, qxa = X_pair(a)
    pyb, qyb = Y_pair(b)
    pyc, qyc = Y_pair(c)
    Pm = pxa * pyb * pyc
    Qm = qxa * qyb * qyc

    PN = (N+1)**2 * (2*N-k+1)**2 * (2*N-k+2)**2
    QN = (
        (N-k+1)
        * (N+a*m-k+1)
        * (N+b*m-k+1)
        * (N-b*m+1)
        * (N+c*m-k+1)
        * (N-c*m+1)
    )

    return {
        'Pk': Pk, 'Qk': Qk,
        'Pm': Pm, 'Qm': Qm,
        'PN': PN, 'QN': QN,
    }


for v in S:
