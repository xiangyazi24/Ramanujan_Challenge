from math import comb


def small_factorials(p):
    fac = [1] * p
    for i in range(1, p):
        fac[i] = fac[i - 1] * i % p
    ifac = [1] * p
    ifac[-1] = pow(fac[-1], p - 2, p)
    for i in range(p - 1, 0, -1):
        ifac[i - 1] = ifac[i] * i % p
    return fac, ifac


def csmall(n, k, p, fac, ifac):
    if k < 0 or k > n:
        return 0
    return fac[n] * ifac[k] % p * ifac[n - k] % p


def lucas_binom(n, k, p, fac, ifac):
    if k < 0 or k > n:
        return 0
    out = 1
    while n or k:
        a, b = n % p, k % p
        if b > a:
            return 0
        out = out * csmall(a, b, p, fac, ifac) % p
        n //= p
        k //= p
    return out


def Lmod(n, k, p, fac, ifac):
    return (lucas_binom(n, k, p, fac, ifac)
            * lucas_binom(n + k, k, p, fac, ifac)) % p


def franel_mod_list(p):
    # F_0,...,F_(p-1); recurrence denominators are units up to this point.
    F = [0] * p
    F[0] = 1
    if p > 1:
        F[1] = 2
    for m in range(1, p - 1):
        rhs = ((7*m*m + 7*m + 2) * F[m]
               + 8*m*m * F[m - 1]) % p
        F[m + 1] = rhs * pow((m + 1) ** 2 % p, p - 2, p) % p
    return F


def apery_mod(j, p, fac, ifac):
    # Folded j satisfies 2*j < p in the q=1 applications.
    return sum(
        csmall(j, k, p, fac, ifac) ** 2
        * csmall(j + k, k, p, fac, ifac) ** 2
        for k in range(j + 1)
    ) % p


def Tcoeff_mod(N, J, d, p, fac, ifac, F):
    # [c^d] T_(N,J)(c), using the exact cutoff coefficient formula.
    if d == 0:
        return sum(Lmod(N, i, p, fac, ifac) * F[i]
                   for i in range(J + 1)) % p

    lo = max(0, J - d + 1)
    hi = min(J, N - d)
    if lo > hi:
        return 0

    ans = 0
    for i in range(lo, hi + 1):
        term = Lmod(N, i + d, p, fac, ifac)
        term *= lucas_binom(i + d, i, p, fac, ifac)
        term *= lucas_binom(d - 1, J - i, p, fac, ifac)
        term *= F[i]
        if (J - i) & 1:
            term = -term
        ans = (ans + term) % p
    return ans


def verify_block_matrix(n, p, t=6):
    assert n // 2 < p <= n
    r = n - p
    j = min(r, p - 1 - r)
    assert 2 * j < p
    assert p > 2 * t

    fac, ifac = small_factorials(p)
    F = franel_mod_list(p)
    Aj = apery_mod(j, p, fac, ifac)

    M = []
    for q in range(1, t + 1):
        N = n + (q - 1) * p       # = q*p + r
        row = []
        for b in range(1, t + 1):
            got = Tcoeff_mod(N, p - 1, b * p,
                             p, fac, ifac, F)
            expected = Aj * Lmod(q, b, p, fac, ifac) % p
            assert got == expected, (n, p, q, b, got, expected)
            if b > q:
                assert got == 0
            row.append(got)
        M.append(row)

    profile_det = 1
    for q in range(1, t + 1):
        profile_det = profile_det * comb(2*q, q) % p

    return {
        "n": n,
        "p": p,
        "j": j,
        "A_j_mod_p": Aj,
        "profile_det_mod_p": profile_det,
        "matrix": M,
    }


def ray_multiplicities(n, p, S=5):
    r = n - p
    out = []
    for s in range(S + 1):
        N = n + s * p
        q = N // p
        assert N % p == r
        out.append((s, N, min(q, p - 1 - q)))
    return out
