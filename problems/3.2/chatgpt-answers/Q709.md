# 12. Reproducible Sage regression code

The proof above is uniform and does not depend on a scan. The following Sage script checks the coefficient formulas, Euclidean reduction, exact large-prime interval, exponent one, the two-branch factorization of D_n, and carrier purification.

```python
from sage.all import ZZ, binomial, gcd, prime_range, prod, factor
from functools import reduce


def franel_list(N):
    F = [ZZ(1)]
    if N == 0:
        return F
    F.append(ZZ(2))
    for m in range(1, N):
        num = (7*m*m + 7*m + 2)*F[m] + 8*m*m*F[m-1]
        den = (m + 1)**2
        assert num % den == 0
        F.append(num // den)
    return F


def apery_list(N):
    A = [ZZ(1)]
    if N == 0:
        return A
    A.append(ZZ(5))
    for m in range(1, N):
        P = 34*m**3 + 51*m**2 + 27*m + 5
        num = P*A[m] - m**3*A[m-1]
        den = (m + 1)**3
        assert num % den == 0
        A.append(num // den)
    return A


def L(n, k):
    return ZZ(binomial(n, k) * binomial(n + k, k))


def content(xs):
    return reduce(gcd, [abs(ZZ(x)) for x in xs], ZZ(0))


def T_coeffs(n, J, F):
    C = [ZZ(0)] * (n + 1)
    C[0] = sum(L(n, i) * F[i] for i in range(J + 1))
    for d in range(1, n + 1):
        lo = max(0, J - d + 1)
        hi = min(J, n - d)
        C[d] = sum(
            (-1)**(J-i)
            * L(n, i+d)
            * binomial(i+d, i)
            * binomial(d-1, J-i)
            * F[i]
            for i in range(lo, hi + 1)
        )
    return C


def kappa(n, J):
    g = ZZ(0)
    for k in range(J, n + 1):
        g = gcd(g, L(n, k) * binomial(k, J))
    return abs(g)


def parameters(n):
    H = (n - 1) // 3
    J0 = H + 1
    U = n - H - 1
    assert J0 == (n + 2) // 3
    assert U == (2*n) // 3
    assert n + J0 == (4*n + 2) // 3
    return H, J0, U


def verify_full(n):
    H, J0, U = parameters(n)
    F = franel_list(U)
    Ap = apery_list(H)

    Gamma = [content(T_coeffs(n, J, F)) for J in range(H, U + 1)]
    kappas = [kappa(n, J) for J in range(J0, U + 1)]

    D_rows = reduce(gcd, Gamma, ZZ(0))
    K = reduce(gcd, kappas, ZZ(0))
    D_euclid = gcd(Gamma[0], K)
    assert D_rows == D_euclid

    # Exact large-prime support and exponent one for K.
    for p in prime_range(2, 2*n + 2):
        if p*p <= 2*n:
            continue
        actual = all(k % p == 0 for k in kappas)
        predicted = (U < p <= n + J0)
        assert actual == predicted, (n, p, actual, predicted)
        if actual:
            assert min(ZZ(k).valuation(p) for k in kappas) == 1

    Pmid = prod(ZZ(p) for p in prime_range(U + 1, n + J0 + 1))
    assert K % Pmid == 0
    small = K // Pmid
    assert all(ZZ(p)**2 <= 2*n for p, e in factor(small))

    # Exact large-prime support of D.
    for p in prime_range(2, 2*n + 2):
        if p*p <= 2*n:
            continue
        j = None
        if U < p <= n:
            j = n - p
        elif n < p <= n + J0:
            j = p - 1 - n
        predicted = (j is not None and Ap[j] % p == 0)
        assert (D_rows % p == 0) == predicted, (n, p, j)

    # Direct carrier purification.
    Bminus = ZZ(binomial(n, J0))
    Pminus = prod(ZZ(p) for p in prime_range(U + 1, n + 1))
    Rminus = prod(
        ZZ(p) for p in prime_range(U + 1, n + 1)
        if Ap[n-p] % p == 0
    )
    assert gcd(D_rows, Pminus) == Rminus

    # The unpurified carrier has the same large direct support.
    G = gcd(D_rows, Bminus)
    for p in prime_range(2, 2*n + 2):
        if p*p > 2*n:
            assert (G % p == 0) == (Rminus % p == 0)

    return D_rows, K, G, Rminus


# Full coefficient/content checks.
for n in range(5, 121):
    verify_full(n)

print("all full checks passed through n=120")

# Hostile local examples without constructing all huge contents.
A = apery_list(2683)

# External q=0 contamination.
assert A[3] % 17 == 0
H, J0, U = parameters(13)
assert 13 < 17 <= 13 + J0

# Direct occurrence of the same zero.
H, J0, U = parameters(20)
assert U < 17 <= 20 and A[20-17] % 17 == 0

# Reflected triple at n=321 is killed because p itself is a support cutoff.
H, J0, U = parameters(321)
for p in [179, 193, 211]:
    assert J0 <= p <= U
    assert kappa(321, p) % p != 0

# Direct triple at n=11576 is retained.
H, J0, U = parameters(11576)
for p, j in [(8893, 2683), (9319, 2257), (11437, 139)]:
    assert p == 11576 - j
    assert j <= H and p > U
    assert A[j] % p == 0

print("hostile examples passed")
```

```plain text
cutoffs H..U
  kill every reflected/lower-quotient large-prime plateau inside n;
  retain exactly the direct q=1 interval inside n;
  also retain an unavoidable q=0 interval above n.
```
