# 11. Reproducible Sage code

The first block computes the cutoff contents exactly for moderate n and checks losses and gains.

```python
from sage.all import ZZ, binomial, gcd, lcm, prod, prime_range


def franel(k):
    return ZZ(sum(binomial(k, a)^3 for a in range(k + 1)))


def L(n, k):
    return ZZ(binomial(n, k) * binomial(n + k, k))


def T_coeffs(n, J):
    """Coefficient vector of T_(n,J)(c), degrees 0..n."""
    F = [franel(i) for i in range(J + 1)]
    C = [ZZ(0)] * (n + 1)

    C[0] = sum(L(n, i) * F[i] for i in range(J + 1))

    for d in range(1, n + 1):
        lo = max(0, J - d + 1)
        hi = min(J, n - d)
        C[d] = sum(
            (-1)^(J - i)
            * L(n, i + d)
            * binomial(i + d, i)
            * binomial(d - 1, J - i)
            * F[i]
            for i in range(lo, hi + 1)
        )
    return C


def content(xs):
    g = ZZ(0)
    for x in xs:
        g = gcd(g, ZZ(x))
    return abs(g)


def Gamma(n, J):
    return content(T_coeffs(n, J))


def kappa(n, J):
    return content([
        L(n, k) * binomial(k, J)
        for k in range(J, n + 1)
    ])


def loss_gain(n, J):
    assert 1 <= J <= n
    gm = Gamma(n, J - 1)
    gp = Gamma(n, J)
    rm = ZZ(gm).radical()
    rp = ZZ(gp).radical()
    common = gcd(rm, rp)
    loss = rm // common
    gain = rp // common

    # Exact transition-content identities.
    kap = kappa(n, J)
    assert gcd(gm, gp) == gcd(gm, kap)
    assert gcd(gm, gp) == gcd(gp, kap)
    return loss, gain


def check_321():
    n = 321
    hits = [(179, 36), (193, 64), (211, 100)]
    for p, j in hits:
        loss, _ = loss_gain(n, p)
        _, gain = loss_gain(n, p + j)
        assert loss % p == 0
        assert gain % p == 0
        print(n, p, j, "loss", p, "gain", p + j)
```

```python
from sage.all import floor, sqrt


def apery_mod(m, p):
    if m == 0:
        return 1 % p
    A0, A1 = 1 % p, 5 % p
    if m == 1:
        return A1
    for k in range(1, m):
        Pk = 34*k^3 + 51*k^2 + 27*k + 5
        den = (k + 1)^3 % p
        A2 = (Pk*A1 - k^3*A0) * inverse_mod(den, p) % p
        A0, A1 = A1, A2
    return A1


def zero_poly_mod(n, J, p):
    return all((x % p) == 0 for x in T_coeffs(n, J))


def check_large_prime_diagonal(n):
    M = n // 2
    for p in prime_range(floor(sqrt(2*n)) + 1, n + 1):
        q, r = divmod(n, p)
        j = min(r, p - 1 - r)
        bad = (apery_mod(j, p) == 0)

        assert q*p > M
        for b in range(1, q + 1):
            J = b*p
            prev_zero = zero_poly_mod(n, J - 1, p)
            cur_zero = zero_poly_mod(n, J, p)
            is_loss = prev_zero and not cur_zero
            assert is_loss == bad, (n, p, b, j)
```

Finally, this computes the minimal diagonal lcm for moderate n and compares its large-prime support with the predicted moving-zero set:

```python
def diagonal_loss_lcm(n):
    M = n // 2
    out = ZZ(1)
    for J in range(M + 1, n + 1):
        loss, _ = loss_gain(n, J)
        out = lcm(out, gcd(J, loss))
    return out


def predicted_large_transition_radical(n):
    B = floor(sqrt(2*n))
    return prod(
        ZZ(p)
        for p in prime_range(B + 1, n + 1)
        if apery_mod(min(n % p, p - 1 - (n % p)), p) == 0
    )


def verify_diagonal_lcm(n):
    D = diagonal_loss_lcm(n)
    B = floor(sqrt(2*n))
    large_part = prod(
        ZZ(p) for p in D.prime_divisors() if p > B
    )
    assert large_part == predicted_large_transition_radical(n)
```
