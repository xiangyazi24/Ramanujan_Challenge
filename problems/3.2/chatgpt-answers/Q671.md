## 8. Exact reproducibility code

The following SageMath code constructs the polynomials H_n, converts them to the binomial basis, verifies the lcm divisibility, verifies integer derivatives, and checks the prime-power Dwork congruence.

```python
from sage.all import *


def franel(m):
    return ZZ(sum(binomial(m, r)^3 for r in range(m + 1)))


def H_polynomials(N):
    R = PolynomialRing(QQ, 'A')
    A = R.gen()
    S = PowerSeriesRing(R, 'x', default_prec=N + 1)
    x = S.gen()

    phi = S(1)
    for m in range(1, N + 1):
        phi += (-1)^m * binomial(A, m) * franel(m) * x^m

    G = (x * phi.derivative() / phi).add_bigoh(N + 1)
    return R, A, [R(0)] + [R(G[n]) for n in range(1, N + 1)]


def newton_coefficient(f, k):
    return ZZ(sum((-1)^(k-j) * binomial(k, j) * f(j)
                  for j in range(k + 1)))


N = 20
R, A, H = H_polynomials(N)

for n in range(1, N + 1):
    coeffs = [newton_coefficient(H[n], k)
              for k in range(H[n].degree() + 1)]
    for k, ck in enumerate(coeffs):
        L = lcm(range(1, k + 1)) if k else 1
        assert ck % L == 0

    dH = H[n].derivative()
    for J in range(0, 30):
        assert dH(J).denominator() == 1

print('binomial-lcm and derivative checks passed through n =', N)


# Exact polynomial P_N(x).
Zx = PolynomialRing(ZZ, 'x')
x = Zx.gen()

def P(N):
    return sum((-1)^m * binomial(N, m) * franel(m) * x^m
               for m in range(N + 1))


def frobenius_x(f, p):
    return Zx({p*i: c for i, c in f.dict().items()})


def check_dwork(p, s, N, M):
    mod = p^s
    lhs = P(N + M*mod) * frobenius_x(P(N // p), p)
    rhs = P(N) * frobenius_x(P(N // p + M*p^(s-1)), p)
    diff = lhs - rhs
    return all(ZZ(c) % mod == 0 for c in diff.coefficients())

for p in [2, 3, 5, 7]:
    for s in [1, 2, 3]:
        for N0 in range(0, 8):
            for M in range(0, 3):
                assert check_dwork(p, s, N0, M)

print('relative Dwork congruences passed')


# Direct check of the stronger difference-ratio statement.
def G_series(N, prec, modulus):
    Rm = Integers(modulus)
    Sm = PowerSeriesRing(Rm, 'z', default_prec=prec)
    z = Sm.gen()
    PN = Sm(sum(Rm((-1)^m * binomial(N, m) * franel(m)) * z^m
                for m in range(N + 1)))
    return (z * PN.derivative() / PN).add_bigoh(prec)

for p in [2, 3, 5, 7]:
    for s in [1, 2, 3]:
        mod = p^s
        for a in range(0, 8):
            for M in range(0, 3):
                b = a + M*mod
                assert G_series(b, 20, mod) == G_series(a, 20, mod)

print('logarithmic difference-ratio checks passed')
```

## Final conclusion

The conjectured integrality is true for all n and all J, with full prime-power control. The decisive input is not an isolated Franel recurrence identity. It is the stronger fact that the cubic-binomial Franel sequence is the constant-term sequence of a Laurent polynomial whose Newton polygon has exactly one interior lattice point. The associated relative Dwork congruence makes the logarithmic derivative locally constant modulo every prime power in the exponent parameter. That is exactly the congruence-preserving property of the polynomials H_n.
