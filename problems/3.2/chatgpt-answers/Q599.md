```python
from sage.all import ZZ, binomial, gcd, prod, prime_range, log


def franel_list(J):
    if J == 0:
        return [ZZ(1)]
    F = [ZZ(1), ZZ(2)]
    for k in range(1, J):
        num = (7*k*k + 7*k + 2)*F[k] + 8*k*k*F[k-1]
        den = (k + 1)**2
        assert num % den == 0
        F.append(num // den)
    return F


def L_list(n):
    L = [ZZ(1)]
    for k in range(0, n):
        num = L[k] * (n-k) * (n+k+1)
        den = (k+1)**2
        assert num % den == 0
        L.append(num // den)
    return L


def q3_coefficients(n):
    J = (n-3)//7
    assert J >= 0
    F = franel_list(J)
    L = L_list(n)
    C = [ZZ(0)]*(n+1)
    C[0] = sum(L[i]*F[i] for i in range(J+1))
    for d in range(1, n+1):
        lo = max(0, J-d+1)
        hi = min(J, n-d)
        s = ZZ(0)
        for i in range(lo, hi+1):
            s += ((-1)**(J-i) * L[i+d]
                  * binomial(i+d, i)
                  * binomial(d-1, J-i)
                  * F[i])
        C[d] = s
    return C


def candidate_primes(n):
    lo = n//4 + 1
    hi = n//3 + 1
    return [ZZ(p) for p in prime_range(lo, hi)
            if 4*p > n and 3*p <= n]


def signed_moment(C, eps, k):
    n = len(C)-1
    s = ZZ(0)
    for d in range(k, n+1):
        sign = 1 if eps == 1 or d % 2 == 0 else -1
        s += sign * binomial(d, k) * C[d]
    return s


def affine_scan(n, eps):
    assert eps in (1, -1)
    C = q3_coefficients(n)
    ps = candidate_primes(n)
    if not ps:
        return None
    pmin = min(ps)
    mmax = pmin - 1
    P = prod(ps)

    D0 = signed_moment(C, eps, 0)
    D1 = signed_moment(C, eps, 1)
    g = abs(D1)
    out = []
    previous_mu = None

    for m in range(1, mmax+1):
        if m > 1:
            # Exact, simple version. For larger n use the modular variant below.
            Dm = signed_moment(C, eps, m)
            g = gcd(g, abs(Dm))

        assert g % P == 0
        r = D0 % g
        mu = g if r == 0 else min(r, g-r)
        assert mu > 0
        if previous_mu is not None:
            assert mu <= previous_mu
        previous_mu = mu
        out.append((m, g, mu))

    return {
        "pmin": pmin,
        "primorial": P,
        "D0": D0,
        "rows": out,
        "final_mu": out[-1][2],
        "final_rate": log(out[-1][2].n())/n,
    }
```

def moment_mod(C, eps, k, modulus):
    if modulus == 1:
        return ZZ(0)
    n = len(C)-1
    s = ZZ(0)
    for d in range(k, n+1):
        sign = 1 if eps == 1 or d % 2 == 0 else -1
        s = (s + sign*(C[d] % modulus)
                   *(binomial(d, k) % modulus)) % modulus
    return s


def affine_scan_modular(n, eps):
    C = q3_coefficients(n)
    ps = candidate_primes(n)
    pmin = min(ps)
    mmax = pmin-1
    P = prod(ps)

    D0 = signed_moment(C, eps, 0)
    g = abs(signed_moment(C, eps, 1))
    rows = []

    for m in range(1, mmax+1):
        if m > 1:
            residue = moment_mod(C, eps, m, g)
            g = gcd(g, residue)
        assert g % P == 0
        r = D0 % g
        mu = g if r == 0 else min(r, g-r)
        rows.append((m, g, mu))

    for a, b in zip(rows, rows[1:]):
        assert b[2] <= a[2]

    return rows
