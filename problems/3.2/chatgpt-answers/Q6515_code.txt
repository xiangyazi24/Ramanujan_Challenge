p = 1009
H = 12
F = GF(p)
R.<X> = PolynomialRing(F)

# 预留所有偶 h 的镜像中心，避免被较早层拿走。
reserved = {(p - 1 - h)//2 for h in range(2, H + 1, 2)}
used = set()
A = {}
P = {}

for h in range(1, H + 1):
    Ah = set()
    if h % 2 == 0:
        c = (p - 1 - h)//2
        assert c not in used
        Ah.add(c)
        used.add(c)

    # 偶 h: 2h+1 个根；奇 h: 2h 个根。都 <= 3h。
    target = 2*h + (1 if h % 2 == 0 else 0)
    for r in range(0, p - h):
        s = p - 1 - h - r
        if r >= s:
            continue
        if r in used or s in used:
            continue
        if r in reserved or s in reserved:
            continue
        Ah.update([r, s])
        used.update([r, s])
        if len(Ah) == target:
            break

    assert len(Ah) == target
    A[h] = Ah
    P[h] = prod(X - F(a) for a in Ah)

    # 镜像函数方程。
    assert P[h](F(p - 1 - h) - X) == (-1)^P[h].degree() * P[h]
    assert P[h].degree() <= 3*h
    assert (P[h].degree() % 2) == (1 if h % 2 == 0 else 0)

# 全部跨层互素/结果式非零。
for h in range(1, H + 1):
    for k in range(h + 1, H + 1):
        assert gcd(P[h], P[k]) == 1
        assert P[h].resultant(P[k]) != 0

S = sum(P[h].degree() for h in range(1, H + 1))
Q = 0                              # 根集两两不交
Pfirst = len(set().union(*A.values()))
assert S == Pfirst and Q == 0
print(S, H^2, S/H)
