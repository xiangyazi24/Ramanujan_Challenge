```python
from sage.all import *

def apery_values(N):
    A = [ZZ(1), ZZ(5)]
    for n in range(1, N):
        P = 34*n^3 + 51*n^2 + 27*n + 5
        A.append((P*A[n] - n^3*A[n-1]) // (n+1)^3)
    return A

def primitive_kernel_vector(M):
    v = M.right_kernel_matrix().row(0)
    den = lcm([x.denominator() for x in v])
    w = vector(ZZ, [den*x for x in v])
    return w // gcd(list(w))

def square_pade(n, a, A):
    H = (n-1)//3
    b = H-a
    M = matrix(QQ, H+1, H+2)
    for j in range(H+1):
        for i in range(a+1):
            M[j,i] = binomial(j,i)
        for k in range(b+1):
            M[j,a+1+k] = -A[j]*binomial(j,k)
    v = primitive_kernel_vector(M)
    Pn = sum(v[i]*binomial(n,i) for i in range(a+1))
    Qn = sum(v[a+1+k]*binomial(n,k) for k in range(b+1))
    assert all(
        sum(v[i]*binomial(j,i) for i in range(a+1))
        == A[j]*sum(v[a+1+k]*binomial(j,k) for k in range(b+1))
        for j in range(H+1)
    )
    return v, ZZ(Pn), ZZ(Qn)
```

```python
def full_pade_ideal(n, A):
    H = (n-1)//3
```
