# 7. Machine-checkable Sage audit

The following script verifies all identities used above at p=13,17,29. It does not expand an integer resultant into p-1 factors; products over the finite field are used only as an independent audit of the symbolic resultant formulas.

```python
from sage.all import *

def apery_row(p):
    F = GF(p)
    b = [F(0)] * p
    b[0] = F(1)
    b[1] = F(5)
    for n in range(1, p - 1):
        Pn = 34*n^3 + 51*n^2 + 27*n + 5
        b[n+1] = (F(Pn)*b[n] - F(n)^3*b[n-1]) / F(n+1)^3
    return b

def dickson_list(F, m):
    RY = PolynomialRing(F, 'Y')
    Y = RY.gen()
    D = [RY(2), Y]
    for j in range(1, m):
        D.append(Y*D[-1] - D[-2])
    return RY, Y, D

for p in [13, 17, 29]:
    F = GF(p)
    N = p - 1
    m = N // 2
    g = F.multiplicative_generator()
    RX = PolynomialRing(F, 'X')
    X = RX.gen()
    RY, Y, D = dickson_list(F, m)

    b = apery_row(p)
    assert all(b[N-r] == b[r] for r in range(N + 1))

    # Full reciprocal and truncated row polynomials.
    B = sum(b[r]*X^r for r in range(N + 1))
    Arow = sum(b[r]*X^r for r in range(N))
    assert B == X^N * B(X=1/X)
    assert X^N * Arow(X=1/X) == Arow + X^N - 1

    Z = [r for r in range(N) if b[r] == 0]
    Rrow = prod(Arow(t) for t in F if t != 0)

    # Exact coefficient/value transposition over F_p.
    c = [Arow(g^j) for j in range(N)]
    C = sum(c[j]*X^j for j in range(N))
    for r in range(N):
        assert C(g^(-r)) == F(N)*b[r]
    Rcorrect = prod(C(g^(-r)) for r in range(N))
    assert Rcorrect == F(N)^N * prod(b[:N])

    # Chebyshev factorization of the written row resultant.
    Q = b[m] + sum(b[m-j]*D[j] for j in range(1, m+1))
    row_plus = (D[m]-2).resultant(Q-1)
    row_minus = (D[m]+2).resultant(Q+1)
    assert Rrow == F((-1)^m) * row_plus * row_minus

    # Chebyshev factorization of the corrected value-circulant.
    S = c[0] + sum(c[j]*D[j] for j in range(1, m))
    d = c[m]
    even_factor = (D[m]-2).resultant(S+d)
    odd_factor = (D[m]+2).resultant(S-d)
    assert even_factor == prod(F(N)*b[r] for r in range(0, N, 2))
    assert odd_factor == prod(F(N)*b[r] for r in range(1, N, 2))
    assert Rcorrect == even_factor * odd_factor

    # CFVZ square / quadratic-square audit.
    q = X^2 - 34*X + 1
    eps = 0 if p % 24 in [1,5,7,11] else 1
    quotient = B // (q^eps)
    fac = quotient.factor()
    assert all(e % 2 == 0 for _, e in fac)
    G = prod(f^(e//2) for f, e in fac)
    assert quotient == G^2

    print({

        'Rcorrect': int(Rcorrect),
        'row_cheb': (int(row_plus), int(row_minus)),
        'correct_parity': (int(even_factor), int(odd_factor)),

        'S': S,
        'd': int(d),
```
