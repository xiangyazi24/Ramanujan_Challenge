# 16. Reproducible exact Sage code

The following code constructs the primitive pairs by signed maximal minors, verifies the Smith formula for every \kappa_b, checks the adjacent polynomial identity, and audits fixed chains. It is written to run through H=50 with exact arithmetic; determinant cost grows quickly, so the HMAX value can be reduced for a first pass.

```python
from sage.all import *
from functools import reduce


def gcd_list(xs):
    xs = [ZZ(abs(x)) for x in xs]
    return reduce(gcd, xs, ZZ(0))


def apery_table(N):
    return [
        ZZ(sum(binomial(m,k)^2 * binomial(m+k,k)^2
               for k in range(m+1)))
        for m in range(N+1)
    ]


def newton_coefficients(A):
    N = len(A)-1
    return [
        ZZ(sum((-1)^(k-j) * binomial(k,j) * A[j]
               for j in range(k+1)))
        for k in range(N+1)
    ]


def multiplication_matrix(H, A):
    c = newton_coefficients(A)
    M = matrix(ZZ, H+1, H+1)
    for k in range(H+1):
        for ell in range(k+1):
            M[k,ell] = binomial(k,ell) * sum(
                binomial(ell,t) * c[k-ell+t]
                for t in range(ell+1)
            )
    return M, c


def signed_maximal_minors(R):
    rows, cols = R.nrows(), R.ncols()
    assert cols == rows + 1
    out = []
    for ell in range(cols):
        keep = [j for j in range(cols) if j != ell]
        out.append((-1)^ell * R.matrix_from_columns(keep).det())
    return out


def primitive_pair(H, b, M):
    a = H-b
    if b == 0:
        q = vector(ZZ, [1])
        delta = ZZ(1)
    else:
        R = M.matrix_from_rows_and_columns(
            list(range(a+1,H+1)), list(range(b+1))
        )
        assert R.rank() == b
        D = signed_maximal_minors(R)
        delta = gcd_list(D)
        q = vector(ZZ, [x//delta for x in D])
        assert gcd_list(q) == 1
        assert R*q == 0

    p = vector(ZZ, [
        sum(M[k,ell]*q[ell] for ell in range(min(k,b)+1))
        for k in range(a+1)
    ])

    tau = M.matrix_from_rows_and_columns(
        list(range(a,H+1)), list(range(b+1))
    ).det()

    assert p[a] == tau//delta
    if b > 0:
        # top denominator coefficient
        tau_prev = M.matrix_from_rows_and_columns(
            list(range(a+1,H+1)), list(range(b))
        ).det()
        assert q[b] == (-1)^b * tau_prev//delta

    return {
        'a': a, 'b': b, 'p': p, 'q': q,
        'delta': delta, 'tau': tau
    }


def binom_poly(x, k):
    if k == 0:
        return x.parent()(1)
    return prod(x-i for i in range(k))/factorial(k)


def poly_from_binomial(coeffs, x):
    return sum(coeffs[k]*binom_poly(x,k)
               for k in range(len(coeffs)))


def eval_binomial(coeffs, n):
    return ZZ(sum(coeffs[k]*binomial(n,k)
                  for k in range(len(coeffs))))


def audit_H(H):
    A = apery_table(H)
    M, c = multiplication_matrix(H, A)
    pairs = [primitive_pair(H,b,M) for b in range(H+1)]

    R = PolynomialRing(QQ, 'x')
    x = R.gen()
    Bx = binom_poly(x,H+1)
    kappas = []

    for b in range(H):
        left = pairs[b]
        right = pairs[b+1]
        Pb = poly_from_binomial(left['p'], x)
        Qb = poly_from_binomial(left['q'], x)
        P1 = poly_from_binomial(right['p'], x)
        Q1 = poly_from_binomial(right['q'], x)

        kappa_lead = (
            binomial(H+1,b+1)
            * left['p'][H-b]
            * right['q'][b+1]
        )
        kappa_smith = (
            (-1)^(b+1) * binomial(H+1,b+1)
            * left['tau']^2
            // (left['delta']*right['delta'])
        )
        assert kappa_lead == kappa_smith
        assert Pb*Q1 - P1*Qb == kappa_smith*Bx
        kappas.append(ZZ(kappa_smith))

    # The explicit b=0,1 edge.
    if H >= 1:
        g = gcd(c[H], H*(c[H]+c[H-1]))
        assert pairs[1]['delta'] == g
        assert pairs[1]['q'][1] == -c[H]//g
        assert kappas[0] == -(H+1)*c[H]^2//g

    return A, M, pairs, kappas


def candidate_audit(H, r=2):
    n = 3*H + 1
    A, M, pairs, kappas = audit_H(H)
    out = []

    for b in range(0,H-r+1):
        Pvals = [eval_binomial(pairs[t]['p'], n)
                 for t in range(b,b+r+1)]
        G = gcd_list(Pvals)
        K = gcd_list(kappas[b:b+r])
        B = ZZ(binomial(n,H+1))
        assert (B*K) % G == 0

        good_pollution = []
        bad_hits = []
        for j in range(H+1):
            p = ZZ(n-j)
            if not p.is_prime():
                continue
            common = all(v % p == 0 for v in Pvals)
            bad = (A[j] % p == 0)
            if bad:
                assert common
                bad_hits.append((j,p))
            elif common:
                assert all(kappas[t] % p == 0
                           for t in range(b,b+r))
                good_pollution.append((j,p))

        out.append({
            'b': b, 'r': r, 'G': G, 'K': K,
            'bad_hits': bad_hits,
            'good_pollution': good_pollution,
        })
    return out


# Exact audit. Start smaller if desired, then raise to 50.
HMAX = 50
for H in range(1,HMAX+1):
    audit_H(H)

# Representative adjacent and triple-chain reports.
for H in [10,20,30,40,50]:
    for row in candidate_audit(H, r=1):
        if row['b'] in [0, H//2, H-1]:
            print('adjacent', H, row)
    if H >= 2:
        for row in candidate_audit(H, r=2):
            if row['b'] in [0, H//2, H-2]:
                print('triple', H, row)
```
