## 10. Reproducible exact code

The following Python/SymPy program reconstructs the rows, cutoff contents, natural minors, Smith factors, and the exact top-minor product formula.

```python
from math import comb, gcd, log
from fractions import Fraction
from functools import reduce
from sympy import Matrix
from sympy.matrices.normalforms import smith_normal_form
from sympy.polys.domains import ZZ


def F(m):
    return sum(comb(m,r)**3 for r in range(m+1))


def L(n,k):
    return comb(n,k)*comb(n+k,k)


def hrow(n,J):
    # h_J(c)=K_c(n,J)g_J(c)
    K=[L(n,J+a)*comb(J+a,J) for a in range(n-J+1)]
    G=[(-1)**b*comb(J,b)*F(J-b) for b in range(J+1)]
    out=[0]*(n+1)
    for a,ka in enumerate(K):
        for b,gb in enumerate(G):
            out[a+b]+=ka*gb
    return out


def content(v):
    g=0
    for x in v:
        g=gcd(g,abs(int(x)))
    return g


def falling(x,k):
    z=1
    for i in range(k):
        z*=x-i
    return z


def top_formula(n,H):
    z=Fraction(comb(2*n,n)**(H+1),1)
    for J in range(H+1):
        z*=comb(n,J)
    for r in range(H+1):
        E=Fraction(0,1)
        for w in range(r+1):
            E+=Fraction(
                comb(r,w)*F(w)*falling(n,r-w),
                falling(2*n,r-w))
        z*=E
    assert z.denominator==1
    return z.numerator


def primes(N):
    ok=[True]*(N+1)
    ok[0]=ok[1]=False
    for p in range(2,int(N**0.5)+1):
        if ok[p]:
            for q in range(p*p,N+1,p):
                ok[q]=False
    return [p for p in range(2,N+1) if ok[p]]


def audit(n):
    H=(n-1)//3
    rows=[hrow(n,J) for J in range(H+1)]

    # Cumulative T_J rows and their contents.
    cur=[0]*(n+1)
    gammas=[]
    for row in rows:
        cur=[x+y for x,y in zip(cur,row)]
        gammas.append(content(cur))

    A=Matrix(rows)
    S=smith_normal_form(A,domain=ZZ)
    diag=[abs(int(S[i,i])) for i in range(H+1)]
    Delta=reduce(lambda x,y:x*y,diag,1)

    top=int(A[:,list(range(n-H,n+1))].det())
    low=int(A[:,list(range(H+1))].det())
    assert abs(top)==abs(top_formula(n,H))

    U1=1
    U2=1
    for p in primes(n):
        if 2*p>n:
            j=min(n-p,2*p-n-1)
            U1*=p**(H-j)
        elif 3*p>n and 2*p<=n:
            j=min(n-2*p,3*p-n-1)
            U2*=p**(H-j)

    assert Delta%(U1*U2)==0

    return {
        "n":n,
        "H":H,
        "gammas":gammas,
        "smith":diag,
        "Delta":Delta,
        "U1":U1,
        "U2":U2,
        "top":top,
        "low":low,
    }


for n in [30,39,45]:
    R=audit(n)
    print("n =",n)
    print("gammas =",R["gammas"])
    print("Smith diagonal =",R["smith"])
    print("log Delta / n^2 =",log(R["Delta"])/n**2)
    print("log abs(top)/(n(H+1)) =",
          log(abs(R["top"]))/(n*(R["H"]+1)))
    print("log abs(low)/(n(H+1)) =",
          log(abs(R["low"]))/(n*(R["H"]+1)))
```
