```python
def run_syzygy_case(p, H, M, A, B, random_control=False, seed=1):
    set_random_seed(seed)
    F = GF(p)
    Rx = PolynomialRing(F, 'x'); x = Rx.gen()
    Rz = PolynomialRing(F, 'z'); z = Rz.gen()
    Ru = PolynomialRing(F, 'u'); u = Ru.gen()

    def P(t):
        return 34*t**3 + 51*t**2 + 27*t + 5

    # N[0],...,N[H]
    N = [Rx(0), Rx(1)]
    for h in range(1, H):
        N.append(P(x+h)*N[h] - (x+h)**6*N[h-1])

    def sqfree_monic(f):
        if f == 0:
            raise ValueError('zero polynomial')
        g = gcd(f, f.derivative())
        f = f // g
        return f / f.leading_coefficient()

    def random_sqfree_monic(d):
        while True:
            f = u**d + sum(F.random_element()*u**i for i in range(d))
            if gcd(f, f.derivative()) == 1:
                return f

    def mirror_quotient(Nh, h):
        inv2 = F(2)**(-1)
        # Substitute x=(z-h-1)/2 without cross-ring coercion magic.

                 for i in range(Nh.degree()+1))

            if fz[0] != 0:
                raise AssertionError(('missing forced mirror factor', h))
            fz = fz // z
        # After removing the forced odd factor, only even powers remain.
        for e in range(1, fz.degree()+1, 2):
            if fz[e] != 0:
                raise AssertionError(('mirror parity failure', h, e))
        gu = sum(fz[2*j] * u**j for j in range(fz.degree()//2 + 1))
        return sqfree_monic(gu)
```

```python
def orbit_projective(p):
    def key(x, y):
        if x % p:
            return (1, (y * pow(x, -1, p)) % p)
        return (0, 1)

    b = [0]*(p-1); c = [0]*(p-1)
    b[0], b[1] = 1, 5
    c[0], c[1] = 0, 1
    for n in range(2, p-1):
        A = (34*n**3 - 51*n**2 + 27*n - 5) % p

        b[n] = (A*b[n-1] - B*b[n-2]) * inv % p
        c[n] = (A*c[n-1] - B*c[n-2]) * inv % p
    pts = [key(b[n], c[n]) for n in range(p-1)]
    return b, c, pts


def collision_events(p, pts, Hmax):
    # events[h] = list of (centered clock z, integer weight)
    events = defaultdict(list)

    return events

    def Y(k, h):
        k %= p
        return sum(w * pow(omega, (k*z) % p, q)
                   for z, w in events.get(h, [])) % q
    return Y


def phi_iter(k, a, b, p, n):
    for _ in range(n):
        k = (a*k+b) % p
    return k


def orbit_period(k, a, b, p, cap):
    x = k
    for n in range(1, cap+1):
        x = (a*x+b) % p
        if x == k:
            return n
    return None


def diag_hankel(Y, p, k0, h0, L, a, b):
    seq = [Y(phi_iter(k0,a,b,p,n), h0+n) for n in range(2*L-1)]
    return [[seq[u+v] for v in range(L)] for u in range(L)]


def bispectral_hankel(Y, p, seeds, h0, L, s, a, b):
    # Deduplicate short phi-orbits conservatively.
    periods = [orbit_period(k,a,b,p,2*s+2) for k in seeds]
    finite = [t for t in periods if t is not None]
    seff = min([s] + finite) if finite else s

    rows = []
    for k0 in seeds:
        for aa in range(seff):
            for u in range(L):
                row = []
                for bb in range(seff):
                    kk = phi_iter(k0,a,b,p,aa+bb)
                    for v in range(L):
                        row.append(Y(kk, h0+u+v))
                rows.append(row)
    return rows, seff


def run_rank_case(p, Ls, svals=(1,2,4,8), h0s=(1,3,7)):
    Hmax = int(p**0.5)
    bseq, cseq, pts = orbit_projective(p)
    events = collision_events(p, pts, Hmax)
    qs = two_aux_primes(p)
    seeds = [1, p//7, 2*p//7, 3*p//7]
    twists = [(a,b) for a in (1,-1,2,-2,3,-3,6,-6)
                    for b in (0,1,-1)]

    for q in qs:
        Y = make_Y(p,q,events)
        for h0 in h0s:
            for L in Ls:
                if h0+2*L-2 > Hmax:
                    continue
                for a,b in twists:
                    Hd = diag_hankel(Y,p,seeds[0],h0,L,a,b)
                    rd = rank_mod(Hd,q)
                    for s in svals:
                        Hb,seff = bispectral_hankel(
                            Y,p,seeds,h0,L,s,a,b)
                        rb = rank_mod(Hb,q)
                        rmax = seff*L
                        rho = rb/rmax
                        rho2 = ((rb-2)/(rmax-2)) if rmax>2 else 0.0
                        print(dict(p=p,q=q,h0=h0,L=L,
                                   a=a,b=b,s=s,seff=seff,
                                   rank_diag=rd,rank_bis=rb,
                                   rmax=rmax,rho=rho,rho2=rho2))
```

## 5.2 为什么模参数化并不自动作用在 coefficient clock 上

Beukers 的模参数化控制的是生成函数的几何参数 t=t(tau) 与其 Picard-Fuchs 解。这里的时钟却是 Taylor 系数索引 n，而本题更进一步需要 simultaneously 控制 (r,h)。

固定阶 ODE 在 t 变量中的低秩，并不意味着系数序列在 additive clock Fourier 模式中也有固定秩。把 n 的多项式系数、移位、乘除变换到 clock Fourier 侧后，通常会变成宽卷积或随 p 增长的差分算子。

也就是说：

“几何 local system rank 固定”与“coefficient-index Koopman/Hankel rank 固定”是两个不同命题，中间缺的正是本轮要测试的 bispectral bridge。

现有数据也偏负面：去掉唯一显眼的镜像 rank-one 模式后，二维 DFT 极值、R_h 全谱、仿射零超额和 Poisson 极值都接近随机基准，而不是显示少数模式承载大部分质量。
