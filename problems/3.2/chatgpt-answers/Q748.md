# 14. Reproducible Sage tests

The first block constructs all cutoff polynomials exactly for moderate (n), computes their contents, loss quotients, the exactly-once radical, the lcm-layer formula, and the Smith determinantal-divisor formula.

```python
from sage.all import (
    ZZ, binomial, gcd, lcm, prod, prime_range, floor, sqrt
)


def radical(a):
    a = ZZ(abs(a))
    if a <= 1:
        return a
    return prod(p for p, _ in a.factor())


def once(a):
    a = ZZ(abs(a))
    r = radical(a)
    return r // gcd(r, a // r)


def vec_content(v):
    g = ZZ(0)
    for x in v:
        g = gcd(g, ZZ(x))
    return abs(g)


def franel_list(N):
    return [
        ZZ(sum(binomial(k, a)^3 for a in range(k + 1)))
        for k in range(N + 1)
    ]


def L_list(n):
    return [
        ZZ(binomial(n, k) * binomial(n + k, k))
        for k in range(n + 1)
    ]


def apery(k):
    return ZZ(sum(
        binomial(k, a)^2 * binomial(k + a, a)^2
        for a in range(k + 1)
    ))


def all_cutoff_contents(n):
    """
    Compute T_(n,J) successively from
      T_J - T_(J-1) = K_(n,J) * g_J.
    Complexity is O(n^3) integer operations; use for moderate n.
    """
    F = franel_list(n)
    Lv = L_list(n)
    T = [ZZ(0)] * (n + 1)
    Gamma = []

    for J in range(n + 1):
        # K_(n,J)(c) coefficient at c^a.
        K = [
            Lv[J + a] * binomial(J + a, J)
            for a in range(n - J + 1)
        ]

        # g_J(c) coefficient at c^b.
        g = [
            (-1)^b * binomial(J, b) * F[J - b]
            for b in range(J + 1)
        ]

        h = [ZZ(0)] * (n + 1)
        for a, ka in enumerate(K):
            for b, gb in enumerate(g):
                h[a + b] += ka * gb

        T = [T[d] + h[d] for d in range(n + 1)]
        Gamma.append(vec_content(T))

    # The full cutoff is the constant Apéry number.
    assert T[0] == apery(n)
    assert all(x == 0 for x in T[1:])
    return Gamma


def loss_data(n):
    Gamma = all_cutoff_contents(n)
    R = [radical(g) for g in Gamma]
    lambdas = []
    counts = {}

    for J in range(1, n + 1):
        lam = R[J - 1] // gcd(R[J - 1], R[J])
        assert lam == radical(lam)
        lambdas.append(lam)
        for p in lam.prime_divisors():
            counts[p] = counts.get(p, 0) + 1

    P = prod(lambdas, ZZ(1))
    O = prod((p for p, e in counts.items() if e == 1), ZZ(1))
    assert O == once(P)

    # Lcm-layer identity Once = L1/L2.
    L1 = lcm(lambdas) if lambdas else ZZ(1)
    pair_gcds = [
        gcd(lambdas[i], lambdas[j])
        for i in range(len(lambdas))
        for j in range(i + 1, len(lambdas))
    ]
    L2 = lcm(pair_gcds) if pair_gcds else ZZ(1)
    assert O == L1 // L2

    # Smith determinantal-divisor identity.
    Dn = prod(lambdas, ZZ(1))
    Dn1 = prod((p^(max(e - 1, 0)) for p, e in counts.items()), ZZ(1))
    Dn2 = prod((p^(max(e - 2, 0)) for p, e in counts.items()), ZZ(1))
    assert O == (Dn * Dn2) // (Dn1^2)

    return Gamma, lambdas, counts, O


def folded_index(n, p):
    r = n % p
    return min(r, p - 1 - r)


def q1_radical(n):
    M = n // 2
    return prod((
        ZZ(p)
        for p in prime_range(M + 1, n + 1)
        if apery(folded_index(n, p)) % p == 0
    ), ZZ(1))


def check_small(n):
    Gamma, lambdas, counts, O = loss_data(n)
    M = n // 2
    Pi = prod((ZZ(p) for p in prime_range(M + 1, n + 1)), ZZ(1))
    R1 = q1_radical(n)
    assert gcd(O, Pi) == R1

    # Verify the large-prime multiplicity theorem on lower-digit bad primes.
    for p in prime_range(floor(sqrt(n)) + 1, n + 1):
        j = folded_index(n, p)
        if apery(j) % p == 0:
            q = n // p
            assert q < p
            assert counts.get(ZZ(p), 0) == min(q, p - 1 - q)

    return {
        'n': n,
        'once': O,
        'q1': R1,
        'small_or_impostor': O // gcd(O, R1),
    }


for n in range(2, 61):
    check_small(n)

print('all exact cutoff checks passed through n=60')
```

```python
from sage.all import ZZ, inverse_mod


def apery_mod(k, p):
    p = ZZ(p)
    if k == 0:
        return ZZ(1) % p
    a0, a1 = ZZ(1), ZZ(5) % p
    if k == 1:
        return a1

    for m in range(1, k):
        Pm = 34*m^3 + 51*m^2 + 27*m + 5
        num = (Pm*a1 - m^3*a0) % p
        den = ((m + 1)^3) % p
        assert den != 0
        a2 = (num * inverse_mod(den, p)) % p
        a0, a1 = a1, a2
    return a1


def check_known(n, primes):
    out = []
    for p in primes:
        q = n // p
        r = n % p
        j = min(r, p - 1 - r)
        assert apery_mod(j, p) == 0
        expected_losses = min(q, p - 1 - q)
        out.append((p, q, r, j, expected_losses))
    return out


print(check_known(321, [179, 193, 211]))
# Expected folded indices: 36, 64, 100; all have q=1 and one loss.

# Expected folded indices: 2683, 2257, 139; all have q=1 and one loss.

print(check_known(258, [17]))
# Expected: (17, 15, 3, 3, 1), the q=p-2 impostor.
```

```python
from sage.all import prime_range


def lcm_upto(N):
    out = ZZ(1)
    for p in prime_range(2, N + 1):
        pp = ZZ(p)
        while pp * p <= N:
            pp *= p
        out *= pp
    return out


def verify_lcm_filter(n):
    _, _, _, O = loss_data(n)
    M = n // 2
    C = lcm_upto(n) // lcm_upto(M)
    X = gcd(O, radical(C))
    R1 = q1_radical(n)

    assert X % R1 == 0
    extra = X // R1
    assert all(p <= sqrt(n) for p in extra.prime_divisors())
    return X, R1, extra
```
