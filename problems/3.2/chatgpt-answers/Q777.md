# 13. Reproducible exact code

The following standard-Python code uses only integer arithmetic. It constructs the cutoff coefficients, the constant Hasse minor, and the finite-difference counterexample. It is provided for reproduction; it was not executed in this connector-only response.

```python
from math import comb, gcd


def C(n: int, k: int) -> int:
    if k < 0 or k > n:
        return 0
    return comb(n, k)


def franel(k: int) -> int:
    return sum(C(k, a) ** 3 for a in range(k + 1))


def apery(k: int) -> int:
    return sum(C(k, a) ** 2 * C(k + a, a) ** 2
               for a in range(k + 1))


def L(N: int, k: int) -> int:
    return C(N, k) * C(N + k, k)


def coeff_T(N: int, J: int, d: int) -> int:
    """Coefficient [c^d] T_(N,J)(c)."""
    if J < 0 or d < 0 or d > N:
        return 0

    if d == 0:
        return sum(L(N, i) * franel(i) for i in range(J + 1))

    lo = max(0, J - d + 1)
    hi = min(J, N - d)
    if lo > hi:
        return 0

    return sum(
        (-1) ** (J - i)
        * L(N, i + d)
        * C(i + d, i)
        * C(d - 1, J - i)
        * franel(i)
        for i in range(lo, hi + 1)
    )


def det_bareiss(A: list[list[int]]) -> int:
    A = [row[:] for row in A]
    n = len(A)
    if n == 0:
        return 1
    sign = 1
    prev = 1

    for k in range(n - 1):
        if A[k][k] == 0:
            pivot = next((r for r in range(k + 1, n)
                          if A[r][k] != 0), None)
            if pivot is None:
                return 0
            A[k], A[pivot] = A[pivot], A[k]
            sign = -sign

        pivot = A[k][k]
        for i in range(k + 1, n):
            for j in range(k + 1, n):
                num = A[i][j] * pivot - A[i][k] * A[k][j]
                assert num % prev == 0
                A[i][j] = num // prev
        prev = pivot

    return sign * A[-1][-1]


def Delta(t: int) -> int:
    return det_bareiss([
        [L(s + 1, a) for a in range(t)]
        for s in range(t)
    ])


def ray_minor(n: int, J: int, t: int) -> int:
    """D_t(n,J) = det [c^(aJ)] T_(n+sJ,J-1)."""
    M = [
        [coeff_T(n + s * J, J - 1, a * J)
         for a in range(t)]
        for s in range(t)
    ]
    return det_bareiss(M)


def vp(x: int, p: int) -> int:
    x = abs(x)
    if x == 0:
        return 10**9
    e = 0
    while x % p == 0:
        x //= p
        e += 1
    return e


def check_target(n: int, p: int, t: int = 2) -> None:
    assert n // 2 < p <= n
    j = min(n - p, 2 * p - 1 - n)
    assert apery(j) % p == 0
    assert p > 2 * t

    D = ray_minor(n, p, t)
    assert D % (p ** t) == 0
    print("target", n, p, j, "v_p(D)=", vp(D, p))


def check_good(n: int, p: int, t: int = 2) -> None:
    assert n // 2 < p <= n
    j = min(n - p, 2 * p - 1 - n)
    assert apery(j) % p != 0
    assert p > 2 * t

    D = ray_minor(n, p, t)
    assert D % p != 0
    assert D % p == (Delta(t) * pow(apery(j), t, p)) % p


def finite_difference_counterexample() -> None:
    n = 6
    p = 5
    J = p

    c0 = coeff_T(n, J - 1, 0)
    c1 = coeff_T(n + J, J - 1, 0)

    assert c0 == 1_188_265
    assert c1 == 159_262_225
    assert vp(c1 - c0, p) == 1

    print("c0 =", c0)
    print("c1 =", c1)
    print("v5(c1-c0) =", vp(c1 - c0, 5))


def regression() -> None:
    assert Delta(1) == 1
    assert Delta(2) == 4
    assert Delta(3) == 60

    finite_difference_counterexample()

    # Known q=1 triples.
    for p in (179, 193, 211):
        check_target(321, p, 2)

    for p in (8893, 9319, 11437):
        check_target(11576, p, 2)


if __name__ == "__main__":
    regression()
```
