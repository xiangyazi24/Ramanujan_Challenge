# 10. Why Q692 does not apply—and what replaces it

Q692 studies determinants of several cutoff rows. If q rows are equal modulo p, a determinant has q-1 universal powers of p; a bad fiber creates only one additional power. This produces a multiplicity-versus-height obstruction.

The present gcd behaves differently:

```plain text
determinant: adds row valuations;
common gcd:  takes the minimum row valuation.
```

Thus the universal multiplicity is removed automatically. There is no determinant of quadratic height and no need to divide a large repeated-row factor.

However, the same rank-one local geometry reappears as a plateau condition. The gcd keeps a prime precisely when one zero plateau covers the entire cutoff interval. The q=0 and reflected q=3 plateaux are as legitimate as the desired q=1 plateau.

# 12. Exact verification code

The following Sage code verifies the consecutive-cutoff identity, computes I_n, and checks the predicted large-prime support.

```python
from sage.all import ZZ, PolynomialRing, binomial, gcd, prime_range


def franel_table(N):
    return [ZZ(sum(binomial(k, a)^3 for a in range(k + 1)))
            for k in range(N + 1)]


def L(n, k):
    if k < 0 or k > n:
        return ZZ(0)
    return ZZ(binomial(n, k) * binomial(n + k, k))


def apery_table(N):
    A = [ZZ(1), ZZ(5)]
    for m in range(1, N):
        Pm = 34*m^3 + 51*m^2 + 27*m + 5
        num = Pm*A[m] - m^3*A[m-1]
        A.append(ZZ(num // (m + 1)^3))
    return A


def content(poly):
    ans = ZZ(0)
    for a in poly.list():
        ans = gcd(ans, ZZ(a))
    return abs(ans)


def g_poly(R, F, J):
    c = R.gen()
    return R(sum((-1)^(J-i) * binomial(J, i) * F[i] * c^(J-i)
                 for i in range(J + 1)))


def K_poly(R, n, J):
    c = R.gen()
    return R(sum(L(n, J+d) * binomial(J+d, J) * c^d
                 for d in range(n-J+1)))


def cutoff_data(n):
    R = PolynomialRing(ZZ, 'c')
    F = franel_table(n)
    T = R(0)
    Gamma = []
    kappa = []

    for J in range(n + 1):
        KJ = K_poly(R, n, J)
        gJ = g_poly(R, F, J)
        hJ = KJ * gJ
        assert content(gJ) == 1
        assert content(hJ) == content(KJ)
        T += hJ
        Gamma.append(content(T))
        kappa.append(content(KJ))

    H = (n - 1) // 3
    M = n // 2
    I = ZZ(0)
    for J in range(H, M + 1):
        I = gcd(I, Gamma[J])

    rhs = Gamma[H]
    for J in range(H + 1, M + 1):
        rhs = gcd(rhs, kappa[J])
    assert I == rhs
    return I, Gamma, kappa


def predicted_large_radical(n, A):
    H = (n - 1) // 3
    cutoff = ZZ(2*n).isqrt()
    R = ZZ(1)
    seen = set()

    # q = 0 branch
    for p in prime_range(n + 1, n + H + 2):
        j = p - 1 - n
        if p > cutoff and A[j] % p == 0:
            seen.add(ZZ(p))

    # q = 1 branch
    for p in prime_range(n // 2 + 1, n + 1):
        r = n - p
        j = min(r, p - 1 - r)
        if p > cutoff and A[j] % p == 0:
            seen.add(ZZ(p))

    # reflected q = 3 branch
    lo = (n + 4) // 4                 # ceil((n+1)/4)
    hi = (4*n + 2) // 15
    for p in prime_range(lo, hi + 1):
        j = 4*p - n - 1
        if p > cutoff and A[j] % p == 0:
            seen.add(ZZ(p))

    for p in seen:
        R *= p
    return R


N = 400
A = apery_table(N)
for n in range(10, N + 1):
    I, Gamma, kappa = cutoff_data(n)
    cutoff = ZZ(2*n).isqrt()

    large_rad = ZZ(1)
    x = I
    for p in prime_range(cutoff + 1, 2*n + 2):
        if x % p == 0:
            large_rad *= p
            while x % p == 0:
                x //= p

    assert large_rad == predicted_large_radical(n, A)

print('all exact cutoff and large-support checks passed')
```
