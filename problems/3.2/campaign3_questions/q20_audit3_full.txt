\subsubsection*{Structure of the difference correspondences}

Write $\delta_h(x)=N_h(x)/\prod_{j=1}^h(x+j)^3$, viewed as a degree-$3h$
map $\mathbb P^1\to\mathbb P^1$, so that $\Delta_{r,h}=\delta_h(r)$ on the
window, and let
$X_{h,k}\subset\mathbb P^1\times\mathbb P^1$ be the closure of
$\{\delta_h(x)=\delta_k(y)\}$, of bidegree $(3h,3k)$.

\begin{lemma}[exact addition law]\label{lem:coinc-addition}
Extend $N_0=0$.  For all $h\ge1$, $d\ge1$,
\begin{align}
 N_{h+d}(x)&=N_d(x+h)\,N_{h+1}(x)-(x+h+1)^6N_{d-1}(x+h+1)\,N_h(x),
 \label{eq:coinc-U1}\\
 \delta_{h+d}(x)&=(x+h+1)^3\bigl(\delta_{h+1}(x)\,\delta_d(x+h)
 -\delta_h(x)\,\delta_{d-1}(x+h+1)\bigr).\label{eq:coinc-D-update}
\end{align}
\end{lemma}

\begin{proof}
In the transfer product $G_{m+g}(s)=G_g(s+m)G_m(s)$ of
Lemma~\ref{lem:oe-transfer-product}, take $s=x+1$, $m=h$, $g=d-1$, and
read off the bottom-right entry: the second row of $G_{d-1}(x+h+1)$ is
$(-N_{d-1}(x+h+1),\,N_d(x+h))$ and the second column of $G_h(x+1)$ is
$((x+h+1)^6N_h(x),\,N_{h+1}(x))^{T}$, giving~\eqref{eq:coinc-U1};
dividing by $\prod_{j=1}^{h+d}(x+j)^3=\prod_{j=1}^{h}(x+j)^3\cdot
\prod_{j=1}^{d}(x+h+j)^3$ gives~\eqref{eq:coinc-D-update}.
\end{proof}

The law expresses a long-gap Casoratian as a bilinear combination of
short-gap Casoratians at the two ends of the block; it is the exact
self-similarity of the family, but it is a coupled identity at one base
point, not a convolution of value distributions.

\begin{lemma}[honest triple poles]\label{lem:coinc-poles}
For $1\le j\le h$,
\[
 N_h(-j)=(-1)^{j-1}\bigl((j-1)!\bigr)^3b_{j-1}\bigl((h-j)!\bigr)^3b_{h-j}
 \ne0 ,
\]
so over $\mathbb Q$ every finite pole of $\delta_h$ has exact order $3$,
and $\delta_h$ vanishes to exact order $3$ at $x=\infty$.
\end{lemma}

\begin{proof}
Apply~\eqref{eq:coinc-U1} with $(h,d)=(j-1+(h-j+1))$ arranged so the
first factor is evaluated at the singular index, or directly: at
$x=-j$ the transfer product $G_{h-1}(x+1)$ factors through the two
regular blocks on either side of the vanishing step, and the two blocks
contribute the two normalized restart values, which
by~\eqref{eq:coinc-gap-identity} are the displayed factorial--Ap\'ery
products; nonvanishing holds because the Ap\'ery numbers are positive
integers.  (Machine-verified exactly for all $1\le j\le h\le10$.)
\end{proof}

\begin{proposition}[generic genus of the difference correspondence]
\label{prop:coinc-genus}
Fix $h\ne k$ and suppose $X_{h,k}$ is irreducible with $\delta_h$,
$\delta_k$ having simple finite critical points, distinct critical
values, and branch loci disjoint from each other away from
$\{0,\infty\}$ (all verified exactly for $2\le h<k\le8$).  Then the
normalization of $X_{h,k}$ has genus
\[
 g=6hk-3h-3k-2 .
\]
In particular the Weil error per correspondence is of order $hk\sqrt p$,
and summing over $h,k\le H$ meets the $O(pH)$ budget
of~\eqref{eq:coinc-hypothesis} only for $H\le p^{1/6}$: pairwise point
counting cannot reach the ranges where
Conjecture~\ref{conj:coinc-beta} gains anything.
\end{proposition}

\begin{proof}
Riemann--Hurwitz on the normalized fiber product: over $\infty$ both
fibers consist of order-$3$ points ($h$ resp.\ $k$ of them, by
Lemma~\ref{lem:coinc-poles}), contributing defect $6hk$; over $0$ the
fibers are $3(h-1)$ resp.\ $3(k-1)$ simple points plus one order-$3$
point at $x=\infty$ resp.\ $y=\infty$, contributing $6(h+k-1)$; each of
the $4h-4$ simple branch points of $\delta_h$ away from $\{0,\infty\}$
contributes $3k$, and symmetrically, by branch-loci disjointness.  Then
$2-2g=2(3h)(3k)-6hk-6(h+k-1)-(4h-4)\cdot3k-(4k-4)\cdot3h$ gives the
displayed $g$.
\end{proof}

\begin{theorem}[component dichotomy under full monodromy]
\label{thm:coinc-components}
Let $\mathrm{MON}_h$ denote the statement that the monodromy group of
$\delta_h:\mathbb P^1\to\mathbb P^1$ over $\overline{\mathbb Q}$ is the
full symmetric group $S_{3h}$.  Assume $\mathrm{MON}_h$ and
$\mathrm{MON}_k$.
\begin{enumerate}
\item If $h\ne k$, then $X_{h,k}$ is irreducible over
$\overline{\mathbb Q}$.
\item If $h=k\ge2$, then $X_{h,h}$ has exactly two irreducible
components over $\overline{\mathbb Q}$: the diagonal $x=y$ and one
further component $H_h$.
\end{enumerate}
Consequently, under $\mathrm{MON}_h$ for all $h\le H$, the total
component excess is $\sum_{h,k\le H}(c_{h,k}-1)=H+O(1)$, exactly the
allowance of Conjecture~\ref{conj:coinc-beta}.
\end{theorem}

\begin{proof}
Components of $X_{h,k}$ over $\overline{\mathbb Q}$ correspond to orbits
of the joint monodromy group $G\le S_{3h}\times S_{3k}$ (the image of
the fundamental group of the common punctured base acting on pairs of
sheets) on $\{1,\dots,3h\}\times\{1,\dots,3k\}$.  By Goursat, $G$ is the
fiber product of the two full symmetric images over a common quotient;
the only proper common quotient of $S_{3h}$ and $S_{3k}$ is
$\mathbb Z/2$, so $G$ contains $A_{3h}\times A_{3k}$.  For $h\ne k$
each alternating factor is transitive, so $G$ is transitive on the
product: one orbit, one component.  For $h=k$ the two projections are
identified along the same cover; $G$ contains the diagonal image of
$A_{3h}$, and $S_{3h}$ acting diagonally on ordered pairs has exactly
two orbits (diagonal and off-diagonal) by $2$-transitivity.
\end{proof}
