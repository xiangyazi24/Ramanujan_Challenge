\begin{proposition}[The strict-triangle Frobenius defect]
\label{prop:gap-kernel-cartier-defect}
Let $p\geq5$ be prime and $1\leq a\leq p-1$.  Put
\[
 M=p-a-1,
 \qquad
 F_{p,a}^{\triangle}(z)=
 \sum_{m=0}^{M}\overline{V_m(a)}z^m\in\mathbb F_p[z],
\]
where the bar is well defined because all recurrence denominators on this
strict triangle are $p$-adic units.  Retain the operators
\[
 \mathscr D_a=(\theta+a)^3-zP(\theta+a)
                   +z^2(\theta+a+1)^3,
 \qquad
 \mathscr L_0=\theta^3-zP(\theta)+z^2(\theta+1)^3.
\]
Then the following identities hold.

\begin{enumerate}
\item The first coefficient beyond the triangle has the integral residue
  \begin{equation}
   p^3V_{p-a}(a)\equiv a^3b_{a-1}\pmod p,
   \label{eq:green-first-Frobenius-boundary}
  \end{equation}
  and the finite polynomial has the exact defect
  \begin{equation}
   \boxed{
   \mathscr D_aF_{p,a}^{\triangle}(z)
    =a^3-a^3b_{a-1}z^{p-a}.}
   \label{eq:green-strict-triangle-defect}
  \end{equation}
  Equivalently,
  \begin{equation}
   \boxed{
   \mathscr L_0\bigl(z^aF_{p,a}^{\triangle}(z)\bigr)
    =a^3z^a-a^3b_{a-1}z^p.}
   \label{eq:green-conjugated-Frobenius-defect}
  \end{equation}
  If
  $\mathcal C_p(\sum_nf_nz^n)=\sum_nf_{pn}z^n$ is the Cartier operator,
  then
  \begin{equation}
   \mathcal C_p\mathscr L_0
    \bigl(z^aF_{p,a}^{\triangle}\bigr)
   =-a^3b_{a-1}z.
   \label{eq:green-Cartier-reads-Hasse}
  \end{equation}

\item The vanishing of the defect on a marked projective fiber is
  tautological.  For $q\in\mathbb F_p$, put $d_n=b_n-qc_n$.  If
  $d_{a-1}=0$, then $d_a\ne0$ and
  \begin{equation}
   \boxed{
   \overline{V_m(a)}=\frac{d_{a+m}}{d_a}}
   \qquad(0\leq m\leq p-a-1).
   \label{eq:green-marked-fiber-tail}
  \end{equation}
  In particular, on the distinguished fiber $b_{a-1}=0$, the truncated
  Green column is just the normalized future Ap\'ery row
  $b_{a+m}/b_a$.

\item Define the full strict-triangle polynomial
  \[
   \Phi_p(s,z)=
   \sum_{a=1}^{p-1}\sum_{m=0}^{p-a-1}
    \overline{V_m(a)}s^az^m
  \]
  and put
  \[
   \Theta=\theta_s+\theta_z,
   \qquad
   \mathscr D_{s,z}^{\triangle}
   =\Theta^3-zP(\Theta)+z^2(\Theta+1)^3.
  \]
  If $H_p(T)=\sum_{r=0}^{p-1}b_rT^r\in\mathbb F_p[T]$, then
  \begin{align}
   \mathscr D_{s,z}^{\triangle}\Phi_p(s,z)
   &={\sum_{a=1}^{p-1}a^3s^a}
     -{\sum_{a=1}^{p-1}a^3b_{a-1}s^az^{p-a}}
     \label{eq:bivariate-strict-triangle-defect}\\
   &={\sum_{a=1}^{p-1}a^3s^a}
     -sz^{p-1}
       \left.(\theta_T+1)^3H_p(T)\right|_{T=s/z}.
     \notag
  \end{align}
  The apparent Laurent expression on the second line is a polynomial:
  the $T^{p-1}$ coefficient is multiplied by $p^3=0$.

\item Coefficient extraction can be made exact, but its frame moves through
  all character indices.  For $0\leq m\leq M$,
  \begin{equation}
   \overline{V_m(a)}
   =-\sum_{t\in\mathbb F_p^\times}
       F_{p,a}^{\triangle}(t)t^{-m},
   \qquad
   \mathcal C_p\bigl(z^{p-m}F_{p,a}^{\triangle}(z)\bigr)
   =\overline{V_m(a)}z.
   \label{eq:green-complete-Mellin-Cartier-extraction}
  \end{equation}
  Moreover, one of the top two coefficients of
  $F_{p,a}^{\triangle}$ is nonzero.  Thus
  \begin{equation}
   \deg F_{p,a}^{\triangle}\geq p-a-2
   \qquad(1\leq a\leq p-2),
   \label{eq:green-truncation-linear-degree}
  \end{equation}
  so the direct polynomial model has degree $\Omega(p)$ whenever
  $a\leq p/2$.
\end{enumerate}
\end{proposition}

\begin{proof}
The companion denominator congruence and the Casoratian formula give
\begin{align*}
 p^3V_{p-a}(a)
 &=a^3\bigl(b_{a-1}(p^3c_p)-c_{a-1}(p^3b_p)\bigr)\\
 &\equiv a^3b_{a-1}\pmod p,
\end{align*}
which proves~\eqref{eq:green-first-Frobenius-boundary}.  Write
$v_m=\overline{V_m(a)}$ for $0\leq m\leq M$ and $v_m=0$ otherwise.
The coefficient of $z^n$ in $\mathscr D_aF_{p,a}^{\triangle}$ is
\begin{equation}
 (a+n)^3v_n-P(a+n-1)v_{n-1}+(a+n-1)^3v_{n-2}.
 \label{eq:green-defect-coefficient}
\end{equation}
It equals $a^3$ at $n=0$ and vanishes for $1\leq n\leq M$ by the
normalized recurrence.  At $n=M+1=p-a$, it is
$-p^3V_{p-a}(a)$, and at $n=M+2$ the only possible term has the factor
$p^3$; all later coefficients vanish.  This proves
\eqref{eq:green-strict-triangle-defect}.  Conjugating by $z^a$ proves
\eqref{eq:green-conjugated-Frobenius-defect}, and Cartier kills $z^a$ but
sends $z^p$ to~$z$, proving~\eqref{eq:green-Cartier-reads-Hasse}.

Replacing $b$ by $d=b-qc$ does not change the Casoratian in
\eqref{eq:green-casoratian}.  If $d_{a-1}=0$, it gives
\[
 V_m(a)=-a^3c_{a-1}d_{a+m}.
\]
The case $m=0$ says $1=-a^3c_{a-1}d_a$, so $d_a\ne0$ and proves
\eqref{eq:green-marked-fiber-tail}.

Multiplying~\eqref{eq:green-strict-triangle-defect} by $s^a$ and summing
over $1\leq a\leq p-1$ replaces $a+\theta_z$ by~$\Theta$ and gives the
first line of~\eqref{eq:bivariate-strict-triangle-defect}.  Setting
$r=a-1$ rewrites its boundary as
\[
 sz^{p-1}\sum_{r=0}^{p-2}(r+1)^3b_r(s/z)^r.
\]
The omitted $r=p-1$ term is zero in~$\mathbb F_p$, proving the second
line.

For the Mellin identity, multiplicative orthogonality gives
\[
 \sum_{t\ne0}t^{k-m}
 =\begin{cases}-1,&k=m,\\0,&k\ne m,
 \end{cases}
 \qquad(0\leq k,m\leq p-2).
\]
For the Cartier identity, the exponents of
$z^{p-m}F_{p,a}^{\triangle}$ lie strictly between~$0$ and~$2p$, and the
unique multiple of~$p$ occurs at the coefficient indexed by~$m$.
Finally, reflection gives
\[
 [z^M]F_{p,a}^{\triangle}=-a^3c_{a-1},
 \qquad
 [z^{M-1}]F_{p,a}^{\triangle}
 =a^3(b_{a-1}-5c_{a-1}).
\]
If the first value vanishes, the second is nonzero because the solution row
$(b_{a-1},c_{a-1})$ never vanishes.  This proves
\eqref{eq:green-truncation-linear-degree}.
\end{proof}

\begin{remark}[Why the natural Frobenius compression is circular]
\label{rem:green-Cartier-circularity}
The rank-four differential compression of the Green kernel is genuine, but
\eqref{eq:green-Cartier-reads-Hasse} shows that its first finite
nonwrapping boundary stores the original Ap\'ery Hasse coefficient.
Likewise,~\eqref{eq:bivariate-strict-triangle-defect} stores the whole Hasse
section on total degree~$p$.  Complete Mellin or Cartier extraction is
exact, but uses a moving frame for every coefficient, while the direct
polynomial degree grows linearly with~$p$.  These identities do not rule
out a new characteristic-dependent mixed crystal; they show that merely
truncating, completing, or applying Cartier to the existing Green equation
does not produce an independent far-bridge condition or a selected-mass
estimate.
\end{remark}
