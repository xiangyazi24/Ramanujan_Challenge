\tag{7.1}

## 7.2 Reverse reciprocity and keep p in a Kloosterman phase

\tag{7.2}

\[

 =(-1)^d
 e\!\left(\frac{d(\ell-1/2)}{pq}\right)

\tag{7.3}
\]

This swaps the inverse variable from `p` to `q`, allowing the first scalar Cauchy to be taken in `q`; the `p,p'` terms then remain inside a fixed `q` square.

However the BC parameter is now the **moving gap** `g`.  The published scalar theorem has one fixed `theta`.  Applying it separately to the `g`-blocks and then recombining is a direct sum and costs the `ell^1`/nuclear sum over `g`, precisely the Q4596 obstruction.

\label{lem:ad-centered-shadow-factorization}

\begin{lemma}[Affine tight frame]
\label{lem:ad-affine-tight-frame}
Let $A\subset\mathbb F_q$, $|A|=Z$, and put
$f=1_A-Z/q$.  For
$g=(\lambda,\beta)\in\operatorname{Aff}_1(\mathbb F_q)$ let

 \sum_g|\rho_gf\rangle\langle\rho_gf|

\[

\begin{proof}

$(q-1)\|f\|_2^2$.  For distinct arguments, sharp two-transitivity of

pairs, giving $-\|f\|_2^2$.  Since
$q\|f\|_2^2=Z(q-Z)$, the operator is $Z(q-Z)P_0$.
The second formula is the corresponding Hilbert--Schmidt identity.
\end{proof}

\begin{proposition}[Mask-robust centered affine coefficient square]
\label{prop:ad-affine-shadow-square}

 u_p\theta_{p,k;q,J}M_{p,k;q,J}

\[
 \sum_{q\asymp N}|u_q|^2\sum_{J\bmod q}|C_q(J)|^2
 \ll_v

\]
In particular, using the canonical first zero moment,
\[
 \boxed{
 \sum_{q\asymp N}|u_q|^2\sum_{J\bmod q}|C_q(J)|^2
 \ll_v N^{7/15+o(1)}.}

\end{proposition}

 C_q^+(J)=\sum_{p,k}|u_p|1_{\mathcal K_q}(k+pJ).
