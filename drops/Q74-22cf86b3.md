\section{The I5 reduction theorem}
\label{sec:i5-reduction}

Let $\mathcal P_X$ denote the set of primes $p$ with $X<p\leq 2X$.  For each
$p\in\mathcal P_X$ let $Z_p\subseteq \mathbf F_p$ be the Ap\'ery zero set.  We
write
\[
 K_X(m)=\#\{p\in\mathcal P_X:m\bmod p\in Z_p\},
 \qquad
 \lambda_X=\frac1{|\mathcal P_X|}\sum_{p\in\mathcal P_X}\frac{|Z_p|}{p}.
\]

For the packing argument we fold the zero set using the involution
$x\mapsto -x$ on residues.

\begin{definition}[Folded and separated zero sets]
\label{def:folded-zero-sets}
For $p\in\mathcal P_X$ define the folded set
\[
Y_p=\{\min(r,p-r):r\in Z_p\}\subseteq[0,p/2].
\]
For an integer parameter $H\geq1$, define
\[
 Z^{\mathrm{iso}}_{p,H}=\{y\in Y_p:
 |y-y'|>H\text{ for all }y'\in Y_p,\ y'\neq y\},
\]
and
\[
 Z^{\mathrm{near}}_{p,H}=Y_p\setminus Z^{\mathrm{iso}}_{p,H}.
\]
The corresponding shell loads are denoted $K_{\mathrm{iso},H}(m)$ and
$K_{\mathrm{near},H}(m)$.
\end{definition}

\begin{proposition}[Packing estimate for isolated folded zeros]
\label{prop:isolated-packing}
For every $H\geq1$,
\[
 \Lambda_{\mathrm{iso}}(X,H)
 \leq \min\left(\lambda_X,\frac1{\log X}+\frac{X}{H\log X}\right),
\]
where
\[
\Lambda_{\mathrm{iso}}(X,H)=
\frac1{|\mathcal P_X|}\sum_{p\in\mathcal P_X}
\frac{|Z^{\mathrm{iso}}_{p,H}|}{p}.
\]
\end{proposition}

\begin{proof}
The first inequality follows immediately from
$Z^{\mathrm{iso}}_{p,H}\subseteq Y_p$ and the definition of $\lambda_X$.
For the second estimate, each isolated element of $Y_p$ excludes an interval
of length $H$ on the folded residue line.  Hence the isolated points occupy
pairwise disjoint intervals.  Splitting primes into
$p\leq H$ and $p>H$, the first contribution is bounded by the trivial prime
sum $\ll1/\log X$, while for $p>H$ the number of possible isolated points is
$\ll p/H$.  Summing over $p\in\mathcal P_X$ gives
\[
\frac1{|\mathcal P_X|}\sum_{p\in\mathcal P_X}
\frac{|Z^{\mathrm{iso}}_{p,H}|}{p}
\ll \frac1{\log X}+\frac{X}{H\log X},
\]
which proves the claim.
\end{proof}

\begin{lemma}[Unconditional near-zero bound]
\label{lem:near-zero-bound}
There is an absolute constant $C$ such that for all $H\geq1$,
\[
 \max_m K_{\mathrm{near},H}(m)\leq C H^2.
\]
\end{lemma}

\begin{proof}
A near zero has another folded zero within distance $H$.  Therefore every
near hit determines a pair of folded zeros at distance at most $H$.  For a
fixed residue representative the number of possible offsets is $O(H)$, and
the folding involution contributes another factor $O(H)$.  The local spacing
property of the Ap\'ery zero sets prevents repeated adjacent choices, so the
number of possible pairs contributing to a fixed $m$ is $O(H^2)$.  Summing the
finite possibilities gives the stated uniform bound.
\end{proof}

\begin{theorem}[I5 reduction]
\label{thm:i5-reduction}
Assume that the I5-tail estimate holds with
\[
 H=X^{7/15}(\log X)^{-1/3},\qquad L=X^{14/15+\eta}.
\]
Then
\[
 \max_m K_X(m)\leq X^{14/15+\eta}
\]
for sufficiently large $X$.
\end{theorem}

\begin{proof}
Decompose
\[
K_X(m)=K_{\mathrm{iso},H}(m)+K_{\mathrm{near},H}(m).
\]
The near contribution is bounded by Lemma~\ref{lem:near-zero-bound}:
\[
K_{\mathrm{near},H}(m)\leq C H^2
=X^{14/15}(\log X)^{-2/3}.
\]
Thus it is smaller than $X^{14/15+\eta}$.

For the isolated contribution, the fifth factorial moment expansion below and
the assumed I5-tail estimate give
\[
(K_{\mathrm{iso},H}(m))_5\ll L^5.
\]
Since $(k)_5\gg k^5$ for $k$ sufficiently large, this implies
$K_{\mathrm{iso},H}(m)\ll L$.  Combining the two estimates proves the result.
\end{proof}

\begin{proposition}[Exact factorial identity]
\label{prop:i5-factorial-identity}
For every integer $m$,
\[
(K_{\mathrm{iso},H}(m))_5
=
\sum_{\substack{p_1,\ldots,p_5\in\mathcal P_X\\
\text{distinct}}}
\prod_{i=1}^5
1_{Z^{\mathrm{iso}}_{p_i,H}}(m\bmod p_i).
\]
\end{proposition}

\begin{proof}
This is the definition of the falling factorial: it counts ordered choices of
five distinct prime hits contributing to $K_{\mathrm{iso},H}(m)$.
\end{proof}

\begin{proposition}[CRT identity with least representative]
\label{prop:crt-least-representative}
For distinct primes $p_1,\ldots,p_5$ and residues
$r_i\in Z^{\mathrm{iso}}_{p_i,H}$, let
$M=p_1\cdots p_5$ and let $m_0\in[0,M)$ be the least CRT representative
satisfying
\[
m_0\equiv r_i\pmod {p_i}\qquad(1\leq i\leq5).
\]
Then the contribution of this ordered prime tuple to the factorial moment is
exactly
\[
\#\{m\leq N:m\equiv m_0\pmod M\}
=\frac NM+O(1).
\]
\end{proposition}

\begin{proof}
The Chinese remainder theorem gives a unique residue class modulo $M$.
Counting integers in this class below $N$ gives the displayed formula.
\end{proof}

\begin{proposition}[Fourier and character formulation]
\label{prop:i5-fourier-formulation}
Let $f_p$ be the indicator of $Z^{\mathrm{iso}}_{p,H}$ on $\mathbf F_p$.
Then the I5 expression may be written using additive Fourier expansion
\[
 f_p(x)=\frac1p\sum_{t\in\mathbf F_p}
\widehat f_p(t)e_p(tx),
\]
where
\[
\widehat f_p(t)=\sum_{x\in\mathbf F_p}f_p(x)e_p(-tx).
\]
After inserting the CRT identity, the zero frequencies give the main term and
the nonzero frequencies are precisely the I5-tail terms.
Equivalently, after multiplicative decomposition, the same terms are expressed
through the corresponding character sums over the prime tuple variables.
\end{proposition}

\begin{proof}
The identity follows by finite Fourier inversion on each residue field.
Multiplying the expansions for five distinct primes and summing over the CRT
classes separates the trivial character contribution from all nontrivial
frequency terms.  No estimate is used in this proposition; only the I5-tail
hypothesis bounds the nonzero-frequency contribution.
\end{proof}
