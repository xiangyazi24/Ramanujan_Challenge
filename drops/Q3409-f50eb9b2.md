# Exact verifier for Q3409.
# No floating point and no Apéry finite scan.

from sage.all import *

# ------------------------------------------------------------------
# Part A. Polynomial-coefficient finite-support antidifference search.
# ------------------------------------------------------------------

def poly_shift_coeffs(d):
    """Coefficients of (a+1)^d in the monomial basis a^k."""
    return [binomial(d, k) for k in range(d + 1)]


def has_poly_solution(h, rhs_scalar, lo, hi, D):
    r"""
    Search

        Q(a) = sum_{i=lo}^hi sum_{d=0}^D c[i,d] a^d w_i

    for

        Q(a+1, shifted by h) - Q(a)
          = rhs_scalar * sum_{j=0}^{h-1} w_j.

    Exact linear algebra over QQ.
    """

    unknowns = [(i, d) for i in range(lo, hi + 1)
                       for d in range(D + 1)]

    # Each column is a sparse vector indexed by (jump_index, a_degree).
    columns = []
    row_keys = set()

    for (i, d) in unknowns:
        col = {}

        # + (a+1)^d w_{i+h}
        for k, ck in enumerate(poly_shift_coeffs(d)):
            key = (i + h, k)
            col[key] = col.get(key, QQ(0)) + QQ(ck)
            row_keys.add(key)

        # - a^d w_i
        key = (i, d)
