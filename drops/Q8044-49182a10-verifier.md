# Q8044 exact verifier

This is the owned Sage worksheet for `Q8044-49182a10.md`. It intentionally
checks only theorem/data interfaces used in that report. It does **not**
manufacture a coefficient-zero-to-Hasse-root mark when the source JSON does
not supply one.

Run from the repository root:

```bash
sage -python - fresh_pairs.json < drops/Q8044-49182a10-verifier.md
```

For ordinary use, copy the fenced Sage block below to
`drops/Q8044-49182a10_verify.sage` and run

```bash
sage drops/Q8044-49182a10_verify.sage path/to/fresh_pairs.json
```

The expected payload is

```json
{
  "records": [
    {
      "output_p": 101,
      "left_triple": [103, 107, 109],
      "right_triple": [113, 127, 131],
      "m": 12345,
      "n": 23456,
      "marked_factors": {
        "101": [1, 3, 1],
        "103": {"factor_index": 0}
      }
    }
  ]
}
```

Polynomial coefficient lists are constant-term first. `n` defaults to `m`.
A marked factor is accepted only if it divides the **actual** `B_p`
reconstructed from the order-two ODE.

```python
from sage.all import (
    GF, QQ, ZZ, RealField, PolynomialRing, binomial, gcd,
    kronecker_symbol, matrix, power_mod, prod
)
import json, sys

RR = RealField(100)

def req(c, msg):
    if not c:
        raise AssertionError(msg)

def apery_row(p):
    F = GF(p)
    b = [F(1), F(5)]
    for n in range(1, p - 1):
        Pn = (2*n + 1) * (17*n*n + 17*n + 5)
        b.append((F(Pn)*b[n] - F(n)^3*b[n-1]) / F(n+1)^3)
    req(len(b) == p, "bad Apéry row length")
    return b

def exact_apery(nmax):
    if nmax == 0:
        return [ZZ(1)]
    b = [ZZ(1), ZZ(5)]
    for n in range(1, nmax):
        Pn = ZZ(2*n + 1) * ZZ(17*n*n + 17*n + 5)
        q, r = (Pn*b[n] - ZZ(n)^3*b[n-1]).quo_rem(ZZ(n+1)^3)
        req(r == 0, "nonintegral Apéry recurrence")
        b.append(q)
    return b

def monic(f):
    req(f != 0, "zero polynomial")
    return f / f.leading_coefficient()

def reciprocal(f):
    R = f.parent(); x = R.gen(); d = f.degree()
    return R(sum(f[i]*x^(d-i) for i in range(d+1)))

def ode_B(p):
    F = GF(p); R = PolynomialRing(F, 'x'); x = R.gen()
    eps = (1 - ZZ(kronecker_symbol(-6, p))) // 2
    d = (p - 1 - 2*eps) // 2
    beta = [F(1)]
    for r in range(d):
        prev = beta[r-1] if r else F(0)
        if eps == 0:
            main, back = 68*r*r + 34*r + 5, (2*r - 1)^2
        else:
            main, back = 68*r*r + 102*r + 39, (2*r + 1)^2
        beta.append((F(2*main)*beta[r] - F(back)*prev) / F(4*(r+1)^2))
    return ZZ(eps), R(sum(beta[i]*x^i for i in range(len(beta))))

def packet(p):
    p = ZZ(p)
    req(p >= 7 and p.is_prime(), "label is not a prime >= 7")
    F = GF(p); R = PolynomialRing(F, 'x'); x = R.gen()
    b = apery_row(p)
    A = R(sum(b[i]*x^i for i in range(p)))
    eps, B = ode_B(p)
    Delta = x^2 - F(34)*x + 1
    req(A == Delta^eps * B^2, "Hasse square identity failed")

    fr = [F(sum(ZZ(binomial(n,k))^3 for k in range(n+1))) for n in range(p)]
    K = R(sum(fr[n]*x^n for n in range(p)))
    rhs = R(sum(b[m]*x^m*(1-F(8)*x)^m*(1+x)^(p-1-m) for m in range(p)))
    req(K^2 == rhs, "Franel pullback failed")

    req(gcd(B, B.derivative()) == 1, "B_p is not squarefree")
    req(power_mod(x, p^2, B) == x % B, "B_p does not divide x^(p^2)-x")
    req(monic(reciprocal(B)) == monic(B), "B_p root set is not reciprocal")

    fs = []
    for f, e in B.factor():
        req(e == 1, "repeated B_p factor")
        f = monic(f)
        req(f.degree() in (1,2), "factor degree exceeds two")
        fp = monic(reciprocal(f))
        req(B % fp == 0, "reciprocal partner missing")
        od = f.degree() if fp == f else f.degree() + fp.degree()
        fs.append((f, ZZ(od)))
    return {"p":p, "row":b, "B":B, "factors":fs}

def digits(n, p):
    n = ZZ(n); ans = []
    if n == 0:
        return [ZZ(0)]
    while n:
        n, r = n.quo_rem(p); ans.append(ZZ(r))
    return ans

def lucas_value(n, P):
    v = GF(P["p"])(1)
    for d in digits(n, P["p"]):
        v *= P["row"][d]
    return v

def parse_factor(spec, P):
    if spec is None:
        return None
    R = P["B"].parent(); x = R.gen()
    if isinstance(spec, list):
        f = R(sum(R.base_ring()(c)*x^i for i,c in enumerate(spec)))
    elif isinstance(spec, dict) and "factor_index" in spec:
        allf = sorted([monic(f) for f,e in P["B"].factor()],
                      key=lambda f:(f.degree(), tuple(ZZ(c) for c in f.list())))
        f = allf[ZZ(spec["factor_index"])]
    else:
        raise AssertionError("bad marked-factor schema")
    f = monic(f)
    req(f.degree() in (1,2), "marked factor degree is not 1/2")
    req(P["B"] % f == 0, "marked factor does not divide actual B_p")
    fp = monic(reciprocal(f))
    req(P["B"] % fp == 0, "marked reciprocal partner missing")
    return f, ZZ(f.degree() if fp == f else f.degree()+fp.degree())

def trace_rank(n, explicit):
    ts = [ZZ(3+i) for i in range(n)]
    dim = ZZ(2)^n
    det0 = prod((t^2-4)^(ZZ(2)^(n-1)) for t in ts)
    out = {"channels":n, "rank":dim, "determinant":str(det0)}
    if explicit:
        G = matrix(QQ, [[2,ts[0]],[ts[0],ts[0]^2-2]])
        for t in ts[1:]:
            G = G.tensor_product(matrix(QQ, [[2,t],[t,t^2-2]]))
        req(G.rank() == dim and G.det() == det0, "trace-pairing rank check failed")
        out["explicit_rank"] = ZZ(G.rank())
    return out

def log_bound(n):
    return RR(n+1).log() + RR(n)*RR(64).log()

req(len(sys.argv) == 2, "usage: sage verifier.sage fresh_pairs.json")
with open(sys.argv[1], 'r', encoding='utf-8') as fh:
    payload = json.load(fh)
records = payload["records"] if isinstance(payload, dict) else payload
req(isinstance(records, list) and records, "no fresh-pair records")

cache = {}; norm = []; maxr = 0
for rec in records:
    p = ZZ(rec["output_p"])
    L = tuple(ZZ(x) for x in rec["left_triple"])
    R = tuple(ZZ(x) for x in rec["right_triple"])
    m = ZZ(rec["m"]); n = ZZ(rec.get("n", m))
    req(len(L)==3 and len(R)==3, "triples must have length three")
    req(len(set((p,)+L+R)) == 7 and set(L).isdisjoint(set(R)),
        "not a seven-label fresh pair")
    for s in (p,)+L+R:
        if s not in cache:
            cache[s] = packet(s)
    indexed = [(p,m),(p,n)] + [(s,m) for s in L] + [(s,n) for s in R]
    for s,u in indexed:
        r = ZZ(u % s); maxr = max(maxr, r)
        req(cache[s]["row"][r] == 0, "not an actual low-digit Apéry hit")
        req(lucas_value(u, cache[s]) == 0, "Lucas/common-index carrier failed")
    norm.append((rec,p,L,R,m,n,indexed))

BZ = exact_apery(ZZ(maxr))
out = {
  "status":"PASS",
  "six_quadratic_rank":trace_rank(6, True),
  "seven_quadratic_rank":trace_rank(7, False),
  "records":[]
}
for rec,p,L,R,m,n,indexed in norm:
    C = prod(BZ[ZZ(u % s)] for s,u in indexed)
    radical = p*prod(L)*prod(R)
    req(C % radical == 0, "seven-label low-index carrier failed")
    marked = rec.get("marked_factors", {})
    fd=[]; fo=[]; pd=None; po=None; missing=[]; report={}
    for s in (p,)+L+R:
        spec = marked.get(str(s), marked.get(int(s)))
        z = parse_factor(spec, cache[s]) if spec is not None else None
        if z is None:
            missing.append(s)
            report[str(s)] = {
              "status":"NO_MARKED_ROOT_BRIDGE",
              "factor_degrees":[ZZ(f.degree()) for f,o in cache[s]["factors"]],
              "orbit_degrees":[o for f,o in cache[s]["factors"]]
            }
        else:
            f,o = z
            report[str(s)] = {"status":"CERTIFIED", "degree":ZZ(f.degree()),
                              "orbit_degree":o, "coeffs":[ZZ(c) for c in f.list()]}
            if s == p: pd,po = ZZ(f.degree()),o
            else: fd.append(ZZ(f.degree())); fo.append(o)
    item = {
      "output_p":p, "left":list(L), "right":list(R), "m":m, "n":n,
      "low_carrier_log_height":RR(abs(C)).log() if abs(C)>1 else RR(0),
      "bm_bn_log_upper":log_bound(m)+log_bound(n),
      "missing_marked_bridge":missing, "factors":report
    }
    if len(fd) == 6:
        item["fresh_factor_rank"] = prod(fd)
        item["fresh_reciprocal_rank"] = prod(fo)
        if pd is not None:
            item["seven_factor_rank"] = pd*prod(fd)
            item["seven_reciprocal_rank"] = po*prod(fo)
    out["records"].append(item)

print(json.dumps(out, indent=2, sort_keys=True, default=str))
```

## What counts as a pass

A record passes only if:

1. the seven labels are distinct and the triples are disjoint;
2. every listed low digit is an actual Apéry zero;
3. the Lucas value at the common integer is zero in every characteristic;
4. the ODE square root, Hasse square, Franel pullback, squarefreeness,
   `B_p | x^(p^2)-x`, and reciprocity all check exactly;
5. the seven-label prime product divides the nonzero low-index integer carrier;
6. every supplied marked factor divides the actual `B_p`;
7. the separated reciprocal-orbit rank is reported rather than silently
   collapsed.

If `marked_factors` is absent, `NO_MARKED_ROOT_BRIDGE` is the intended result,
not a verifier failure. It records the exact coefficient/evaluation gap isolated
in the report.
