\section{Revisit (September 2026): the pointwise top-window problem and its barriers}

\subsection{Reduction and notation}

Let
\[
 b_n=\sum_{k=0}^n \binom{n}{k}^2\binom{n+k}{k}^2
\]
be the Apéry numbers for \(\zeta(3)\), and for a prime \(p\) put
\[
 Z_p=\{0\le r<p:p\mid b_r\}.
\]
For \(q\ge1\) fixed, define the \(q\)-th quotient cell
\[
 I_q(n)=\left\{p\ {\rm prime}:\frac{n}{q+1}<p\le\frac nq\right\}.
\]
For \(p\in I_q(n)\), write
\[
 n=qp+r_p,\qquad 0\le r_p<p,
\]
and set
\[
 N_q(n)=\#\{p\in I_q(n):r_p\in Z_p\},
 \qquad
 M_q(n)=\sum_{\substack{p\in I_q(n)\\r_p\in Z_p}}\log p.
\]
Since \(q\) is fixed and \(p\asymp_q n\) on \(I_q(n)\),
\[
 M_q(n)=(\log n+O_q(1))N_q(n).
\]
Consequently, for every fixed \(q\),
\begin{equation}
 M_q(n)=o(n)
 \quad\Longleftrightarrow\quad
 N_q(n)=o\!\left(\frac{n}{\log n}\right).
 \tag{1}
\end{equation}
Thus the all-fixed-quotient formulation is exactly the family of estimates
\(N_q(n)=o(n/\log n)\), one for each fixed \(q\).

There is an important quantifier point.  The final Problem~3.2 target is
stronger than merely proving (1) separately for every fixed \(q\).  If
\[
 G_n=\gcd(d_na_n,d_nb_n),\qquad
 d_n=\operatorname{lcm}(1,\ldots,n)^3,
\]
the corrected master reduction in the report is
\[
 M(n)=
 \sum_{\sqrt n<p\le n}
   \log p\,
   {\bf 1}_{p\mid b_{\,n\bmod p}},
\]
together with
\begin{equation}
 M(n)\le \log G_n+O(1),
 \qquad
 \log G_n\le 6M(n)+O(n^{2/3}).
 \tag{2}
\end{equation}
Hence \(G_n=e^{o(n)}\) is equivalent to \(M(n)=o(n)\).
The quotient reduction removes
\[
 p\le \frac{n}{f(n)\log n}
\]
for any \(f(n)\to\infty\), leaving quotient classes
\(1\le q<f(n)\log n\).  Therefore the full theorem needs an aggregate or
uniform estimate over this slowly growing range; the fixed-\(q\) statements
in (1) are necessary but, by themselves, are not a uniform summation theorem.

The first quotient is the top window.  Put
\[
 L(n)=\pi(n)-\pi(n/2),
 \qquad
 N_1(n)=
 \#\{n/2<p\le n:p\mid b_{n-p}\}.
\]
The stepping stone used below is
\begin{equation}
 \mathrm{W}:\qquad
 \text{there exists }\delta>0\text{ such that }
 N_1(n)\le(1-\delta)L(n)
 \text{ for all sufficiently large }n.
 \tag{3}
\end{equation}
This is strictly weaker than the \(q=1\) target
\(N_1(n)=o(n/\log n)\).

For an integer \(a\) and \(m\ge1\), write
\[
 e_m(a)=\exp(2\pi i a/m).
\]

\begin{lemma}[Result 1: the one-phase bound]\label{lem:onephase}
Define
\[
 T(n)=\sum_{n/2<p\le n} e_p(b_{n-p}).
\]
Then
\begin{equation}
 N_1(n)\le\frac{L(n)+\operatorname{Re}T(n)}{2}.
 \tag{4}
\end{equation}
In particular, if for some fixed \(\rho<1\),
\[
 \operatorname{Re}T(n)\le \rho L(n)
\]
for all sufficiently large \(n\), then \(\mathrm W\) holds with
\[
 \delta=\frac{1-\rho}{2}.
\]
The stronger hypothesis
\[
 |T(n)|\le(1-\eta)L(n)
\]
implies \(\mathrm W\) with \(\delta=\eta/2\).
\end{lemma}

\begin{proof}
For a bad prime \(p\), namely \(p\mid b_{n-p}\), the corresponding summand is
\(e_p(0)=1\).  For every other prime the real part of the summand is at
least \(-1\).  Hence
\[
 \operatorname{Re}T(n)
 \ge N_1(n)-(L(n)-N_1(n))
 =2N_1(n)-L(n),
\]
which is equivalent to (4).  Substitution of
\(\operatorname{Re}T(n)\le\rho L(n)\) gives
\[
 N_1(n)\le\frac{1+\rho}{2}L(n)
 =\left(1-\frac{1-\rho}{2}\right)L(n).
\]
Finally \(\operatorname{Re}T(n)\le|T(n)|\), so the last assertion is the
special case \(\rho=1-\eta\).
\end{proof}

\begin{proposition}[Result 2: the Malik--Straub conjecture implies \(\mathrm W\)]\label{prop:MS}
Let
\[
 \mathcal E=\{p:Z_p=\varnothing\}
\]
be the set of primes dividing no Apéry number.  Assume Malik--Straub
Conjecture~6.3:
\begin{equation}
 \frac{\#\{p\le x:p\in\mathcal E\}}{\pi(x)}
 \longrightarrow e^{-1/2}.
 \tag{5}
\end{equation}
Then \(\mathrm W\) holds with every fixed
\[
 0<\delta<e^{-1/2}.
\]
\end{proposition}

\begin{proof}
Write \(E(x)=\#\{p\le x:p\in\mathcal E\}\).
From (5) and the prime number theorem,
\[
 E(n)-E(n/2)
 =
 e^{-1/2}\bigl(\pi(n)-\pi(n/2)\bigr)+o(\pi(n)).
\]
Since \(\pi(n)-\pi(n/2)\asymp\pi(n)\),
\[
 \frac{\#(\mathcal E\cap(n/2,n])}{L(n)}
 \longrightarrow e^{-1/2}.
\]
A zero-free prime cannot occur in the set counted by \(N_1(n)\).  Therefore
\[
 N_1(n)
 \le L(n)-\#(\mathcal E\cap(n/2,n]).
\]
Given any \(\delta<e^{-1/2}\), the last display is at most
\((1-\delta)L(n)\) for all sufficiently large \(n\).  This is \(\mathrm W\).
\end{proof}

\begin{remark}
The premise in Proposition~\ref{prop:MS} is conjectural.  Malik and Straub,
\emph{Divisibility properties of sporadic Apéry-like numbers},
Research in Number Theory \textbf{2} (2016), Article~5, Conjecture~6.3,
is the cited source of (5).
\end{remark}

\begin{proposition}[Result 3: the ordinarity barrier]\label{prop:ordinary}
Let
\[
 f(z)=\eta(2z)^4\eta(4z)^4
     =\sum_{m\ge1}\gamma_m q^m,
\]
the normalized weight-\(4\), level-\(8\) newform labelled \(8.4.a.a\).
For every odd prime \(p\), Ahlgren and Ono prove the supercongruence
\begin{equation}
 b_{(p-1)/2}\equiv\gamma_p\pmod{p^2}.
 \tag{6}
\end{equation}
Consequently
\[
 p\in\mathcal E
 \quad\Longrightarrow\quad
 p\nmid\gamma_p.
\]
Thus infinitely many zero-free primes would imply infinitely many ordinary
primes for \(f\); positive density of zero-free primes would imply positive
density of ordinary primes.
\end{proposition}

\begin{proof}
If \(p\in\mathcal E\), then by definition \(p\) divides no Apéry number, so
in particular
\[
 p\nmid b_{(p-1)/2}.
\]
Reducing (6) modulo \(p\) gives
\[
 b_{(p-1)/2}\equiv\gamma_p\pmod p.
\]
Hence \(p\nmid\gamma_p\).  For a normalized eigenform, this is precisely the
definition of ordinarity at \(p\).  Therefore
\[
 \mathcal E\subseteq\{p:f\text{ is ordinary at }p\},
\]
and both the infinitude and positive-density implications follow.
\end{proof}

\begin{remark}
The supercongruence (6) is Theorem~5 of S.~Ahlgren and K.~Ono,
\emph{A Gaussian hypergeometric series evaluation and Apéry number
congruences}, J. reine angew. Math. \textbf{518} (2000), 187--212,
doi:10.1515/crll.2000.004.  The implication above is one-way: it blocks the
zero-free shortcut by tying it to the ordinary-prime problem, but it does
not show that \(\mathrm W\) itself implies any ordinarity statement.
\end{remark}

\begin{proposition}[Result 4: a reflected-pair support model]\label{prop:planted}
Fix \(n\).  For each prime \(p\in(n/2,n]\), put
\[
 r_p=n-p,\qquad s_p=2p-n-1=p-1-r_p.
\]
Omit primes for which one of \(r_p,s_p\) is \(0\) or \(1\), and define
\[
 Z_p^\star=
 \begin{cases}
 \{r_p,s_p\},&r_p\ne s_p,\\
 \{r_p\},&r_p=s_p.
 \end{cases}
\]
Then at most four primes are omitted, and the family \(Z_p^\star\) satisfies
all of the following single-row support laws:
\begin{enumerate}
\item reflection \(r\mapsto p-1-r\);
\item no adjacent zeros;
\item \(|Z_p^\star|\le2\), hence also every global or subinterval support
      bound that is weaker than a constant bound;
\item the ordinary two-zero return law coming from the even-gap factor of
      the gap polynomial.
\end{enumerate}
Nevertheless,
\[
 N_1^\star(n)\ge L(n)-4.
\]
Therefore reflection, no adjacency, single-row support bounds, and the
two-zero return law do not imply \(\mathrm W\).
\end{proposition}

\begin{proof}
The equations \(r_p\in\{0,1\}\) give \(p=n,n-1\), while
\(s_p\in\{0,1\}\) give \(p=(n+1)/2,(n+2)/2\).  Thus at most four primes are
removed.

Reflection is immediate from
\[
 r_p+s_p=p-1.
\]
If \(r_p\ne s_p\), their distance
\[
 h=|r_p-s_p|
\]
is even, because \(p-1\) and \(2r_p\) have the same parity; hence \(h\ge2\),
which proves no adjacency.

To record the return law without confusing the gap polynomial with the
count \(N_q(n)\), write \(N_h^{\rm gap}(X)\) for the report's gap polynomial.
Its reflection symmetry is
\[
 N_h^{\rm gap}(-X-h-1)=(-1)^{h-1}N_h^{\rm gap}(X).
\]
For even \(h\), the polynomial is odd about the fixed point
\(-(h+1)/2\), and therefore
\begin{equation}
 N_h^{\rm gap}(X)=(2X+h+1)M_h(X)
 \tag{7}
\end{equation}
for an integral polynomial \(M_h\).  If
\(x=\min(r_p,s_p)\), then the reflected pair gives
\[
 2x+h+1=p.
\]
Thus (7) implies
\[
 p\mid N_h^{\rm gap}(x),
\]
which is exactly the support-visible two-zero return condition.

Finally \(r_p=n-p\in Z_p^\star\) for every retained prime, so every retained
top-window prime is bad in the planted model.  Hence
\[
 N_1^\star(n)\ge L(n)-4.
\]
This violates every fixed positive saving for large \(n\).

The construction is only a support model.  It is not asserted to be the
actual Apéry row, and it does not encode value-level fibre or energy laws,
a mod-\(p^2\) reflection law, or compatibility of the distinguished initial
Apéry vector across characteristics.  Those are outside the scope of the
no-go statement.
\end{proof}

\begin{lemma}[Result 5: zero-jet canonicity of index interpolation]\label{lem:canonicity}
Fix \(q\ge1\) and \(n\), and put
\[
 K=\left\lceil\frac{n}{q+1}\right\rceil-1,
 \qquad
 B_K(X)=\sum_{k=0}^K
 \binom{X}{k}^2\binom{X+k}{k}^2.
\]
Let
\[
 P_{n,q}=\prod_{p\in I_q(n)}p.
\]
If \(G(X)\in\mathbb Q[X]\) has coefficients that are \(p\)-integral for
every \(p\in I_q(n)\), and if
\[
 G(r)=b_r\qquad(0\le r\le K),
\]
then for every \(p\in I_q(n)\),
\begin{equation}
 G(n)\equiv B_K(n)\pmod p.
 \tag{8}
\end{equation}
Equivalently, after writing
\(G(n)-B_K(n)=A/D\) with \(\gcd(D,P_{n,q})=1\), one has
\[
 P_{n,q}\mid A.
\]
Moreover, for \(p\in I_q(n)\),
\begin{equation}
 b_n\equiv b_q B_K(n)\pmod p.
 \tag{9}
\end{equation}
\end{lemma}

\begin{proof}
Take \(p\in I_q(n)\), and write \(n=qp+r\).
Since \(p>n/(q+1)\), one has \(p>K\).  The same strict inequality also
implies \(0\le r\le K\): if \(n/(q+1)\) is not an integer this follows from
\(r<n/(q+1)<K+1\), while in the integral endpoint case the strict inequality
forces \(p\ge n/(q+1)+1\), again giving \(r\le K\).

Because \(p>K\), all denominators occurring in the polynomial binomial
coefficients in \(B_K(X)\) are units modulo \(p\).  Since \(n\equiv r\pmod p\),
\[
 B_K(n)\equiv B_K(r)\pmod p.
\]
Terms with \(k>r\) vanish at the integer \(r\), so
\[
 B_K(r)=\sum_{k=0}^r
 \binom{r}{k}^2\binom{r+k}{k}^2=b_r.
\]
Similarly, \(G\) is \(p\)-integral and \(G(r)=b_r\), hence
\[
 G(n)\equiv G(r)=b_r\equiv B_K(n)\pmod p.
\]
This proves (8) for every window prime, and Chinese remaindering gives the
statement modulo \(P_{n,q}\) after clearing the denominator \(D\).

Finally the Apéry Lucas congruence gives
\[
 b_{qp+r}\equiv b_qb_r\pmod p.
\]
Combining it with \(b_r\equiv B_K(n)\pmod p\) yields (9).
\end{proof}

\begin{remark}[Exact scope of Result 5]
Lemma~\ref{lem:canonicity} controls only the mod-\(p\) \emph{zero-jet} of a
\(p\)-integral index evaluation.  It does not control a divided difference
\[
 \frac{G(n)-G(n-p)}p,
\]
companion-vector information such as \((a_r,b_r)\), normalizations with a
\(p\)-singular denominator, or operations that combine different primes.
Thus it rules out a free gain from ordinary index interpolation, but it is
not a general no-go theorem for \(p\)-adic or cross-characteristic
constructions.
\end{remark}

\begin{proposition}[Result 6: one integral carrier and the short-triple reduction]\label{prop:carrier}
Let
\[
 F=(1-x-y)(1-z-w)-xyzw,
 \qquad
 C_{s,m}=[x^sy^sz^sw^s](xyzw\,F)^m,
\]
and, for fixed \(n\), define the integral polynomial
\[
 H_n(T)=\sum_{m\ge0}C_{n-1,m}T^m.
\]
For every prime \(p\in(n/2,n]\),
\begin{equation}
 p\mid b_{n-p}
 \quad\Longleftrightarrow\quad
 p\mid[T^{p-1}]H_n(T).
 \tag{10}
\end{equation}

For \(h,k\ge1\), let \(\mathcal C_n(h,k)\) be the number of primes \(p\)
such that
\[
 p,\quad p-h,\quad p-h-k
\]
all lie in \((n/2,n]\), are prime, and are all bad for the same \(n\):
\[
 p\mid b_{n-p},\qquad
 p-h\mid b_{n-p+h},\qquad
 p-h-k\mid b_{n-p+h+k}.
\]
If, for every fixed \(A>0\),
\begin{equation}
 \max_{1\le h,k\le A\log n}\mathcal C_n(h,k)
 =
 o_A\!\left(\frac{n}{\log^3 n}\right),
 \tag{11}
\end{equation}
then
\[
 N_1(n)=o\!\left(\frac{n}{\log n}\right).
\]
\end{proposition}

\begin{proof}
The previously proved diagonal congruence is
\[
 b_r\equiv[x^ry^rz^rw^r]F^{p-1}\pmod p
 \qquad(0\le r<p).
\]
For \(p\in(n/2,n]\), set
\[
 m=p-1,\qquad r=n-p=n-1-m.
\]
Since
\[
 (xyzw\,F)^m=(xyzw)^mF^m,
\]
we obtain
\[
 C_{n-1,p-1}
 =
 [x^ry^rz^rw^r]F^{p-1}
 \equiv b_r=b_{n-p}\pmod p,
\]
which proves (10).

For completeness, expanding \(F^m\) by first choosing \(j\) copies of the
term \(-xyzw\) gives the exact one-sum form
\[
 [x^ry^rz^rw^r]F^m
 =
 \sum_{j=\max(0,\,2r-m)}^r
 (-1)^j\binom mj
 \left(
 \frac{(m-j)!}
 {(r-j)!^2(m-2r+j)!}
 \right)^2.
\]
Thus the carrier is genuinely one characteristic-zero hypergeometric array;
no equality over \(\mathbb Z\) between its column and \(b_r\) is being
asserted beyond the congruence modulo the defining prime.

Now suppose, toward a contradiction to the conclusion, that for some
\(\varepsilon>0\) and infinitely many \(n\),
\[
 M:=N_1(n)\ge\varepsilon\frac{n}{\log n}.
 \tag{12}
\]
Order the corresponding residues as
\[
 0\le r_1<\cdots<r_M<n/2,
\]
and put
\[
 H=\left\lceil\frac{4\log n}{\varepsilon}\right\rceil.
\]
The sum of all adjacent gaps is \(<n/2\).  Hence fewer than
\[
 \frac{n}{2H}\le\frac M8
\]
adjacent gaps exceed \(H\), for all sufficiently large \(n\).
A long adjacent gap spoils at most two consecutive triples
\((r_i,r_{i+1},r_{i+2})\).  Therefore at least \(M/2\) consecutive triples
have both gaps at most \(H\).

There are at most \(H^2\) ordered gap pairs.  Some
\((h,k)\), with \(1\le h,k\le H\), therefore occurs at least
\[
 \frac{M}{2H^2}
 \gg_\varepsilon\frac{n}{\log^3 n}
 \tag{13}
\]
times.  Replacing \(r_i,r_{i+1},r_{i+2}\) by the decreasing primes
\(p=n-r_i,p-h,p-h-k\), every occurrence counted in (13) contributes to
\(\mathcal C_n(h,k)\).  Taking
\(A>4/\varepsilon\), (13) contradicts (11).  Since \(\varepsilon\) was
arbitrary, \(N_1(n)=o(n/\log n)\).
\end{proof}

\begin{remark}[Hardy--Littlewood caution]
The scale in (11) should not be mistaken for a sparse-prime gain by itself.
Hardy--Littlewood heuristics predict order \(n/\log^3n\) prime triples for
admissible short offsets.  Thus (11) asks for an \(o(1)\) proportion of the
available short prime triples to be simultaneously bad.  The gain of the
reduction is algebraic proximity of the three columns, not an automatic
prime-counting saving.
\end{remark}

\begin{proposition}[Result 7: the sharpened exceptional-set bound]\label{prop:exceptional}
For every fixed \(0<\varepsilon\le1\),
\begin{equation}
 \#\{n\le X:\log G_n>\varepsilon n\}
 \ll
 \varepsilon^{-3}(\log X)^2+O_\varepsilon(1).
 \tag{14}
\end{equation}
More precisely, there are absolute constants \(c,C>0\) such that every
interval
\[
 J\subset(N,2N],
 \qquad
 |J|=c\varepsilon^2\frac{N}{\log N},
\]
contains at most \(C/\varepsilon\) exceptional integers, once \(N\) is
sufficiently large in terms of \(\varepsilon\).
\end{proposition}

\begin{proof}
For \(n\in(N,2N]\), write
\[
 \mathcal P(n)=
 \{p>\sqrt n:n\bmod p\in Z_p\},
 \qquad
 t(n)=|\mathcal P(n)|.
\]
The corrected valuation bound and the companion-height estimate already
proved in the report imply that there is an absolute \(c_0>0\) such that
\begin{equation}
 \log G_n>\varepsilon n
 \quad\Longrightarrow\quad
 t(n)\ge
 T:=c_0\varepsilon\frac{N}{\log N}
 \tag{15}
\end{equation}
for all sufficiently large \(N\).

All prime labels lie in \((\sqrt N,2N]\).  Hence, for an absolute \(C_L\),
their universe has size
\[
 L_N\le C_L\frac{N}{\log N}.
 \tag{16}
\]
For distinct \(m<n\) in the block, with \(h=n-m\), the repaired codegree
argument gives
\begin{equation}
 |\mathcal P(m)\cap\mathcal P(n)|\le C_{\rm cd}h
 \tag{17}
\end{equation}
for an absolute \(C_{\rm cd}\).  Indeed, primes \(p\le h\) are counted
trivially.  For \(p>h\), a no-wrap common hit forces \(p\) to divide the
nonzero gap carrier \(N_h^{\rm gap}(m)\), while a wrap forces \(p\) to
divide a product of \(h\) consecutive integers.  Both integers have
logarithmic height \(O(h\log N)\).  Since every eligible prime has
\(\log p\gg\log N\), each carrier has only \(O(h)\) eligible prime divisors,
which proves (17).

Take an interval \(J\) of length
\[
 Y=c\varepsilon^2\frac{N}{\log N}
\]
and let \(n_1,\ldots,n_M\) be its exceptional integers.  For each prime let
\[
 d_p=\#\{j:p\in\mathcal P(n_j)\},
 \qquad
 I=\sum_pd_p.
\]
By (15),
\[
 I\ge TM.
 \tag{18}
\]
Cauchy--Schwarz gives
\begin{equation}
 \sum_p\binom{d_p}{2}
 =
 \frac12\left(\sum_pd_p^2-I\right)
 \ge
 \frac12\left(\frac{I^2}{L_N}-I\right).
 \tag{19}
\end{equation}
On the other hand, summing (17) over pairs of exceptional integers gives
\begin{equation}
 \sum_p\binom{d_p}{2}
 \le
 C_{\rm cd}
 \sum_{i<j}|n_i-n_j|
 \le
 C_{\rm cd}YM^2.
 \tag{20}
\end{equation}

Choose the absolute constant \(c\) so small that
\[
 C_{\rm cd}Y\le\frac18\,\frac{T^2}{L_N}.
 \tag{21}
\]
This is possible because (16) implies
\[
 \frac{T^2}{L_N}
 \ge
 \frac{c_0^2}{C_L}
 \varepsilon^2\frac{N}{\log N}.
\]

Assume \(M>2L_N/T\).  Then \(TM>2L_N\), and the function
\(x^2/L_N-x\) is increasing throughout the range containing \(I\ge TM\).
Thus (19) may be evaluated at the lower endpoint \(TM\):
\[
 \sum_p\binom{d_p}{2}
 \ge
 \frac12\left(\frac{T^2M^2}{L_N}-TM\right).
\]
Combining this with (20)--(21) yields
\[
 \frac38\frac{T^2M^2}{L_N}\le\frac12TM,
\]
hence
\[
 M\le\frac{4L_N}{3T},
\]
contradicting \(M>2L_N/T\).  Therefore
\[
 M\le\frac{2L_N}{T}
 \le
 \frac{2C_L}{c_0}\,\varepsilon^{-1}.
 \tag{22}
\]

There are
\[
 O\!\left(\frac{\log N}{\varepsilon^2}\right)
\]
critical intervals in \((N,2N]\).  By (22), the number of exceptions in
one dyadic block is \(O(\varepsilon^{-3}\log N)\).  Summing over the
\(O(\log X)\) dyadic blocks up to \(X\) gives (14); the finitely many blocks
below the threshold \(N_0(\varepsilon)\) contribute \(O_\varepsilon(1)\).
\end{proof}

\begin{proposition}[Result 8: CRT--Lucas factorization and the terminal-prime obstruction]\label{prop:crt}
Let \(d\ne m\) be primes with \(dm\le n\).  Let
\(\overline m\) be the inverse of \(m\) modulo \(d\), and
\(\overline d\) the inverse of \(d\) modulo \(m\).  Then
\begin{align}
 e_{dm}(b_{n-dm})
 &=
 e_d\!\left(
   \overline m\,
   b_{\,n\bmod d}\,
   b_{\lfloor n/d\rfloor-m}
 \right)
 \nonumber\\
 &\qquad\times
 e_m\!\left(
   \overline d\,
   b_{\,n\bmod m}\,
   b_{\lfloor n/m\rfloor-d}
 \right).
 \tag{23}
\end{align}
This exposes a fixed-modulus factor after one variable is frozen, but an
exact Buchstab/Harman-type decomposition based only on proper factors
cannot deduce cancellation in \(T(n)\): the terminal sector consisting of
integers with exactly one prime factor is the original prime sum \(T(n)\).
\end{proposition}

\begin{proof}
For every integer \(A\), Chinese remaindering gives
\[
 e_{dm}(A)=e_d(\overline m A)e_m(\overline d A).
 \tag{24}
\]
Indeed,
\[
 m\overline m+d\overline d\equiv1\pmod{dm},
\]
so the two exponents in the right side of (24) sum to \(A/(dm)\) modulo
\(\mathbb Z\).

Write
\[
 n=d\left\lfloor\frac nd\right\rfloor+(n\bmod d).
\]
Since \(dm\le n\),
\[
 n-dm
 =
 d\left(\left\lfloor\frac nd\right\rfloor-m\right)+(n\bmod d).
\]
The Apéry Lucas congruence therefore gives
\[
 b_{n-dm}
 \equiv
 b_{\,n\bmod d}\,
 b_{\lfloor n/d\rfloor-m}
 \pmod d.
\]
The analogous congruence modulo \(m\) is
\[
 b_{n-dm}
 \equiv
 b_{\,n\bmod m}\,
 b_{\lfloor n/m\rfloor-d}
 \pmod m.
\]
Substitution into (24) proves (23).

For the second assertion, a proper-factor decomposition can freeze a modulus
on a composite integer by selecting a proper prime divisor.  A prime has no
proper nontrivial divisor.  Hence any exact decomposition of a function
which agrees with \(e_p(b_{n-p})\) on primes contains a terminal
\(\Omega=1\) sector.  Its restriction is
\[
 \sum_{n/2<p\le n}e_p(b_{n-p})=T(n).
\]
Estimates which only control sectors possessing a proper factor therefore
leave the original sum untouched.  To obtain \(T(n)=o(L(n))\) one needs
either direct information on this terminal sector or genuinely stronger
integer Type~I/II cancellation that already includes it.
\end{proof}

\begin{proposition}[Result 9: the Jacobi-sum lift and the fixed-field obstruction]\label{prop:jacobi}
Let \(p\) be an odd prime, let \(\omega\) be a generator of the character
group of \(\mathbb F_p^\times\), and let
\[
 J(\chi,\psi)=\sum_{x\in\mathbb F_p}\chi(x)\psi(1-x)
\]
be the Jacobi sum.  Define
\begin{equation}
 E_p(r)=
 \sum_{k\bmod p-1}
 \left[
 J(\omega^{-k},\omega^{k-r})
 J(\omega^{-k},\omega^{-r})
 \right]^2.
 \tag{25}
\end{equation}
For \(a\in(\mathbb Z/(p-1)\mathbb Z)^\times\), let
\(\sigma_a\) be the cyclotomic Galois automorphism sending character values
to their \(a\)-th powers.  Then, exactly,
\begin{equation}
 \sigma_a E_p(r)=E_p(ar).
 \tag{26}
\end{equation}

Assume additionally the two assertions suggested by the finite computation:
\begin{enumerate}
\item for a prime \(\mathfrak P\) above \(p\),
      \(E_p(r)\equiv b_r\pmod{\mathfrak P}\);
\item the Galois orbit values \(E_p(ar)\) are distinct up to the involution
      \(a\sim-a\).
\end{enumerate}
Then, with
\[
 \nu_{p,n}
 =
 \frac{p-1}{\gcd(n-1,p-1)},
\]
one has
\begin{equation}
 [\mathbb Q(E_p(n-p)):\mathbb Q]
 =
 \frac{\varphi(\nu_{p,n})}{2}
 \tag{27}
\end{equation}
apart from the harmless orders \(1\) and \(2\).  Consequently a single fixed
bounded-degree coefficient field cannot contain the off-centre lifts for a
positive proportion of top-window primes as \(n\to\infty\).
\end{proposition}

\begin{proof}
Applying \(\sigma_a\) to (25) raises every character to its \(a\)-th power:
\[
 \sigma_a E_p(r)
 =
 \sum_k
 \left[
 J(\omega^{-ak},\omega^{a(k-r)})
 J(\omega^{-ak},\omega^{-ar})
 \right]^2.
\]
Multiplication by \(a\) permutes the residue classes \(k\bmod p-1\).
Renaming \(ak\) as \(k\) gives (26).

For the top-window residue \(r=n-p\),
\[
 r\equiv n-1\pmod{p-1}.
\]
Hence the character \(\omega^r\) has exact order
\[
 \nu_{p,n}=\frac{p-1}{\gcd(n-1,p-1)}.
\]
The Galois action on the cyclic subgroup generated by \(\omega^r\) has
\(\varphi(\nu_{p,n})\) possibilities.  Under the stated distinctness
assumption, only the involution \(a\sim-a\) identifies two values, giving
(27).

Fix \(R\).  If \(\nu_{p,n}\le R\), then
\[
 p-1=td,\qquad t\le R,\qquad d\mid n-1.
\]
Thus the number of such candidate primes is at most
\[
 R\,\tau(n-1)=n^{o(1)}.
\]
Now fix a number-field degree bound \(D\).  Since
\(\varphi(t)\to\infty\), choose \(R=R(D)\) so that
\(\varphi(t)/2>D\) for all \(t>R\).  By (27), any lift lying in a field of
degree at most \(D\) must come from one of the \(n^{o(1)}\) bounded-order
candidates.  This cannot cover a positive proportion of the
\(\asymp n/\log n\) top-window primes.
\end{proof}

\begin{remark}[Status and scope of Result 9]
The Galois covariance (26) is algebraic.  The congruence
\(E_p(r)\equiv b_r\pmod{\mathfrak P}\) was checked numerically for
\(p=11,\ldots,43\) in \(222\) cases, and the normalized centre agreed with
the modular coefficient in \(7\) tested primes.  The distinctness needed
for the exact degree formula was checked only for \(p=17,\ldots,31\).
Accordingly, the lift and the exact degree formula are not promoted here to
unconditional theorems.

Even if those inputs are proved, the obstruction has a deliberately narrow
scope: it rules out one fixed bounded-complexity compatible system whose
coefficient field contains these off-centre values.  It does \emph{not}
exclude \(n\)-dependent families, fields whose complexity grows with \(n\),
or Galois-invariant detectors such as orbit norms.  Those require separate
quantitative arguments.
\end{remark}

\subsection{Negative results}

\begin{remark}[Where the attempted lanes first fail]\label{rem:negative}
The following are negative conclusions of the revisit, with the first failed
implication made explicit.

\begin{enumerate}
\item \emph{Minors of the \(C_{s,m}\) array.}
The fixed and reflected rows satisfy different minimal order-\(3\)
recurrences and are not co-solutions, so there is no direct Casoratian.
The primitive \(2\times2\) minors then have the full sum-of-column height,
about \(2.7n\)--\(2.9n\), so divisibility by two nearby primes is far too
small to contradict their archimedean size.

\item \emph{A common LCLM Ore system.}
Passing to a direct-sum/LCLM system does not manufacture a badness detector:
the direct-sum Casoratian ratio depends only on the recurrence coefficients,
not on the distinguished solutions.  Reintroducing the boundary vector
returns to the same full-height minors.

\item \emph{The first \(p\)-adic jet.}
On the bad locus the second-digit identity has the form
\[
 \frac{b_n}{p}\equiv
 -11\,\frac{B_K(n)}p+R_n\pmod p,
\]
with a new free digit \(R_n\).  Hence badness supplies no second thin
congruence; such a congruence appears only on the much rarer locus
\(p^2\mid b_n\).

\item \emph{The companion solution.}
On bad window primes the companion relation
\[
 5(p^2a_n)\equiv6(b_n/p)\pmod p,
 \qquad
 v_p(a_n)=-3+v_p(b_n),
\]
lies on the same one-dimensional line as the scalar Apéry condition.
It does not create an independent local equation.

\item \emph{Large/larger sieve and CRT alone.}
They can bound how many integers satisfy many local restrictions, but at a
single prescribed integer they saturate at one candidate.  The reflected
model in Proposition~\ref{prop:planted} shows that nonempty sparse supports
can be planted on that singleton without violating the available
single-row support laws.

\item \emph{Vaughan/Fermat-quotient analogy.}
Formula (23) corrects the claim that there is no structure, but it is not
bilinear: fixing one modulus leaves the opposite CRT phase dependent on the
same pair of variables.  The multiplicative identity
\(q_p(uv)=q_p(u)+q_p(v)\) used in Fermat-quotient arguments has no Apéry
analogue, and the terminal-prime sector leaves \(T(n)\) itself.
\end{enumerate}
\end{remark}

\subsection{Open statements}

The smallest open statements isolated by the revisit are listed below in the
requested order.  They are \emph{not} a linear implication chain; the exact
relations are stated after the list.

\begin{enumerate}
\item[\(\mathrm{H1}\).]
\emph{Constant phase saving.}
There exists \(\eta>0\) such that, for all sufficiently large \(n\),
\[
 |T(n)|\le(1-\eta)L(n).
\]
By Lemma~\ref{lem:onephase}, \(\mathrm{H1}\Rightarrow\mathrm W\).

\item[\(\mathrm W\).]
\emph{Pointwise top-window constant saving.}
There exists \(\delta>0\) such that
\[
 N_1(n)\le(1-\delta)L(n)
\]
for all sufficiently large \(n\).

\item[\(\mathrm{L3C}\).]
\emph{Logarithmic three-characteristic codegree.}
For every fixed \(A>0\),
\[
 \max_{1\le h,k\le A\log n}\mathcal C_n(h,k)
 =
 o_A\!\left(\frac{n}{\log^3n}\right).
\]
Proposition~\ref{prop:carrier} shows that this implies the stronger
\(q=1\) conclusion \(N_1(n)=o(n/\log n)\).

\item[\(\mathrm{FP}\).]
\emph{Medium/far pair decorrelation for the master bad-prime sets.}
For each fixed \(0<\varepsilon\le1\), there is
\(\kappa(\varepsilon)=o(\varepsilon^2)\) such that, for all sufficiently
large \(N\) and all distinct \(m,n\in(N,2N]\) satisfying
\[
 |n-m|\ge c\varepsilon^2\frac{N}{\log N},
\]
one has
\[
 |\mathcal P(m)\cap\mathcal P(n)|
 \le
 \kappa(\varepsilon)\frac{N}{\log N}.
\]
Together with the existing \(O(h)\) codegree bound in the complementary
short range, this would improve the exceptional set to
\(O_\varepsilon(\log X)\).  A statement only for
\(|n-m|\asymp N\) is insufficient.

\item[\(\mathrm{Full}\).]
\emph{Problem~3.2.}
\[
 G_n=e^{o(n)},
\]
equivalently \(M(n)=o(n)\).  In quotient language this requires an aggregate
\(o(n)\) bound over all surviving classes
\(1\le q<f(n)\log n\), not merely separate nonuniform assertions for fixed
\(q\).
\end{enumerate}

The logical relations matter.  We have
\[
 \mathrm{H1}\Longrightarrow\mathrm W,
 \qquad
 \mathrm{L3C}\Longrightarrow
 N_1(n)=o(n/\log n)\Longrightarrow\mathrm W.
\]
On the other hand, \(\mathrm W\) and \(\mathrm{FP}\) are incomparable:
\(\mathrm W\) is a marginal bound and need not make pair intersections
small, while \(\mathrm{FP}\) can hold even if one isolated integer has
maximal top-window load.  The full target implies the corresponding master
pair decorrelation, but \(\mathrm{FP}\) by itself still allows an exceptional
integer on each dyadic scale.  Thus the revisit does not turn any of these
intermediate statements into an unconditional solution.
