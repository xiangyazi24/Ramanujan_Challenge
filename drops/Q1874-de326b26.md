```python
from math import isqrt
from time import perf_counter

LIMIT = 2_000_000
SEG = 1 << 20


def base_primes(n):
    a = bytearray(b'\1') * (n + 1)
    a[:2] = b'\0\0'
    for i in range(2, isqrt(n) + 1):
        if a[i]:
            a[i*i:n+1:i] = b'\0' * (((n-i*i)//i)+1)
    return [i for i,v in enumerate(a) if v]


def segmented_primes(n):
    root = isqrt(n)
    small = base_primes(root)
    lo = 2
    while lo < n:
        hi = min(lo + SEG, n)
        mark = bytearray(b'\1') * (hi-lo)
        for p in small:
            s = max(p*p, ((lo+p-1)//p)*p)
            for x in range(s, hi, p):
                mark[x-lo] = 0
        for i,v in enumerate(mark):
            if v:
                yield lo+i
        lo = hi


def scan_prime(p):
    m = (p-1)//2
    # u[t] is kept only for t=0..m
    u0, u1 = 1 % p, 0
    vals = [u0, u1]
    ok = True
    for t in range(1, m):
        # (2t+1)^3 u[t+1] = 4t(68t^2+3)u[t] - (2t-1)^3u[t-1]
        den = pow(2*t+1, 3, p)
        if den == 0:
            return None
        un = (4*t*(68*t*t+3)*u1 - (2*t-1)**3*u0) % p
        un = un * pow(den, -1, p) % p
        vals.append(un)
        u0, u1 = u1, un

    if (vals[m-1] - 5*vals[m]) % p != 0:
        return None
    if any(vals[t] == 0 for t in range(2, m+1)):
        return ('E', p, 'non-unit')

    sigma = pow(3, -1, p)
    for t in range(2, m+1):
        sigma -= 2*(t*t-1)*pow(vals[t], -1, p)
        sigma %= p
    return ('E', p, sigma)


if __name__ == '__main__':
    t0 = perf_counter()
    hits = []
    for p in segmented_primes(LIMIT):
        if p >= 5:
            r = scan_prime(p)
            if r:
                hits.append(r)
    print('wall seconds', perf_counter()-t0)
    print('hits', hits)
```
