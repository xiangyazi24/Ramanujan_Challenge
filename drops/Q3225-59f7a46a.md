#!/usr/bin/env python3
from math import comb, gcd, isqrt, prod

from q32_newton import (
    apery_numbers,
    evaluate_newton,
    forward_differences,
)


def primes_upto(n):
    mark = bytearray(b"\x01") * (n + 1)
    if n >= 0:
        mark[0] = 0
    if n >= 1:
        mark[1] = 0
    for d in range(2, isqrt(n) + 1):
        if mark[d]:
            mark[d*d:n+1:d] = b"\x00" * (((n-d*d)//d)+1)
    return [p for p in range(2, n + 1) if mark[p]]


def sigma(m, H, s):
    out = H + 1 + ((-m - 1 - (H + 1)) % s)
    assert H + 1 <= out <= H + s
    assert (m + 1 + out) % s == 0
    return out


def neg_binom(X, K):
    # binom(-X,K), X>0.
    return (-1)**K * comb(X + K - 1, K)


def truncated(apery, sig, slope, K, X):
    data = [apery[sig + slope*t] for t in range(K)]
    coeff = forward_differences(data)
    return evaluate_newton(coeff, -X)


def first_far_edge(m, H):
    sig2 = sigma(m, H, 2)
    X2 = (m + 1 + sig2) // 2
    delta2 = (X2 - 2*sig2 - H + 1) % 3
    F2 = 3*sig2 + 2*H - 3 + 2*delta2
    return F2


def audit(m, H):
    F2 = first_far_edge(m, H)
    sig4 = sigma(m, H, 4)
    sig7 = sigma(m, H, 7)
    X4 = (m + 1 + sig4) // 4
    X7 = (m + 1 + sig7) // 7

    Phi = (m - F2) // 6

    I4hi = min(Plo - 1, 1 + (m - sig4)//4)
    I7lo = max(1, Phi - X7 + 1)
    I7hi = min(Plo - 1, 1 + (m - sig7)//7)
    assert I4lo <= I4hi
    assert I7lo <= I7hi

    ap = apery_numbers(m)
    core = [p for p in primes_upto(Phi) if Plo <= p <= Phi]
    Pi = prod(core) if core else 1

    # One cutoff from each interval already sees the full target coordinate.
    K4 = I4lo
    K7 = I7lo
    U4 = neg_binom(X4, K4)
    U7 = neg_binom(X7, K7)
    T4 = truncated(ap, sig4, 4, K4, X4)
    T7 = truncated(ap, sig7, 7, K7, X7)

    assert U4 % Pi == 0
    assert U7 % Pi == 0
    assert (T4 - 5*T7) % Pi == 0

    target_product = 1
    for p in core:
        q, j = divmod(m, p)
        assert q == 6
        assert j >= F2
        assert p - 1 - j >= sig7

        r4 = 2*p - X4
        r7 = p - X7
        assert 0 <= r4 < K4 < p
        assert 0 <= r7 < K7 < p

        assert neg_binom(X4, K4) % p == 0
        assert neg_binom(X7, K7) % p == 0
        assert T4 % p == (5 * ap[j]) % p
        assert T7 % p == ap[j] % p

        # S=p-1 edge K=p is a unit selector.
        assert neg_binom(X4, p) % p == (-2) % p
        assert neg_binom(X7, p) % p == (-1) % p

        # Fixed-q Lucas equivalence (away from fixed divisors of b_6).
        assert ap[m] % p == (ap[6] * ap[j]) % p

        if ap[j] % p == 0:
            target_product *= p

    # First Smith divisor on the two chosen rows.
    Delta1 = gcd(gcd(abs(U4), abs(T4)), gcd(abs(U7), abs(T7)))
    assert gcd(Delta1, Pi) == target_product

    # Every 2x2 minor is ambient-divisible; one cross-slope minor suffices
    # as a regression witness.
    minor = U4*T7 - U7*T4
    assert minor % Pi == 0

    # Verify cutoff freezing at the next cutoff when legal.
    if K4 < I4hi:
        T4n = truncated(ap, sig4, 4, K4 + 1, X4)
        assert (T4n - T4) % Pi == 0
    if K7 < I7hi:
        T7n = truncated(ap, sig7, 7, K7 + 1, X7)
        assert (T7n - T7) % Pi == 0

    print(
        "PASS",
        "m=", m,
        "H=", H,
        "F2=", F2,
        "core_primes=", len(core),
        "I4=", (I4lo, I4hi),
        "I7=", (I7lo, I7hi),
        "targets=", target_product,
    )


if __name__ == "__main__":
    # Regression choices only.  Increase them as desired; no finite range
    # is used as proof of the asymptotic statements above.
    for m, H in ((600, 4), (900, 5), (1200, 5)):
        audit(m, H)

# Bottom line

The multi-`K`, slope-4/slope-7 construction has reached its exact algebraic endpoint:

```text
all selectors        -> ambient q=6 interval;
all Newton values    -> one target coordinate b_j;
first Smith divisor  -> fixed-q=6 target radical;
higher minors        -> ambient interval again.
```

So there is no hidden cross-slope Bezout/Smith/Wronskian compression left inside this architecture.  The next genuinely new theorem is `(FQ6)`, a pointwise fixed-band prime-divisor estimate for the single Apéry number `b_m`.
