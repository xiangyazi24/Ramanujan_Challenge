# 10. Independent executable certificates

The following standard-library Python checker verifies the packet affine increments, the exact combinatorial adjacent-word inequality, and every exponent used above.  It is not an asymptotic proof of the resultant theorem.

```python
#!/usr/bin/env python3
from __future__ import annotations

from fractions import Fraction
from itertools import combinations


def packet_digits(q: int, p: int, t: int) -> tuple[int, int, int]:
    d = q - p
    r = t + 6 * d
    a_plus = t + 13 * d - q
    a_minus = d - t - 1
    return r, a_plus, a_minus


def legal_plus(q: int, p: int, t: int) -> bool:
    d = q - p
    r, a, _ = packet_digits(q, p, t)
    return (
        d > 0
        and d % 2 == 0
        and 0 <= r < p
        and 0 <= a < p
        and 13 * d >= q - t
        and 7 * d <= q - t - 1
    )


def legal_minus(q: int, p: int, t: int) -> bool:
    d = q - p
    r, _, a = packet_digits(q, p, t)
    return (
        d > 0
        and d % 2 == 0
        and 0 <= r < p
        and 0 <= a < p
        and d >= t + 1
        and 7 * d <= q - t - 1
    )


def adjacent_words(us: list[int], H: int):
    us = sorted(set(us))
    gaps = [b - a for a, b in zip(us, us[1:])]
    good = []
    for i in range(len(gaps) - 1):
        if gaps[i] <= H and gaps[i + 1] <= H:
            good.append((us[i], gaps[i], gaps[i + 1]))
    return gaps, good


def check_word_inequality(us: list[int], H: int) -> None:
    us = sorted(set(us))
    if not us:
        return
    gaps, good = adjacent_words(us, H)
    large = sum(g > H for g in gaps)
    # Every bad triple touches at least one large adjacent gap,
    # and each large gap can spoil at most two triple starts.
    assert len(us) <= 2 + 2 * large + len(good)
    span = us[-1] - us[0]
    assert large * (H + 1) <= span


def exhaustive_word_check(n: int = 12) -> None:
    ambient = list(range(n))
    for mask in range(1 << n):
        us = [ambient[i] for i in range(n) if (mask >> i) & 1]
        for H in range(1, n + 1):
            check_word_inequality(us, H)


def packet_increment_check() -> None:
    # A concrete deep legal example.  q and p are odd and d is even.
    q = 1009
    p = 929
    M = 2
    g = 6 * M
    t0 = 20

    rows = []
    for u in range(3):
        t = t0 + g * u
        assert 0 <= t < q
        assert legal_plus(q, p, t)
        assert legal_minus(q, p, t)
        rows.append((t, *packet_digits(q, p, t)))

    for left, right in zip(rows, rows[1:]):
        t0_, r0, ap0, am0 = left
        t1_, r1, ap1, am1 = right
        assert t1_ - t0_ == g
        assert r1 - r0 == g
        assert ap1 - ap0 == g
        assert am1 - am0 == -g

    # Three consecutive u-values have adjacent numerical gaps (g,g).
    assert rows[1][0] - rows[0][0] == g
    assert rows[2][0] - rows[1][0] == g


def exponent_check() -> None:
    M = Fraction(2, 5)
    Z = Fraction(3, 5)
    K = Fraction(2, 5)
    H = Fraction(4, 25)

    # U = T Z / H after the balancing, and also M^2 H^4.
    U = Fraction(1, 1) + Z - H
    assert U == Fraction(36, 25)
    assert 2 * M + 4 * H == U

    pair = U + 2 * K
    excess = U + K
    critical_pair = Fraction(12, 5)
    critical_excess = Fraction(2, 1)

    assert pair == Fraction(56, 25)
    assert excess == Fraction(46, 25)
    assert critical_pair - pair == Fraction(4, 25)
    assert critical_excess - excess == Fraction(4, 25)

    # Full Cartesian selector.
    full_U = 2 * M
    assert full_U == Fraction(4, 5)
    assert full_U + 2 * K == Fraction(8, 5)
    assert full_U + K == Fraction(6, 5)


def main() -> None:
    exhaustive_word_check()
    packet_increment_check()
    exponent_check()
    print("Q6011 finite affine/combinatorial/exponent checks: PASS")


if __name__ == "__main__":
    main()
```
