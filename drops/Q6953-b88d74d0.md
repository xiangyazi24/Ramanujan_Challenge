# 1. Scale and scalar state

Use a real-valued normalized scale with initial value `1`. This is `1/6` of the gamma normalization appearing in the fourfold integral, but the recurrence is homogeneous, so this is the normalization compatible with the requested `scale n <= 10^n` and the `512` majorant.

```lean
private def dualScaleFactor (n : ℕ) : ℝ :=
  2 * (2 * (n : ℝ) + 5) / ((n : ℝ) + 1)

private def dualScale : ℕ → ℝ
  | 0 => 1
  | n + 1 => dualScale n * dualScaleFactor n

@[simp] private theorem dualScale_zero : dualScale 0 = 1 := rfl

@[simp] private theorem dualScale_succ (n : ℕ) :
    dualScale (n + 1) = dualScale n * dualScaleFactor n := rfl

private theorem dualScaleFactor_pos (n : ℕ) :
    0 < dualScaleFactor n := by
  unfold dualScaleFactor
  positivity

private theorem dualScaleFactor_nonneg (n : ℕ) :
    0 ≤ dualScaleFactor n :=
  (dualScaleFactor_pos n).le

private theorem dualScaleFactor_le_ten (n : ℕ) :
    dualScaleFactor n ≤ 10 := by
  have hn : (0 : ℝ) < (n : ℝ) + 1 := by positivity
  rw [dualScaleFactor, div_le_iff₀ hn]
  have hnn : (0 : ℝ) ≤ (n : ℝ) := by positivity
  nlinarith

private theorem dualScale_pos : ∀ n : ℕ, 0 < dualScale n
  | 0 => by norm_num [dualScale]
  | n + 1 => by
      rw [dualScale_succ]
      exact mul_pos (dualScale_pos n) (dualScaleFactor_pos n)

private theorem dualScale_nonneg (n : ℕ) : 0 ≤ dualScale n :=
  (dualScale_pos n).le

private theorem dualScale_le_pow_ten : ∀ n : ℕ,
    dualScale n ≤ (10 : ℝ) ^ n
  | 0 => by norm_num [dualScale]
  | n + 1 => by
      calc
        dualScale (n + 1)
            = dualScale n * dualScaleFactor n := dualScale_succ n
        _ ≤ (10 : ℝ) ^ n * dualScaleFactor n :=
          mul_le_mul_of_nonneg_right
            (dualScale_le_pow_ten n) (dualScaleFactor_nonneg n)
        _ ≤ (10 : ℝ) ^ n * 10 :=
          mul_le_mul_of_nonneg_left
            (dualScaleFactor_le_ten n) (pow_nonneg (by norm_num) n)
        _ = (10 : ℝ) ^ (n + 1) := by rw [pow_succ]
```

Define the scalar sequence and its three-term state:

```lean

private theorem fastScalar_abs_le (n : ℕ) :

    _ = 512 * (((10 : ℝ) / 64) ^ n) := by rw [div_pow]
    _ = 512 * fastRho ^ n := by norm_num [fastRho]
```

```lean
private theorem qNorm_le_seventeen_sixteen_v3Inf
    (z : Chart) (x : Vec3) (hz : InBox z) :
    qNorm z x ≤ (17 / 16 : ℝ) * v3Inf x := by
  rcases hz with ⟨hu0, hu1, hv0, hv1⟩
  let M : ℝ := v3Inf x

  have hx0 : |x 0| ≤ M := by
    dsimp [M, v3Inf]
    exact le_max_left _ _
  have hx1 : |x 1| ≤ M := by
    dsimp [M, v3Inf]
    exact le_max_of_le_right (le_max_left _ _)
  have hx2 : |x 2| ≤ M := by
    dsimp [M, v3Inf]
    exact le_max_of_le_right (le_max_right _ _)
  have hM : 0 ≤ M := (abs_nonneg (x 0)).trans hx0

  have hu_mul : z 0 * |x 0| ≤ (1 / 16 : ℝ) * M := by
    calc
      z 0 * |x 0| ≤ (1 / 16 : ℝ) * |x 0| :=
        mul_le_mul_of_nonneg_right hu1 (abs_nonneg _)
      _ ≤ (1 / 16 : ℝ) * M :=
        mul_le_mul_of_nonneg_left hx0 (by norm_num)

  have hv_mul : z 1 * |x 0| ≤ (1 / 256 : ℝ) * M := by
    calc
      z 1 * |x 0| ≤ (1 / 256 : ℝ) * |x 0| :=
        mul_le_mul_of_nonneg_right hv1 (abs_nonneg _)
      _ ≤ (1 / 256 : ℝ) * M :=
        mul_le_mul_of_nonneg_left hx0 (by norm_num)

  have hq0 : |defect0 z x| ≤ (17 / 16 : ℝ) * M := by
    calc
      |defect0 z x| = |x 1 - z 0 * x 0| := rfl
      _ ≤ |x 1| + |z 0 * x 0| := abs_sub _ _
      _ = |x 1| + z 0 * |x 0| := by
        rw [abs_mul, abs_of_nonneg hu0]
      _ ≤ M + (1 / 16 : ℝ) * M := add_le_add hx1 hu_mul
      _ = (17 / 16 : ℝ) * M := by ring

  have hq1 : |defect1 z x| ≤ (17 / 16 : ℝ) * M := by
    calc
      |defect1 z x| = |x 2 - z 1 * x 0| := rfl
      _ ≤ |x 2| + |z 1 * x 0| := abs_sub _ _
      _ = |x 2| + z 1 * |x 0| := by
        rw [abs_mul, abs_of_nonneg hv0]
      _ ≤ M + (1 / 256 : ℝ) * M := add_le_add hx2 hv_mul
      _ ≤ (17 / 16 : ℝ) * M := by nlinarith

  exact max_le hq0 hq1
```

# 8. From zero defect to the graph line

A robust zero characterization for the sup defect is:

```lean
private theorem qNorm_eq_zero_iff (z : Chart) (x : Vec3) :
    qNorm z x = 0 ↔
      x 1 = z 0 * x 0 ∧ x 2 = z 1 * x 0 := by
  constructor
  · intro h
    have h0abs : |defect0 z x| = 0 := by
      apply le_antisymm
      · exact (le_max_left _ _).trans_eq h
      · exact abs_nonneg _
    have h1abs : |defect1 z x| = 0 := by
      apply le_antisymm
      · exact (le_max_right _ _).trans_eq h
      · exact abs_nonneg _
    have h0 : defect0 z x = 0 := abs_eq_zero.mp h0abs
    have h1 : defect1 z x = 0 := abs_eq_zero.mp h1abs
    exact ⟨sub_eq_zero.mp h0, sub_eq_zero.mp h1⟩
  · rintro ⟨h0, h1⟩
    simp [qNorm, defect0, defect1, h0, h1]
```
