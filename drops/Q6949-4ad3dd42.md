

```text

c0 + (c1-v')*u + c2*v = 0.
```

```text
v = u'/d = (1/d)*u' = u*u',
```


A useful identity for the implementation is therefore


# 2. Exact defect recurrence

Define

```text
q0 = x1-u*x0,
q1 = x2-v*x0,
q'0 = x'1-u'*x'0,
q'1 = x'2-v'*x'0.
```

For the first coordinate,

```text
q'0
 = x2-u'*x1
 = (q1+v*x0)-u'*(q0+u*x0)
 = -u'*q0+q1+(v-u*u')x0
 = -u'*q0+q1.
```

For the second coordinate,

```text
q'1
 = c0*x0+c1*x1+c2*x2-v'*x1
 = (c1-v')q0+c2*q1

```

Hence the claimed matrix is exactly correct:

```text
G = [[-u',1],[c1-v',c2]].
```

Its determinant is

```text
det G = (-u')c2-(c1-v')
```

## 5.1 Pullback graph cancellation

Prefer storing the cross-multiplied equation

```text
c0*d = -c1+v'-c2*u'
```

rather than repeatedly unfolding `a,b,r` and dividing by `c0`.

```lean
lemma companion_pullback_graph_relations
    (c0 c1 c2 d u v un vn : ℝ)
    (hc0_lo : 1 ≤ c0)
    (hc1_hi : c1 ≤ -35)
    (hc2_hi : c2 ≤ 37)
    (hun_lo : 0 ≤ un)
    (hun_hi : un ≤ (1 : ℝ) / 16)
    (hvn_lo : 0 ≤ vn)
    (hd_cross : c0 * d = -c1 + vn - c2 * un)
    (hu : u = 1 / d)
    (hv : v = un / d) :
    v = u * un ∧ c0 + (c1 - vn) * u + c2 * v = 0 := by
  have hcu : c2 * un ≤ (37 : ℝ) / 16 := by
    calc
      c2 * un ≤ 37 * un :=
        mul_le_mul_of_nonneg_right hc2_hi hun_lo
      _ ≤ 37 * ((1 : ℝ) / 16) :=
        mul_le_mul_of_nonneg_left hun_hi (by norm_num)
      _ = (37 : ℝ) / 16 := by ring

  have hDelta : (523 : ℝ) / 16 ≤ -c1 + vn - c2 * un := by
    nlinarith

  have hprod : 0 < c0 * d := by
    rw [hd_cross]
    nlinarith

  have hd_pos : 0 < d := by
    rcases (mul_pos_iff.mp hprod) with h | h
    · exact h.2
    · nlinarith [hc0_lo, h.1]

  have hd_ne : d ≠ 0 := ne_of_gt hd_pos

  have hvu : v = u * un := by
    rw [hu, hv]
    field_simp [hd_ne]

  have hdu : d * u = 1 := by
    rw [hu]
    field_simp [hd_ne]

  refine ⟨hvu, ?_⟩
  rw [hvu]
  calc
    c0 + (c1 - vn) * u + c2 * (u * un)
        = c0 + (c1 - vn + c2 * un) * u := by ring
    _ = c0 - (-c1 + vn - c2 * un) * u := by ring
    _ = c0 - (c0 * d) * u := by rw [← hd_cross]
    _ = 0 := by rw [hdu]; ring
```

## 5.3 Transverse sup lower bound

This is the corrected standalone statement. Notice that `c0`, `u_n`, and `v_n` disappear: after the two defect-step identities are known, only `c1,c2,u_{n+1},v_{n+1}` matter.

```lean
lemma companion_defect_sup_lower
    (c1 c2 un vn q0 q1 Q0 Q1 : ℝ)
    (hc1_lo : -49 ≤ c1)
    (hc1_hi : c1 ≤ -35)
    (hc2_lo : 35 ≤ c2)
    (hc2_hi : c2 ≤ 37)
    (hun_lo : 0 ≤ un)
    (hun_hi : un ≤ (1 : ℝ) / 16)
    (hvn_lo : 0 ≤ vn)
    (hvn_hi : vn ≤ (1 : ℝ) / 256)
    (hQ0 : Q0 = -un * q0 + q1)
    (hQ1 : Q1 = (c1 - vn) * q0 + c2 * q1) :
    (1 / 2 : ℝ) * max |q0| |q1| ≤ max |Q0| |Q1| := by
  let Δ : ℝ := -c1 + vn - c2 * un
  let M : ℝ := max |Q0| |Q1|

  have hc2_nonneg : 0 ≤ c2 := by linarith
  have hvc_nonneg : 0 ≤ vn - c1 := by linarith

  have hcu : c2 * un ≤ (37 : ℝ) / 16 := by
    calc
      c2 * un ≤ 37 * un :=
        mul_le_mul_of_nonneg_right hc2_hi hun_lo
      _ ≤ 37 * ((1 : ℝ) / 16) :=
        mul_le_mul_of_nonneg_left hun_hi (by norm_num)
      _ = (37 : ℝ) / 16 := by ring

  have hdet : (523 : ℝ) / 16 ≤ Δ := by
    dsimp [Δ]
    nlinarith

  have h32 : (32 : ℝ) ≤ Δ := by
    have h : (32 : ℝ) ≤ 523 / 16 := by norm_num
    exact h.trans hdet

  have hΔ_nonneg : 0 ≤ Δ := by linarith

  have hQ0M : |Q0| ≤ M := by
    dsimp [M]
    exact le_max_left _ _
  have hQ1M : |Q1| ≤ M := by
    dsimp [M]
    exact le_max_right _ _
  have hM_nonneg : 0 ≤ M :=
    (abs_nonneg Q0).trans hQ0M

  -- Cross-multiplied inverse coordinates.
  have hinv0 : Δ * q0 = c2 * Q0 - Q1 := by
    dsimp [Δ]
    rw [hQ0, hQ1]
    ring

  have hinv1 : Δ * q1 = (vn - c1) * Q0 - un * Q1 := by
    dsimp [Δ]
    rw [hQ0, hQ1]
    ring

  have habs0 : Δ * |q0| = |c2 * Q0 - Q1| := by
    have h := congrArg abs hinv0
    simpa [abs_mul, abs_of_nonneg hΔ_nonneg] using h

  have habs1 : Δ * |q1| = |(vn - c1) * Q0 - un * Q1| := by
    have h := congrArg abs hinv1
    simpa [abs_mul, abs_of_nonneg hΔ_nonneg] using h

  have hc2Q : c2 * |Q0| ≤ 37 * M :=
    mul_le_mul hc2_hi hQ0M (abs_nonneg Q0) (by norm_num)

  have hrow0 : Δ * |q0| ≤ 38 * M := by
    calc
      Δ * |q0| = |c2 * Q0 - Q1| := habs0
      _ ≤ |c2 * Q0| + |Q1| := abs_sub _ _
      _ = c2 * |Q0| + |Q1| := by
        rw [abs_mul, abs_of_nonneg hc2_nonneg]
      _ ≤ 37 * M + M := add_le_add hc2Q hQ1M
      _ = 38 * M := by ring

  have hvc_le : vn - c1 ≤ 50 := by
    nlinarith
  have hun_le_one : un ≤ 1 := by
    nlinarith

  have hvcQ : (vn - c1) * |Q0| ≤ 50 * M :=
    mul_le_mul hvc_le hQ0M (abs_nonneg Q0) (by norm_num)

  have hunQ : un * |Q1| ≤ 1 * M :=
    mul_le_mul hun_le_one hQ1M (abs_nonneg Q1) (by norm_num)

  have hrow1 : Δ * |q1| ≤ 51 * M := by
    calc
      Δ * |q1| = |(vn - c1) * Q0 - un * Q1| := habs1
      _ ≤ |(vn - c1) * Q0| + |un * Q1| := abs_sub _ _
      _ = (vn - c1) * |Q0| + un * |Q1| := by
        rw [abs_mul, abs_of_nonneg hvc_nonneg,
          abs_mul, abs_of_nonneg hun_lo]
      _ ≤ 50 * M + 1 * M := add_le_add hvcQ hunQ
      _ = 51 * M := by ring

  have hq0_scaled : 32 * |q0| ≤ Δ * |q0| :=
    mul_le_mul_of_nonneg_right h32 (abs_nonneg q0)

  have hq1_scaled : 32 * |q1| ≤ Δ * |q1| :=
    mul_le_mul_of_nonneg_right h32 (abs_nonneg q1)

  have h38 : 38 * M ≤ 64 * M :=
    mul_le_mul_of_nonneg_right (by norm_num) hM_nonneg
  have h51 : 51 * M ≤ 64 * M :=
    mul_le_mul_of_nonneg_right (by norm_num) hM_nonneg

  have hq0_two : |q0| ≤ 2 * M := by
    have h : 32 * |q0| ≤ 64 * M :=
      hq0_scaled.trans (hrow0.trans h38)
    nlinarith

  have hq1_two : |q1| ≤ 2 * M := by
    have h : 32 * |q1| ≤ 64 * M :=
      hq1_scaled.trans (hrow1.trans h51)
    nlinarith

  have hmax : max |q0| |q1| ≤ 2 * M :=
    max_le hq0_two hq1_two

  calc
    (1 / 2 : ℝ) * max |q0| |q1|
        ≤ (1 / 2 : ℝ) * (2 * M) :=
          mul_le_mul_of_nonneg_left hmax (by norm_num)
    _ = M := by ring
    _ = max |Q0| |Q1| := by rfl
```

The proof deliberately uses only:

- the two exact coordinate identities;
- `abs_mul`, `abs_sub`, and `abs_of_nonneg`;
- `le_max_left`, `le_max_right`, and `max_le`;
- scalar order arithmetic (`ring`, `norm_num`, `nlinarith`).

There is no matrix norm or inverse API.

# 6. Sup norm versus sum norm

The sup norm is not merely shorter; the proposed factor `1/2` fails for the sum norm under the same assumptions.

Choose the allowed endpoint data

```text
c1=-35,
c2=37,
u'=1/16,
v'=0.
```

Then

```text
Delta = 35-37/16 = 523/16.
```

Take the next defect to be

```text
(q'0,q'1)=(1,0).
```

The exact inverse identities give

```text
q0 = 37/Delta = 592/523,
q1 = 35/Delta = 560/523.
```

Therefore

```text
|q0|+|q1| = 1152/523,
(1/2)(|q0|+|q1|) = 576/523 > 1
                     = |q'0|+|q'1|.
```

This example is compatible with the full graph assumptions: take `c0=1`, so

```text
d=Delta=523/16,
u=16/523 <=1/16,
v=u*u'=1/523 <=1/256.
```

Thus a sum-norm lemma with factor `1/2` is false. The sup-norm statement is the correct transverse estimate.
