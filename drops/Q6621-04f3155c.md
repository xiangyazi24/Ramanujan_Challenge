\begin{theorem}[Abstract inverse-square clique theorem]\label{thm:abstract-h2}
Let a word satisfy $R_d\le Cd$ for $d\le H$. Then
$Q_H\le22CH^2(1+\log H)$.
\end{theorem}

\begin{proof}
Partition into intervals of length $H$. Apply Lemma~\ref{lem:clique} to each
color class and Lemma~\ref{lem:localize} to sum the inverse-square energies.
This gives $Q_H\le11H^2\cdot2C(1+\log H)$.
\end{proof}

\begin{lemma}[Clique]\label{lem:clique}
For a color class $A$ in an interval of length at most $2H$,
$\binom{|A|}{3}\le11H^2E_H(A)$.
\end{lemma}

\begin{proof}
For three points the claim follows because one of the two consecutive gaps is
at most $H$. For $m\ge4$, a half interval contains $n\ge m/2$ points.
If its consecutive gaps are $g_i$, then
$(\sum g_i^{-2})(\sum g_i)^2\ge(n-1)^3$, and the span bound gives
$E_H(A)\ge m^3/(64H^2)$. Since $\binom m3\le m^3/6$, the result follows.
\end{proof}

\begin{lemma}[Localization]\label{lem:localize}
For blocks $B_j=[jH,(j+1)H)$ and windows $W_j=[jH,(j+2)H)$,
$\sum_{j,a}E_H(A_{j,a})\le2\sum_{d\le H}R_d/d^2\le2C(1+\log H)$.
\end{lemma}

\begin{proof}
Assign triples to the block of their least point. Each contributing pair of
separation $d$ occurs in at most two windows, proving the estimate.
\end{proof}

\begin{proposition}[Lower word]\label{prop:lower-word}
There are words with $R_d\le(5/2)d$ and
$Q_H=(1/8+o(1))H^2$.
\end{proposition}

\begin{proof}
Use triples $b_e,b_e+t,b_e+2t$ with $b_e=e(H+1)$,
$1\le s\le t\le\lfloor H/2\rfloor$, private filler symbols, and no shared
symbols between blocks. Direct counting gives
$R_d=2n_d+\mathbf1_{2|d}n_{d/2}\le(5/2)d$ and
$Q_H=M(M+1)/2$.
\end{proof}

\begin{proof}

\end{proof}

\end{remark}
