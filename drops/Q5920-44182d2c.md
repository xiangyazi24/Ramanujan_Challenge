## 8. Executable Sage audit

The following code verifies the divided Newton row and the direct fixed-minor formula without guessing any factorization.

```sage
from sage.all import *


def harmonic_mod(k, p):
    return sum(inverse_mod(j, p) for j in range(1, k + 1)) % p


def newton_weight(d, L, i):
    return (-1)^i * binomial(d + i, i) * binomial(d + L + 1, L - i)


def divided_newton_row(d, L, p):
    t = p - 1 - d
    assert 0 <= t <= L < p
    out = {}
    for i in range(L + 1):
        node = d + i
        if i == t:
            value = harmonic_mod(L - t, p) - harmonic_mod(t, p)
        else:
            value = (
                (-1)^(i + t)
                * binomial(L, i)
                * inverse_mod(binomial(L, t), p)
                * inverse_mod(i - t, p)
            )
        out[node] = ZZ(value % p)

        weight = ZZ(newton_weight(d, L, i))
        delta = ZZ(1 if i == t else 0)
        assert (weight - delta) % p == 0
        assert ((weight - delta) // p - out[node]) % p == 0
    return out


def carrier(values, d, L):
    return sum(
        newton_weight(d, L, i) * values[d + i]
        for i in range(L + 1)
    )


def divided_jet(values, d, L, p):
    raw = ZZ(carrier(values, d, L))
    assert raw % p == 0
    return ZZ((raw // p) % p)


def quotient_column(values, dminus, D, L, p, QI):
    cofactor = ZZ(QI // p)
    scale = inverse_mod(cofactor, p)
    return vector(GF(p), [
        scale * divided_jet(values, dminus, L, p),
        scale * divided_jet(values, D, L, p),
    ])


def fixed_minor_direct(W, X1, dminus, D, L, p, QI):
    left = quotient_column(W, dminus, D, L, p, QI)
    right = quotient_column(X1, dminus, D, L, p, QI)
    return matrix(GF(p), 2, 2, [left[0], right[0], left[1], right[1]]).det()
```

# Srows[s] is d |-> C_{p+s}(d-1)+C_{p+s}(d+1).

def projected_jets(G, Srows, Yrows, J, b, p, srange):
    xi = {}
    eta = {}
    for s in srange:
        a = ZZ(G(Srows[s]) - J[s])
        c = ZZ(G(Yrows[s]) - b[s])
        assert a % p == 0 and c % p == 0
        xi[s] = (a // p) % p
        eta[s] = (c // p) % p
    return xi, eta


def dot_apply(coefficients, argument_values, s):
    # coefficients is a list of polynomial functions v_j(s)
    return sum(coefficients[j].derivative()(s) * argument_values[s+j]
               for j in range(len(coefficients)))


def apply_operator(coefficients, argument_values, s):
    return sum(coefficients[j](s) * argument_values[s+j]
               for j in range(len(coefficients)))


def ore_cartier_minor(U, Lop, M, xi_minus, xi_plus,
                       eta_minus, eta_plus, J, b, r, p, QI_over_p):
    wm = (apply_operator(U, xi_minus, r)
          + dot_apply(U, J, r)
          - M(r)*eta_minus[r] - M.derivative()(r)*b[r]) % p
    wp = (apply_operator(U, xi_plus, r)
          + dot_apply(U, J, r)
          - M(r)*eta_plus[r] - M.derivative()(r)*b[r]) % p
    xm = (apply_operator(Lop, xi_minus, r+1)
          + dot_apply(Lop, J, r+1)) % p
    xp = (apply_operator(Lop, xi_plus, r+1)
          + dot_apply(Lop, J, r+1)) % p
    return ((wm*xp-wp*xm) * inverse_mod(QI_over_p, p)^2) % p

def equal_jet_residue(dminus, D, L, p, marked_value):
    hm = divided_newton_row(dminus, L, p)
    hp = divided_newton_row(D, L, p)
    nodes = sorted(set(hm) | set(hp))
