\label{lem:ad-high-prime-shadow}

$c_p=(p-1)/2$, $c_q=(q-1)/2$, and write actual zero positions as

\[
 a_{p,q}(k,\ell)=\langle k+pJ\rangle_{pq}

\begin{lemma}[Exact affine-correlation second moment]
\label{lem:ad-affine-second-moment}
Let $q$ be prime and $K\subset\mathbb F_q$ have cardinality $z$.  Put
\[
 A_K(\lambda,\beta)
 =\#\{x\in K:\lambda x+\beta\in K\},
 \qquad \lambda\in\mathbb F_q^*,\ \beta\in\mathbb F_q.
\]

\[
 \boxed{
 \sum_{\lambda\ne0}\sum_{\beta\in\mathbb F_q}
 \left(A_K(\lambda,\beta)-\frac{z^2}{q}\right)^2
 =qz^2\left(1-\frac zq\right)^2.}
\]
\end{lemma}

\begin{proof}
One has
\[
 \sum_{\lambda,\beta}A_K(\lambda,\beta)=(q-1)z^2.
\]
For the second moment, coincident source points force coincident target
points and contribute $(q-1)z^2$.  For ordered distinct source and target
pairs there is a unique affine map, contributing $z^2(z-1)^2$.  Thus
\[
 \sum_{\lambda,\beta}A_K(\lambda,\beta)^2
 =(q-1)z^2+z^2(z-1)^2.
\]
Expansion around the mean $z^2/q$ gives the formula.
\end{proof}
