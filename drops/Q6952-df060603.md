# Finite rational trajectory/Krylov block

The block below is intended to be pasted inside `namespace RamanujanChallenge.P25` in the current `Problem25Moment.lean`.  It uses exact Cramer coordinates for the companion coefficients.  These are rational expressions in the displayed trajectory and are definitionally independent of any matrix inverse.  The locally simplified Q6938/Q6941 formulas can later be installed as rewrite lemmas for `c0`, `c1`, and `c2`; none of the downstream statements changes.

```lean
namespace MillerKrylov

set_option maxHeartbeats 0
set_option maxRecDepth 100000

abbrev Vec3 := Fin 3 → ℝ
abbrev Mat3 := Matrix (Fin 3) (Fin 3) ℝ
abbrev Chart := Fin 2 → ℝ

private lemma n1_ne (n : ℕ) : (n : ℝ) + 1 ≠ 0 := by positivity
private lemma n2_ne (n : ℕ) : (n : ℝ) + 2 ≠ 0 := by positivity
private lemma n3_ne (n : ℕ) : (n : ℝ) + 3 ≠ 0 := by positivity
private lemma n4_ne (n : ℕ) : (n : ℝ) + 4 ≠ 0 := by positivity
private lemma n5_ne (n : ℕ) : (n : ℝ) + 5 ≠ 0 := by positivity
private lemma two_n3_ne (n : ℕ) : 2 * (n : ℝ) + 3 ≠ 0 := by positivity
private lemma two_n5_ne (n : ℕ) : 2 * (n : ℝ) + 5 ≠ 0 := by positivity
private lemma two_n7_ne (n : ℕ) : 2 * (n : ℝ) + 7 ≠ 0 := by positivity
private lemma two_n9_ne (n : ℕ) : 2 * (n : ℝ) + 9 ≠ 0 := by positivity

/-- Real cast of the literal challenge matrix. -/
def M (n : ℕ) : Mat3 :=
  fun i j => (challengeMatrix (n : ℤ) i j : ℝ)

/-- The exact Q6938/Q6941 rational trajectory. -/
def T (n : ℕ) : Mat3 :=
  let x : ℝ := n
  !![
    4 * (x + 2) * (17*x^3 + 111*x^2 + 240*x + 171) /
      ((x + 1) * (x + 3) * (2*x + 3) * (2*x + 5)),
    (x + 2) * (24*x^2 + 101*x + 102) /
      ((x + 1) * (2*x + 3)),
    (x + 2) * (2*x + 5) * (16*x^2 + 81*x + 90) /
      (2 * (x + 1) * (2*x + 3));

    (96*x^4 + 780*x^3 + 2384*x^2 + 3273*x + 1723) /
      ((x + 1) * (x + 2) * (x + 3) * (2*x + 3) * (2*x + 5)),
    (68*x^3 + 398*x^2 + 778*x + 523) /
      (2 * (x + 1) * (x + 2) * (2*x + 3)),
    (96*x^4 + 884*x^3 + 2970*x^2 + 4360*x + 2403) /
      (4 * (x + 1) * (x + 2) * (2*x + 3));

    -5 * (24*x^2 + 117*x + 143) /
      ((x + 1) * (x + 2) * (x + 3) * (2*x + 3) * (2*x + 5)),
    -5 * (16*x + 41) /
      (2 * (x + 1) * (x + 2) * (2*x + 3)),
    (8*x^3 - 44*x^2 - 478*x - 801) /
      (4 * (x + 1) * (x + 2) * (2*x + 3))
  ]

/-- Scalar in `T(n)ᵀ M(n) = sigma(n) I`. -/
def sigma (n : ℕ) : ℝ :=
  -2 * ((n : ℝ) + 2)^2 * ((n : ℝ) + 3)^2 *
    (2 * (n : ℝ) + 5) * (2 * (n : ℝ) + 7)^2

lemma sigma_ne (n : ℕ) : sigma n ≠ 0 := by
  unfold sigma
  positivity

/-- This proof is nine independent rational polynomial identities. -/
theorem T_transpose_mul_M (n : ℕ) :
    (T n).transpose * M n = sigma n • (1 : Mat3) := by
  ext i j
  fin_cases i <;> fin_cases j
  all_goals
    simp only [Matrix.mul_apply, Matrix.transpose_apply, Fin.sum_univ_three,
      Matrix.smul_apply, Matrix.one_apply]
    simp [T, M, sigma, challengeMatrix,
      m11, m12, m13, m21, m22, m23, m31, m32, m33,
      Matrix.cons_val_two]
    push_cast
    field_simp [n1_ne n, n2_ne n, n3_ne n,
      two_n3_ne n, two_n5_ne n]
    ring

/-- Useful independent determinant check. -/
theorem det_T (n : ℕ) :
    Matrix.det (T n) =
      (((n : ℝ) + 3) * (2 * (n : ℝ) + 7)^2) /
        (((n : ℝ) + 1) * (2 * (n : ℝ) + 3)^2) := by
  rw [Matrix.det_fin_three]
  simp [T, Matrix.cons_val_two]
  field_simp [n1_ne n, n2_ne n, n3_ne n,
    two_n3_ne n, two_n5_ne n]
  ring

lemma det_T_ne (n : ℕ) : Matrix.det (T n) ≠ 0 := by
  rw [det_T]
  positivity

/-- First standard basis column. -/
def e0 : Vec3 := ![1, 0, 0]
def e2 : Vec3 := ![0, 0, 1]

/-- The first and second nontrivial Krylov columns. -/
def p (n : ℕ) : Vec3 := T n *ᵥ e0
def q (n : ℕ) : Vec3 := T n *ᵥ p (n + 1)
def r (n : ℕ) : Vec3 := T n *ᵥ q (n + 1)

/-- `U(n)=[e0,T(n)e0,T(n)T(n+1)e0]`, represented by columns. -/
def Ucol (n : ℕ) (j : Fin 3) : Vec3 := ![e0, p n, q n] j

def U (n : ℕ) : Mat3 := fun i j => Ucol n j i

/-- Positive polynomial in the exact Krylov determinant. -/
def krylovPoly (n : ℕ) : ℝ :=
  let x : ℝ := n
  3072*x^7 + 61824*x^6 + 531184*x^5 + 2525522*x^4 +
    7175793*x^3 + 12183785*x^2 + 11446123*x + 4589916

lemma krylovPoly_pos (n : ℕ) : 0 < krylovPoly n := by
  unfold krylovPoly
  positivity

/-- Full exact determinant certificate from the Krylov computation. -/
theorem det_U (n : ℕ) :
    Matrix.det (U n) =
      -15 * krylovPoly n /
        (2 * ((n : ℝ) + 1) * ((n : ℝ) + 2)^3 *
          ((n : ℝ) + 3)^2 * ((n : ℝ) + 4) *
          (2 * (n : ℝ) + 3)^2 * (2 * (n : ℝ) + 5)^2 *
          (2 * (n : ℝ) + 7)) := by
  rw [Matrix.det_fin_three]
  simp [U, Ucol, p, q, e0, T, Matrix.mulVec,
    Fin.sum_univ_three, Matrix.cons_val_two]
  field_simp [n1_ne n, n2_ne n, n3_ne n, n4_ne n,
    two_n3_ne n, two_n5_ne n, two_n7_ne n]
  ring

lemma det_U_ne (n : ℕ) : Matrix.det (U n) ≠ 0 := by
  rw [det_U]
  have hp := krylovPoly_pos n
  positivity

/-- Because the first column of `U` is `e0`, this is its determinant. -/
def krylovDelta (n : ℕ) : ℝ :=
  p n 1 * q n 2 - p n 2 * q n 1

lemma krylovDelta_eq_det (n : ℕ) :
    krylovDelta n = Matrix.det (U n) := by
  rw [Matrix.det_fin_three]
  simp [krylovDelta, U, Ucol, e0, Matrix.cons_val_two]
  ring

lemma krylovDelta_ne (n : ℕ) : krylovDelta n ≠ 0 := by
  rw [krylovDelta_eq_det]
  exact det_U_ne n

/-- Exact Cramer coefficients of the third Krylov column. -/
def c1 (n : ℕ) : ℝ :=
  (r n 1 * q n 2 - r n 2 * q n 1) / krylovDelta n

def c2 (n : ℕ) : ℝ :=
  (p n 1 * r n 2 - p n 2 * r n 1) / krylovDelta n

def c0 (n : ℕ) : ℝ :=
  r n 0 - c1 n * p n 0 - c2 n * q n 0

/-- Companion matrix; this is an exact rational definition, not an inverse. -/
def C (n : ℕ) : Mat3 :=
  !![0, 0, c0 n;
     1, 0, c1 n;
     0, 1, c2 n]

lemma det_C (n : ℕ) : Matrix.det (C n) = c0 n := by
  rw [Matrix.det_fin_three]
  simp [C, Matrix.cons_val_two]

lemma r_reconstruct (n : ℕ) :
    r n = c0 n • e0 + c1 n • p n + c2 n • q n := by
  ext i
  fin_cases i
  · simp [c0, e0, Matrix.cons_val_two]
  · simp [c1, c2, krylovDelta, e0, Matrix.cons_val_two]
    field_simp [krylovDelta_ne n]
    ring
  · simp [c1, c2, krylovDelta, e0, Matrix.cons_val_two]
    field_simp [krylovDelta_ne n]
    ring

/-- The two first columns are definitional; the last is `r_reconstruct`. -/
theorem T_mul_U_succ (n : ℕ) :
    T n * U (n + 1) = U n * C n := by
  ext i j
  fin_cases j
  · simp [U, Ucol, C, p, Matrix.mul_apply, Matrix.mulVec,
      Fin.sum_univ_three, e0, Matrix.cons_val_two]
  · simp [U, Ucol, C, p, q, Matrix.mul_apply, Matrix.mulVec,
      Fin.sum_univ_three, e0, Matrix.cons_val_two]
  · have h := congrFun (r_reconstruct n) i
    simpa [U, Ucol, C, r, Matrix.mul_apply, Matrix.mulVec,
      Fin.sum_univ_three, Matrix.cons_val_two, add_assoc] using h

/-- Simplified positive formula for the leading companion coefficient.
This follows from determinants and is useful for the pull matrix. -/
theorem c0_closed (n : ℕ) :
    c0 n =
      ((n : ℝ) + 2)^2 * (2 * (n : ℝ) + 7) * krylovPoly (n + 1) /
        (((n : ℝ) + 4) * ((n : ℝ) + 5) *
          (2 * (n : ℝ) + 9) * krylovPoly n) := by
  have h := congrArg Matrix.det (T_mul_U_succ n)
  rw [Matrix.det_mul, Matrix.det_mul, det_T, det_U, det_U, det_C] at h
  have hDn : krylovPoly n ≠ 0 := (krylovPoly_pos n).ne'
  have hDnp : krylovPoly (n + 1) ≠ 0 := (krylovPoly_pos (n + 1)).ne'
  field_simp [n1_ne n, n2_ne n, n3_ne n, n4_ne n, n5_ne n,
    two_n3_ne n, two_n5_ne n, two_n7_ne n, two_n9_ne n,
    hDn, hDnp] at h ⊢
  linear_combination h

lemma c0_pos (n : ℕ) : 0 < c0 n := by
  rw [c0_closed]
  have h0 := krylovPoly_pos n
  have h1 := krylovPoly_pos (n + 1)
  positivity

lemma c0_ne (n : ℕ) : c0 n ≠ 0 := (c0_pos n).ne'

/-- Literal `C(n)^{-T}`; no use of `Matrix.inv`. -/
def pullMatrix (n : ℕ) : Mat3 :=
  !![-c1 n / c0 n, -c2 n / c0 n, 1 / c0 n;
     1,             0,             0;
     0,             1,             0]

/-- Exact inverse-transpose certificate used by the graph transform. -/
theorem pullMatrix_mul_Ctranspose (n : ℕ) :
    pullMatrix n * (C n).transpose = (1 : Mat3) := by
  ext i j
  fin_cases i <;> fin_cases j
  all_goals
    simp [pullMatrix, C, Matrix.mul_apply, Fin.sum_univ_three,
      Matrix.one_apply, Matrix.cons_val_two]
    field_simp [c0_ne n]
    ring

/-- The exact initial Krylov matrix. -/
theorem U_zero :
    U 0 =
      !![1, 152/5, 195477/175;
         0, 1723/90, 1963751/2800;
         0, -143/18, -165201/560] := by
  ext i j
  fin_cases i <;> fin_cases j <;>
    norm_num [U, Ucol, p, q, e0, T, Matrix.mulVec,
      Fin.sum_univ_three, Matrix.cons_val_two]

/-- Requested exact value.  Transposition does not change determinant. -/
theorem det_U_zero_transpose :
    Matrix.det (U 0).transpose = -(382493 : ℝ) / 5040 := by
  rw [Matrix.det_transpose, U_zero, Matrix.det_fin_three]
  norm_num [Matrix.cons_val_two]

lemma det_U_zero_transpose_ne : Matrix.det (U 0).transpose ≠ 0 := by
  rw [det_U_zero_transpose]
  norm_num

end MillerKrylov
```

## Optional column-2 terminal alignment

This is the corrected Q6945 coordinate order.  Set `N=h+2`; then the two reverse steps are exactly `pullMatrix h` followed by `pullMatrix (h+1)` on the terminal vector.

```lean
namespace MillerKrylov

/-- Homogeneous chart convention used by the pullback theorem. -/
def lift (z : Chart) : Vec3 := ![1, z 0, z 1]

def u2Num (N : ℕ) : ℝ :=
  (N : ℝ)^2 * (2*(N : ℝ)+1) * (2*(N : ℝ)+3) *
    (48*(N : ℝ)^4 + 306*(N : ℝ)^3 + 715*(N : ℝ)^2 +
      729*(N : ℝ) + 275)

def R2 (N : ℕ) : ℝ :=
  6528*(N : ℝ)^8 + 54480*(N : ℝ)^7 + 187144*(N : ℝ)^6 +
    344616*(N : ℝ)^5 + 371338*(N : ℝ)^4 + 239883*(N : ℝ)^3 +
    91396*(N : ℝ)^2 + 19200*(N : ℝ) + 1800

def u2 (N : ℕ) : ℝ := u2Num N / R2 N

def terminal2Chart (h : ℕ) : Chart := ![u2 (h + 2), 0]

def terminal2Vec (h : ℕ) : Vec3 :=
  pullMatrix h *ᵥ
    (pullMatrix (h + 1) *ᵥ ((U (h + 2)).transpose *ᵥ e2))

/-- Defining the scalar as the first coordinate avoids any normalization
ambiguity.  Q6951's factored scalar is a rewrite theorem for this definition. -/
def terminal2Scale (h : ℕ) : ℝ := terminal2Vec h 0

lemma R2_pos (N : ℕ) : 0 < R2 N := by
  unfold R2
  positivity

/-- One heavy rational identity (coordinate 1); coordinates 0 and 2 are
respectively definitional and forced by the two companion shifts. -/
set_option maxHeartbeats 0 in
theorem terminal2_formula_shift (h : ℕ) :
    terminal2Vec h = terminal2Scale h • lift (terminal2Chart h) := by
  ext i
  fin_cases i
  · simp [terminal2Scale, lift, Matrix.cons_val_two]
  · simp [terminal2Vec, terminal2Scale, terminal2Chart, lift,
      pullMatrix, U, Ucol, p, q, e0, e2, u2, u2Num, R2,
      Matrix.mulVec, Fin.sum_univ_three, Matrix.cons_val_two]
    field_simp [c0_ne h, c0_ne (h + 1),
      krylovDelta_ne h, krylovDelta_ne (h + 1),
      krylovDelta_ne (h + 2), krylovDelta_ne (h + 3),
      R2_pos (h + 2)]
    ring
  · simp [terminal2Vec, terminal2Scale, terminal2Chart, lift,
      pullMatrix, U, Ucol, e0, e2,
      Matrix.mulVec, Fin.sum_univ_three, Matrix.cons_val_two]

/-- Unshifted statement, with `N-2` and `N-1` in the required order. -/
theorem terminal2_formula (N : ℕ) (hN : 2 ≤ N) :
    pullMatrix (N - 2) *ᵥ
        (pullMatrix (N - 1) *ᵥ ((U N).transpose *ᵥ e2)) =
      terminal2Scale (N - 2) • lift ![u2 N, 0] := by
  obtain ⟨h, rfl⟩ := Nat.exists_eq_add_of_le hN
  simpa [Nat.add_sub_cancel_left, Nat.add_sub_cancel] using
    terminal2_formula_shift h

end MillerKrylov
```
