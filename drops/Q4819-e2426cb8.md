\begin{lemma}[Selected reciprocal fibre: consecutive-gap reduction]
\label{lem:ad-selected-reciprocal-gap-reduction}
Fix pairwise distinct shell primes $p,q,p'\asymp N$, a circular sign
$\sigma\in\{\pm1\}$, and a determinant $\delta$.  Put
\[
 \mathcal R_\delta
 =\left\{\ell\asymp N:
 \sigma\delta\overline{p'\ell}\pmod p\in\mathcal Z_p,
 \quad
 -\sigma\delta\overline{pq}\pmod\ell\in\mathcal Z_\ell
 \right\},
 \qquad R(\delta)=|\mathcal R_\delta|.
\]
For $\ell\in\mathcal R_\delta$ write
\[
 r_\delta(\ell)
 =\sigma\delta\overline{p'\ell}\pmod p\in\mathcal Z_p.
\]
After a bounded subdivision of the fixed ratio boxes, the map
$\ell\mapsto r_\delta(\ell)$ is injective.  Let $\mathcal C_H(\delta)$
be the number of non-circular consecutive gaps of length at most $H$
among the ordered values $r_\delta(\ell)$.  Then, uniformly for
$1\le H<p$,
\[
 R(\delta)\ll_v1+\frac pH+\mathcal C_H(\delta).
\tag{SG1}
\]
Consequently, if $R(\delta)>N^\tau$ and
$H=C_vN^{1-\tau}$ with a sufficiently large fixed constant, then
\[
 \mathcal C_H(\delta)\gg_v N^\tau.
\tag{SG2}
\]
\end{lemma}

\begin{proof}
On one injective subbox list the distinct labels as
$0\le r_1<\cdots<r_m<p$.  Every consecutive gap not counted by
$\mathcal C_H$ is larger than $H$, while the sum of the non-circular
gaps is less than $p$.  Hence
$m-1-\mathcal C_H\le p/H$.  Summing over the bounded number of subboxes
proves \textup{(SG1)}, and \textup{(SG2)} follows by the stated choice
of $H$.
\end{proof}
