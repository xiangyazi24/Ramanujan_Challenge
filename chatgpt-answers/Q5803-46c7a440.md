# 11. Dependency-free exact denominator audit

This standard-library checker verifies the closed factorial formula for the canonical backward A_M transport on the hostile full-margin rows. It is an audit of the forbidden transfer implementation, not a substitute for the asymptotic proof.

```python
#!/usr/bin/env python3

from math import factorial, prod

ROWS = (
    (200, 128, 63),
    (272, 180, 63),
    (300, 180, 57),
    (321, 168, 53),
)


def transfer_denominator(moment, residue, steps):
    direct = abs(prod(
        (x + 1) * (x - 2 * moment) ** 2
        for x in range(residue - steps, residue)
    ))
    closed = (
        factorial(residue) // factorial(residue - steps)
        * (
            factorial(2 * moment - residue + steps)
            // factorial(2 * moment - residue)
        ) ** 2
    )
    assert direct == closed
    return direct


for n, D, N in ROWS:
    M = n - 1
    m = min(D - ((n - 1) // 2), n - D - N + 1)
    d0 = D - m + 1
    L = N + m - 2
    r = M - d0
    h = L + 1

    assert 0 < h <= r < M

        "denominator_bits", den.bit_length(),
```
