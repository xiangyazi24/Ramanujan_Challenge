#!/usr/bin/env python3
from collections import defaultdict
from functools import lru_cache
from math import comb, gcd
from pathlib import Path
import struct


def C(n, k):
    return comb(n, k) if 0 <= k <= n else 0


def delta(Y, d, order):
    return sum((-1) ** (order - i) * C(order, i) * Y[d + i]
               for i in range(order + 1))


def weight(d, L, i):
    return (-1) ** i * C(d + i, i) * C(d + L + 1, L - i)


def G_from_dict(Y, d, L):
    return sum(weight(d, L, i) * Y[d + i] for i in range(L + 1))


def row(d, L, lo, hi):
    out = [0] * (hi - lo + 1)
    for i in range(L + 1):
        out[d + i - lo] = weight(d, L, i)
    return out


def rank_mod(rows, p):
    A = [[x % p for x in r] for r in rows]
    rank = 0
    ncol = len(A[0]) if A else 0
    for col in range(ncol):
        pivot = next((r for r in range(rank, len(A)) if A[r][col]), None)
        if pivot is None:
            continue
        A[rank], A[pivot] = A[pivot], A[rank]
        inv = pow(A[rank][col], -1, p)
        A[rank] = [(x * inv) % p for x in A[rank]]
        for r in range(len(A)):
            if r != rank and A[r][col]:
                a = A[r][col]
                A[r] = [(A[r][c] - a * A[rank][c]) % p
                        for c in range(ncol)]
        rank += 1
    return rank


def primes_upto(n):
    out = []
    for q in range(2, n + 1):
        if all(q % a for a in range(2, int(q ** 0.5) + 1)):
            out.append(q)
    return out


def audit_identities():
    Y = {j: j ** 5 - 7 * j ** 2 + 11 for j in range(200)}
    for d in range(3, 15):
        for L in range(0, 8):
            g = G_from_dict(Y, d, L)
            assert sum(weight(d, L, i) for i in range(L + 1)) == 1
            if L < 7:
                vertical = G_from_dict(Y, d, L + 1) - g
                assert vertical == ((-1) ** (L + 1)
                                    * C(d + L + 1, L + 1)
                                    * delta(Y, d, L + 1))
            horizontal = g - G_from_dict(Y, d + 1, L)
            assert horizontal == ((-1) ** (L + 1)
                                  * C(d + L + 1, L)
                                  * delta(Y, d, L + 1))
            if L < 7:
                assert ((d + 1) * horizontal
                        == (L + 1) * vertical)


def audit_rectangle_smith(D, R, A, B, q):
    assert q > B
    j = q - 1
    assert D + R <= j <= D + A
    lo = D
    hi = D + R + B

    boundary = [row(D + t, A, lo, hi) for t in range(R + 1)]
    boundary += [row(D + R, L, lo, hi) for L in range(A + 1, B + 1)]
    k = R + (B - A) + 1
    assert len(boundary) == k

    # Every original boundary row is the same coordinate row modulo q.
    ej = [0] * (hi - lo + 1)
    ej[j - lo] = 1
    assert all([x % q for x in r] == ej for r in boundary)
    assert rank_mod(boundary, q) == 1

    transformed = [boundary[0]]
    for t in range(R):
        diff = [boundary[t][c] - boundary[t + 1][c]
                for c in range(len(ej))]
        assert all(x % q == 0 for x in diff)
        transformed.append([x // q for x in diff])

    right_offset = R
    for u in range(B - A):
        low = boundary[right_offset + u]
        high = boundary[right_offset + u + 1]
        diff = [high[c] - low[c] for c in range(len(ej))]
        assert all(x % q == 0 for x in diff)
        transformed.append([x // q for x in diff])

    assert rank_mod(transformed, q) == k


def audit_generic_countermodel():
    D, R, A, B = 10, 3, 5, 7
    j = D + R
    T = 1009
    Y = {m: m + 1 + (T if m == j else 0) for m in range(100)}
    vals = [G_from_dict(Y, d, L)
            for d in range(D, D + R + 1)
            for L in range(A, B + 1)]
    assert vals and all(v % T == 0 for v in vals)
    assert any(v for v in vals)


def binom_table_mod_prime(p):
    fact = [1] * p
    for i in range(1, p):
        fact[i] = fact[i - 1] * i % p
    invfact = [1] * p
    invfact[-1] = pow(fact[-1], -1, p)
    for i in range(p - 1, 0, -1):
        invfact[i - 1] = invfact[i] * i % p
    return fact, invfact


def small_binom_mod(n, k, p, fact, invfact):
    if k < 0 or k > n:
        return 0
    return fact[n] * invfact[k] % p * invfact[n - k] % p


def lucas_binom_mod(n, k, p, fact, invfact):
    if k < 0 or k > n:
        return 0
    out = 1
    while n or k:
        ni, ki = n % p, k % p
        if ki > ni:
            return 0
        out = out * small_binom_mod(ni, ki, p, fact, invfact) % p
        n //= p
        k //= p
    return out


def shell_mod(M, d, p):
    a = M // d
    fact, invfact = binom_table_mod_prime(p)
    out = 0
    for t in range(M + 1):
        cm = lucas_binom_mod(M, t, p, fact, invfact)
        xp = sum(lucas_binom_mod(M, M - t + d * u,
                                 p, fact, invfact)
                 for u in range(-a, a + 1)) % p
        yz = sum(lucas_binom_mod(2 * M - t, M - t + d * v,
                                 p, fact, invfact)
                 for v in range(-a, a + 1)) % p
        out = (out + cm * xp * yz * yz) % p
    return out


@lru_cache(None)
def shell_exact(M, d):
    a = M // d
    out = 0
    for t in range(M + 1):
        xp = sum(C(M, M - t + d * u) for u in range(-a, a + 1))
        yz = sum(C(2 * M - t, M - t + d * v)
                 for v in range(-a, a + 1))
        out += C(M, t) * xp * yz * yz
    return out


def G_shell_exact(M, d, L):
    return sum(weight(d, L, i) * shell_exact(M, d + i)
               for i in range(L + 1))

def load_hits(path):
    by_n = defaultdict(list)
    data = Path(path).read_bytes()
    assert len(data) % 8 == 0
    for off in range(0, len(data), 8):
        p, r = struct.unpack_from("<II", data, off)
        by_n[p + r].append(p)
    return by_n
