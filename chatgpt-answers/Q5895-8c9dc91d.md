def sig(f, k):
    return K(f)(n=n+k)


def coeff_of_product(A, B, t):
    out = K(0)
    for i in range(A.order()+1):
        j = t-i
        if 0 <= j <= B.order():
            out += K(A[i]) * sig(B[j], i)
    return out

# Seven coefficient-by-coefficient checks.
for t in range(AM.order()+Uop.order()+1):
    assert coeff_of_product(AM, Uop, t) == coeff_of_product(Qop, Lop, t)


def gcd_many_local(values):
    g = ZZ(0)
    for x in values:
        g = gcd(g, abs(ZZ(x)))
    return abs(g)


def d2(M):
    assert M.nrows() == 2
    return gcd_many_local(M.minors(2))


def col_matrix(columns):
    return matrix(ZZ, [list(vector(ZZ, c)) for c in columns]).transpose()


def cleared_time_coefficients(t):
    vals = [K(AM[i])(n=t) for i in range(3)] \
         + [K(Qop[i])(n=t) for i in range(2)]
    den = lcm(v.denominator() for v in vals)
    ints = [ZZ(den*v) for v in vals]
    return den, tuple(ints[:3]), tuple(ints[3:])


def endpoint_pair(values, d, length, D):
    return vector(ZZ, (
        newton_carrier(values, d, length),
        newton_carrier(values, D, length),
    ))


def phi_values(moment, nodes):
    neighbours = list(range(min(nodes)-1, max(nodes)+2))
    row = shell_batch(moment, neighbours)
    return {x: row[x-1] + row[x+1] for x in nodes}


def operator_shell_values(operator, t, nodes):
    rows = {
        j: phi_values(t+j, nodes)
        for j in range(operator.order()+1)
    }
    out = {}
    for x in nodes:
        value = sum(K(operator[j])(n=t) * rows[j][x]
                    for j in range(operator.order()+1))
        assert value.denominator() == 1
        out[x] = ZZ(value)
    return out


def shifted_W_values(t, nodes):
    z = operator_shell_values(Uop, t, nodes)
    y = shell_batch(t-1, nodes)
    mu = ZZ(Mpoly(n=t))
    return {x: z[x] - mu*y[x] for x in nodes}


def syzygy_margin_record(global_index, D, N, margin):
    d = D-margin+1
    length = N+margin-2
    nodes = list(range(d, D+length+1))
    QI = prod(prime_range(D+1, D+N+1))

    # Visible columns.
    Wrows = {k: shifted_W_values(global_index+k, nodes) for k in range(6)}
    Lrows = {
        k: operator_shell_values(Lop, global_index+k, nodes)
        for k in range(5)
    }
    Z = []
    for k in range(6):
        raw = endpoint_pair(Wrows[k], d, length, D)
        assert all(x % QI == 0 for x in raw)
        Z.append(raw / QI)
    Rghost = []
    for k in range(5):
        raw = endpoint_pair(Lrows[k], d, length, D)
        assert all(x % QI == 0 for x in raw)
        Rghost.append(raw / QI)

    Rcols = [Z[0]] + Rghost
    Rmat = col_matrix(Rcols)
    delta = d2(Rmat)

    # Four exact projected Ore relations.
    Gamma = []
    Bcols = []
    q1_values = []
    for k in range(4):
        den, aa, qq = cleared_time_coefficients(global_index+k)
        q0, q1 = qq
        q1_values.append(q1)
        gamma = q0*Rghost[k] + q1*Rghost[k+1]

        yrows = {
            r: shell_batch(global_index+k+r-1, nodes)
            for r in range(3)
        }
        defect_values = {
            x: sum(aa[r] * ZZ(Mpoly(n=global_index+k+r)) * yrows[r][x]
                   for r in range(3))
            for x in nodes
        }
        defect_raw = endpoint_pair(defect_values, d, length, D)
        assert all(x % QI == 0 for x in defect_raw)
        B = defect_raw / QI
        rhs = sum((aa[r]*Z[k+r] for r in range(3)), vector(ZZ,[0,0])) + B
        assert gamma == rhs
        Gamma.append(gamma)
        Bcols.append(B)

    Gcols = [Z[0], Rghost[0]] + Gamma
    Gmat = col_matrix(Gcols)
    epsilon = d2(Gmat)

    # Explicit coefficient transform.
    C = matrix(ZZ, 6, 6, 0)
    C[0,0] = 1
    C[1,1] = 1
    for k in range(4):
        _, _, qq = cleared_time_coefficients(global_index+k)
        q0, q1 = qq
        C[k+1,k+2] = q0
        C[k+2,k+2] = q1
    assert Gmat == Rmat*C
    detC = prod(q1_values)
    assert C.det() == detC
    assert delta.divides(epsilon)
    assert epsilon.divides(detC^2 * delta)

    # The exact two-time-Casoratian element.
    theta_visible = Z[0][0]*Gamma[0][1] - Z[0][1]*Gamma[0][0]
    _, aa0, _ = cleared_time_coefficients(global_index)
    omega01 = Z[0][0]*Z[1][1] - Z[0][1]*Z[1][0]
    omega02 = Z[0][0]*Z[2][1] - Z[0][1]*Z[2][0]
    beta = Z[0][0]*Bcols[0][1] - Z[0][1]*Bcols[0][0]
    assert theta_visible == aa0[1]*omega01 + aa0[2]*omega02 + beta
    assert delta.divides(theta_visible)

    return {
        "margin": margin,
        "delta": delta,
        "epsilon": epsilon,
        "detC": detC,
        "theta": theta_visible,
        "Z0": Z[0],
        "Gamma0": Gamma[0],
        "Rmat": Rmat,
        "Gmat": Gmat,
    }


def adjacent_check(global_index, D, N):
    moment = global_index-1
    m = min(D-moment//2, moment-D-N+2)
    prev = syzygy_margin_record(global_index, D, N, m-1)
    cur  = syzygy_margin_record(global_index, D, N, m)

    common_delta = gcd(prev["delta"], cur["delta"])
    assert common_delta.divides(gcd(prev["theta"], cur["theta"]))

    dZ = cur["Z0"] - prev["Z0"]
    dG = cur["Gamma0"] - prev["Gamma0"]
    rhs = (dZ[0]*prev["Gamma0"][1] - dZ[1]*prev["Gamma0"][0]) \
        + (prev["Z0"][0]*dG[1] - prev["Z0"][1]*dG[0]) \
        + (dZ[0]*dG[1] - dZ[1]*dG[0])
    assert cur["theta"] - prev["theta"] == rhs
    return prev, cur
