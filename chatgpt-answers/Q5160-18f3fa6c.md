# 8. SageMath verification code

```python
from sage.all import *


def S(n):
    return ZZ(sum(
        binomial(n, k)^2
        * binomial(n + k, n)
        * binomial(n + 2*k, n)
        for k in range(n + 1)
    ))


vals = [S(n) for n in range(21)]
for n, value in enumerate(vals):
    print(n, value)


def Q(n):
    return 946*n^2 - 731*n + 153


def P(n):
    return (
        104060*n^6 + 127710*n^5 + 12788*n^4
        - 34525*n^3 - 8482*n^2 + 3298*n + 1071
    )


def R(n):
    return (
        3784*n^5 - 1032*n^4 - 1925*n^3
        + 853*n^2 + 328*n - 184
    )


# Verify the minimal four-term recurrence for S_n.
for n in range(2, 20):
    residual = (
        2*Q(n)*(2*n + 1)*(n + 1)^3*vals[n + 1]
        - 2*P(n)*vals[n]
        + 2*n*R(n)*vals[n - 1]
        - Q(n + 1)*n*(n - 1)^3*vals[n - 2]
    )
    assert residual == 0


# Full AESZ coefficients.
avec = [binomial(2*n, n)*vals[n] for n in range(21)]


def c3(n):
    return (n + 3)^4*(946*n^2 + 3053*n + 2475)


def c2(n):
    return -2*(
        104060*n^6 + 1376430*n^5 + 7533488*n^4
        + 21825779*n^3 + 35282480*n^2
        + 30160606*n + 10648707
    )


def c1(n):
    return 4*(2*n + 3)*(
        3784*n^5 + 36808*n^4 + 141179*n^3
        + 267255*n^2 + 250336*n + 93060
    )


def c0(n):
    return -4*(n + 1)^2*(2*n + 1)*(2*n + 3)*(
        946*n^2 + 4945*n + 6474
    )


for n in range(18):
    assert (
        c3(n)*avec[n + 3]
        + c2(n)*avec[n + 2]
        + c1(n)*avec[n + 1]
        + c0(n)*avec[n]
    ) == 0


# Edge-polynomial irreducibility: proves minimal shift order 3.
Rlam.<lam> = PolynomialRing(QQ)
edge = 4*lam^3 - 220*lam^2 + 8*lam - 1
assert edge.is_irreducible()
print("edge polynomial:", edge)


# Hypergeometric forms.
# Sage's symbolic hypergeometric function can be used to check small n,
# but the binomial sum above is faster and exact.


# Cooper sequence and centered transform.
def cooper_terms(N):
    T = [QQ(1), QQ(4)]
    for n in range(1, N - 1):
        tm2 = T[n - 2] if n >= 2 else QQ(0)
        nxt = (
            2*(2*n + 1)*(5*n^2 + 5*n + 2)*T[n]
            - 8*n*(7*n^2 + 1)*T[n - 1]
            + 22*n*(2*n - 1)*(n - 1)*tm2
        ) / (n + 1)^3
        T.append(QQ(nxt))
    return T


T = cooper_terms(10)


def centered(T, n, alpha):
    return QQ(sum(
        binomial(2*n, m)*(-2)^(2*n - m)*T[m]
        for m in range(2*n + 1)
    )) / alpha^n


print("Cooper:", T[:5])
print("K_4:", [centered(T, n, 4) for n in range(3)])
print("K_256:", [centered(T, n, 256) for n in range(3)])
```
