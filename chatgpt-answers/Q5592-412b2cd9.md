# 10. Self-contained exact-arithmetic checker and diagnostic plan

The following script uses only Python integers. It does not require factoring b_n, and it verifies every symbolic identity above. An optional sympy.factorint call may be added only for displaying the residual small factors.

```python
from __future__ import annotations

from dataclasses import dataclass
from functools import reduce
from math import gcd, log, log2
from typing import Iterable


def xgcd(a: int, b: int) -> tuple[int, int, int]:
    """Return g,s,t with g>=0 and s*a+t*b=g."""
    old_r, r = a, b
    old_s, s = 1, 0
    old_t, t = 0, 1
    while r:
        q = old_r // r
        old_r, r = r, old_r - q * r
        old_s, s = s, old_s - q * s
        old_t, t = t, old_t - q * t
    if old_r < 0:
        return -old_r, -old_s, -old_t
    return old_r, old_s, old_t


def bezout_many(values: list[int]) -> tuple[int, list[int]]:
    """Return g and z with sum(z[i]*values[i])=g=gcd(values)."""
    g = 0
    coeffs: list[int] = []
    for value in values:
        gg, s, t = xgcd(g, value)
        coeffs = [s * c for c in coeffs] + [t]
        g = gg
    assert sum(c * v for c, v in zip(coeffs, values)) == g
    return g, coeffs


def gcd_many(values: Iterable[int]) -> int:
    return reduce(gcd, (abs(v) for v in values), 0)


def vp(x: int, p: int) -> int:
    if x == 0:
        raise ValueError("v_p(0) is not used in this checker")
    x = abs(x)
    e = 0
    while x % p == 0:
        x //= p
        e += 1
    return e


def primes_upto(limit: int) -> list[int]:
    if limit < 2:
        return []
    sieve = bytearray(b"\x01") * (limit + 1)
    sieve[0:2] = b"\x00\x00"
    for p in range(2, int(limit**0.5) + 1):
        if sieve[p]:
            sieve[p * p : limit + 1 : p] = b"\x00" * (
                (limit - p * p) // p + 1
            )
    return [p for p in range(2, limit + 1) if sieve[p]]


def log_int(x: int) -> float:
    """Natural log without converting a huge integer directly to float."""
    if x <= 0:
        raise ValueError("log_int expects a positive integer")
    bits = x.bit_length()
    shift = max(0, bits - 53)
    return log(x >> shift) + shift * log(2.0)


@dataclass
class Row:
    k: int
    c: int
    x: int
    y: int
    rho: int
    mu: int
    sigma: int
    carrier: int


@dataclass
class Audit:
    n: int
    H: int
    L: int
    G: int
    a: int | None
    beta: int | None
    m: int | None
    U: int
    m_sat: int | None
    hard: int | None
    target_rad: int
    upper_rad: int
    band_rad: int


def apery_w_row(n: int) -> list[int]:
    """W_k=binom(n,k)binom(n+k,k), 0<=k<=n, by exact recurrence."""
    W = [1]
    for k in range(n):
        numerator = W[-1] * (n - k) * (n + k + 1)
        denominator = (k + 1) ** 2
        assert numerator % denominator == 0
        W.append(numerator // denominator)
    return W


def audit_n(n: int) -> Audit:
    if n < 12:
        raise ValueError("the requested range starts at n=12")

    H = n // 3
    L = n // 2
    W = apery_w_row(n)
    t = [w * w for w in W]

    S: list[int] = []
    total = 0
    for value in t:
        total += value
        S.append(total)
    b_n = S[n]

    rows: list[Row] = []
    for k in range(H + 1, L):
        A = (n - k) * (n + k + 1)
        B = (k + 1) ** 2
        c = gcd(A, B)
        x = B // c
        y = A // c
        assert gcd(x, y) == 1

        d = gcd(W[k], W[k + 1])
        assert W[k] == d * x
        assert W[k + 1] == d * y
        assert gcd(t[k], t[k + 1]) == d * d

        rho = y * y - x * x
        mu = (W[k] * W[k + 1] // d) ** 2
        T_k = S[k] - S[H]
        sigma = mu - rho * T_k
        carrier = sigma - rho * S[H]

        raw_minor = S[k] * S[k] - S[k - 1] * S[k + 1]
        assert raw_minor > 0
        assert raw_minor % gcd(t[k], t[k + 1]) == 0
        assert carrier == raw_minor // gcd(t[k], t[k + 1])
        assert carrier > 0

        rows.append(Row(k, c, x, y, rho, mu, sigma, carrier))

    if not rows:
        raise AssertionError("no carrier row")

    G = gcd_many(row.carrier for row in rows)
    U = gcd_many(W[j] for j in range(H + 1, L + 1))
    U2 = U * U
    assert all(row.sigma % U2 == 0 for row in rows)

    # One-row boundary cases have no scalar coefficient eliminant.
    if len(rows) == 1:
        a = beta = m = m_sat = hard = None
    else:
        rhos = [row.rho for row in rows]
        a, z = bezout_many(rhos)
        beta = sum(zi * row.sigma for zi, row in zip(z, rows))
        reduced_slopes = [rho // a for rho in rhos]
        hs = [
            row.sigma - reduced * beta
            for row, reduced in zip(rows, reduced_slopes)
        ]
        m = gcd_many(hs)

        assert G == gcd(abs(a * S[H] - beta), m)

        # All-pair determinant identity: gcd(det)=a*m.
        det_gcd = 0
        for i in range(len(rows)):
            for j in range(i + 1, len(rows)):
                det = (
                    rows[i].rho * rows[j].sigma
                    - rows[j].rho * rows[i].sigma
                )
                det_gcd = gcd(det_gcd, abs(det))
        assert det_gcd == a * m

        # Adjacent closed formula.
        N = n * (n + 1)
        for left, right in zip(rows, rows[1:]):
            k = left.k
            assert right.k == k + 1
            xvar = k + 1
            hminus = (2 * xvar + 1) * N + xvar * (xvar + 1)
            hplus = (
                (2 * xvar * xvar + 2 * xvar + 1) * N
                - xvar * (xvar + 1) * (2 * xvar * xvar - 1)
            )
            numerator = W[k + 1] ** 2 * hminus * hplus
            denominator = left.c**2 * right.c**2
            assert numerator % denominator == 0
            closed = numerator // denominator
            direct = left.rho * right.carrier - right.rho * left.carrier
            assert direct == closed
            assert closed % G == 0

        assert beta % U2 == 0
        assert m % U2 == 0
        m_sat = m // U2
        hard = gcd(a * S[H], U2)
        assert (hard * m_sat) % G == 0

        if len(rows) >= 4:
            assert 12 % a == 0

    primes = primes_upto(n + H + 1)
    target_primes = [p for p in primes if 2 * p > n and p <= n and b_n % p == 0]
    upper_primes = [p for p in primes if n < p <= n + H + 1 and b_n % p == 0]
    band_primes = [p for p in primes if L < p <= n + H + 1 and b_n % p == 0]

    target_rad = reduce(lambda x, y: x * y, target_primes, 1)
    upper_rad = reduce(lambda x, y: x * y, upper_primes, 1)
    band_rad = reduce(lambda x, y: x * y, band_primes, 1)

    if n >= 30:
        for p in primes:
            if not (L < p <= n + H + 1):
                continue
            assert U % p == 0
            assert a is not None and a % p != 0
            assert (G % p == 0) == (S[H] % p == 0)
            assert (S[H] % p == 0) == (b_n % p == 0)

            A0 = n - H
            B0 = n + H + 1
            expected = int(A0 <= p <= B0) + int(A0 <= 2 * p <= B0)
            assert vp(U, p) == expected

        assert G % band_rad == 0
        assert band_rad == target_rad * upper_rad

    return Audit(
        n=n,
        H=H,
        L=L,
        G=G,
        a=a,
        beta=beta,
        m=m,
        U=U,
        m_sat=m_sat,
        hard=hard,
        target_rad=target_rad,
        upper_rad=upper_rad,
        band_rad=band_rad,
    )


def scan(N: int = 1000) -> None:
    print(
        "n,e,logG/n,loga,logU2/n,logm/n,logmsat/n,loghard/n,"
        "target_log/n,upper_log/n"
    )
    for n in range(12, N + 1):
        A = audit_n(n)

        def ratio(x: int | None) -> str:
```
