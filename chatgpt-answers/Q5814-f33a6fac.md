def gcd_many(values):
    g = ZZ(0)
    for value in values:
        g = gcd(g, ZZ(value))
    return abs(g)


def xgcd_many(values):
    """Return g, coeffs with sum coeffs[i]*values[i] = g."""
    coeffs = []
    g = ZZ(0)
    for value in map(ZZ, values):
        gg, s, t = xgcd(g, value)
        coeffs = [s * coefficient for coefficient in coeffs] + [t]
        g = gg
    if g < 0:
        g = -g
        coeffs = [-coefficient for coefficient in coeffs]
    assert sum(c * ZZ(a) for c, a in zip(coeffs, values)) == g
    return g, coeffs


def origin_values(index):
    values = [ZZ(polynomial(n=index)) for polynomial in ORIGIN_U]
    values.append(ZZ(ORIGIN_MU(n=index)))
    return values


def origin_content(index):
    return gcd_many(origin_values(index))


def polynomial_bezout_constant():
    """One fixed C_U divisible by every specialized origin content."""
    values = list(ORIGIN_U) + [ORIGIN_MU]
    ring = values[0].parent()
    g = values[0]
    coeffs = [ring.one()]
    for value in values[1:]:
        gg, s, t = g.xgcd(value)
        coeffs = [s * coefficient for coefficient in coeffs] + [t]
        g = gg
    assert g.degree() == 0 and g != 0
    coeffs = [coefficient / g[0] for coefficient in coeffs]
    denominators = [
        QQ(a).denominator()
        for coefficient in coeffs
        for a in coefficient.list()
    ]

    integer_coeffs = [denominator * coefficient for coefficient in coeffs]

        coefficient * value

def G(d, length, values):
    """Q5808 convention: nodes d-1,...,d+length-1."""
    return ZZ(sum(
        (-1)^offset
        * binomial(d - 1 + offset, offset)
        * binomial(d + length, length - offset)
        * values[d - 1 + offset]
        for offset in range(length + 1)
    ))


def G_row(d, length):
    return {
        d - 1 + offset:
        ZZ((-1)^offset
           * binomial(d - 1 + offset, offset)
           * binomial(d + length, length - offset))
        for offset in range(length + 1)
    }


def diamond_content(d, length):
    left = G_row(d, length)
    right = G_row(d + 1, length)
    nodes = sorted(set(left) | set(right))
    content = gcd_many(
        left.get(node, 0) - right.get(node, 0) for node in nodes
    )
    # Exact off-by-one check.
    assert content == binomial(d + length, length)
    return content


def normalizer(D, N, margin):
    d = D - margin + 1
    length = N + margin - 2
    contents = [
        diamond_content(d + shift, length)
        for shift in range(margin - 1)
    ]
    closed = [
        binomial(d + shift + length, length)
        for shift in range(margin - 1)
    ]
    assert contents == closed
    return gcd_many(contents)


def maximal_margin(index, D, N):
    return min(D - ((index - 1) // 2), index - D - N + 1)


def nodes_of_G(d, length):
    return set(range(d - 1, d + length))


def required_base_nodes(D, N, margins):
    nodes = set()
    for margin in margins:
        d = D - margin + 1
        length = N + margin - 2
        for corner in (
            (d, length),
            (D, length),
            (d + 1, length - 1),
            (D, length - 1),
        ):
            nodes |= nodes_of_G(*corner)
    return sorted(nodes)

def vector_carrier(d, length, arrays, primitive=True):
    second = arrays["W"] if primitive else arrays["Z"]
    return vector(ZZ, [
        G(d, length, arrays["Y"]),
        G(d, length, second),
    ])


def det2(left, right):
    return ZZ(left[0] * right[1] - left[1] * right[0])


def endpoint_value(index, D, N, margin, arrays, primitive=True):
    d = D - margin + 1
    length = N + margin - 2
    left = vector_carrier(d, length, arrays, primitive=primitive)
    right = vector_carrier(D, length, arrays, primitive=primitive)
    numerator = det2(left, right)
    g = normalizer(D, N, margin)
    assert numerator % g == 0
    return numerator // g, numerator, g


def forward_difference(values, start, order):
    return ZZ(sum(
        (-1)^(order - shift)
        * binomial(order, shift)
        * values[start + shift]
        for shift in range(order + 1)
    ))


def boundary_data(D, N, margin, arrays):
    d = D - margin + 1
    length = N + margin - 2
    A = vector_carrier(d, length, arrays)
    B = vector_carrier(d + 1, length - 1, arrays)
    C = vector_carrier(D, length, arrays)
    E = vector_carrier(D, length - 1, arrays)
    P = A - B
    Rv = C - E

    for coordinate, values in enumerate((arrays["Y"], arrays["W"])):
        p_expected = (
            (-1)^length
            * binomial(d + length, length)
            * forward_difference(values, d - 1, length)
        )
        r_expected = (
            (-1)^length
            * binomial(D + length - 1, length)
            * forward_difference(values, D - 1, length)
        )
        assert P[coordinate] == p_expected
        assert Rv[coordinate] == r_expected

    delta = det2(A, C) - det2(B, E)
    assert delta == det2(P, C) + det2(B, Rv)

    matrix_boundary = matrix(ZZ, [
        [-P[0], P[1], -B[0], B[1], 0],
        [1, 0, 0, 0, -C[1]],
        [0, 1, 0, 0, -C[0]],
        [0, 0, 1, 0, -Rv[1]],
        [0, 0, 0, 1, -Rv[0]],
    ])
    assert matrix_boundary.det() == -delta
    return {
        "A": A, "B": B, "C": C, "E": E,
        "P": P, "R": Rv,
        "delta": ZZ(delta),
        "matrix": matrix_boundary,
    }


def determinantal_divisor(M, size):
    answer = ZZ(0)
    for rows in combinations(range(M.nrows()), size):
        for columns in combinations(range(M.ncols()), size):
            answer = gcd(
                answer,
                abs(M.matrix_from_rows_and_columns(rows, columns).det()),
            )
    return ZZ(answer)


def smith_certificate(boundary):
    M = boundary["matrix"]
    delta = boundary["delta"]

    # First row moved to the bottom: four swaps, determinant sign unchanged.
    order = [1, 2, 3, 4, 0]
    permutation = matrix(ZZ, 5, 5, 0)
    for row, old_row in enumerate(order):
        permutation[row, old_row] = 1
    T = permutation * M

    v = vector(ZZ, list(T[4, 0:4]))
    w = vector(ZZ, [-T[row, 4] for row in range(4)])

    U = identity_matrix(ZZ, 5)
    for column in range(4):
        U[4, column] = -v[column]

    V = identity_matrix(ZZ, 5)
    for row in range(4):
        V[row, 4] = w[row]

    diagonal = U * T * V
    assert diagonal == diagonal_matrix(ZZ, [1, 1, 1, 1, -delta])
    assert abs(permutation.det()) == abs(U.det()) == abs(V.det()) == 1

    divisors = [determinantal_divisor(M, size) for size in range(1, 6)]
    assert divisors == [1, 1, 1, 1, abs(delta)]
    return {
        "permutation": permutation,
        "U": U,
        "V": V,
        "diagonal": diagonal,
        "determinantal_divisors": divisors,
    }
