from math import comb, gcd, isqrt, prod
import sympy as sp


def is_prime(p: int) -> bool:
    if p < 2:
        return False
    if p % 2 == 0:
        return p == 2
    d = 3
    while d * d <= p:
        if p % d == 0:
            return False
        d += 2
    return True


def vp(x: int, p: int) -> int:
    assert x != 0
    x = abs(x)
    e = 0
    while x % p == 0:
        x //= p
        e += 1
    return e


def term(n: int, k: int) -> int:
    return comb(n, k) ** 2 * comb(n + k, k) ** 2


def apery(n: int) -> int:
    return sum(term(n, k) for k in range(n + 1))


def reduced_ratio(n: int, k: int) -> tuple[int, int]:
    X = (n - k) * (n + k + 1)
    e = gcd(X, (k + 1) ** 2)
    a = (X // e) ** 2
    b = ((k + 1) ** 2 // e) ** 2
    assert gcd(a, b) == 1
    return a, b


def build(n: int):
    t = [term(n, k) for k in range(n + 1)]
    S = []
    z = 0
    for x in t:
        z += x
        S.append(z)

    a, b = zip(*(reduced_ratio(n, k) for k in range(n)))
    a, b = list(a), list(b)

    C = []
    for k in range(n):
        prev = 0 if k == 0 else S[k - 1]
        C.append(b[k] * S[k] - a[k] * prev)
        assert C[k] == b[k] * S[k + 1] - a[k] * S[k]

    Delta = [None] * n
    for k in range(1, n):
        Delta[k] = a[k - 1] * b[k] - a[k] * b[k - 1]
        assert Delta[k] > 0
        assert Delta[k] * S[k] == a[k - 1] * C[k] - a[k] * C[k - 1]
        assert Delta[k] * S[k - 1] == b[k - 1] * C[k] - b[k] * C[k - 1]

    for k in range(1, n - 1):
        lhs = (
            b[k] * Delta[k] * C[k + 1]
            - (b[k + 1] * Delta[k] + a[k - 1] * Delta[k + 1]) * C[k]
            + a[k] * Delta[k + 1] * C[k - 1]
        )
        assert lhs == 0

    # Euclidean propagation, i=1.
    H = n // 3
    for K in range(1, H + 1):
        left = prod(a[1:K + 1]) * S[0]
        right = prod(b[1:K + 1]) * S[K]
        for j in range(1, K + 1):
            right -= prod(b[1:j]) * prod(a[j + 1:K + 1]) * C[j]
        assert left == right

    return t, S, a, b, C, Delta


# Symbolic factorization of rho_(k-1)-rho_k.
N, x = sp.symbols("N x")
rho = lambda y: (N - y * (y + 1)) ** 2 / (y + 1) ** 4
F = (2 * x + 1) * N + x * (x + 1)
G = (2 * x * x + 2 * x + 1) * N - x * (x + 1) * (2 * x * x - 1)
assert sp.factor(rho(x - 1) - rho(x) - F * G / (x ** 4 * (x + 1) ** 4)) == 0


def exceptional_moduli(n: int) -> set[int]:
    q, c = divmod(n, 6)
    return {
        0: {4 * q + 1, 4 * q - 1},
        1: {4 * q + 1},
        2: {4 * q + 1},
        3: {4 * q + 3, 4 * q + 1},
        4: {4 * q + 3},
        5: {4 * q + 3},
    }[c]


for n in range(60, 181):
    H, L = n // 3, n // 2
    t, S, a, b, C, Delta = build(n)

    # All odd top candidate moduli, before imposing primality.
    rs = [r for r in range((n - 1) // 2 + 1) if (n - r) % 2 == 1]
    max_m = max(min(r, n - 2 * r - 1) for r in rs)
    expected_max = H if n % 6 in (1, 4) else H - 1
    assert max_m == expected_max

    exc = set()
    for r in rs:
        p = n - r
        s = p - 1 - r
        if max(r, s) <= H + 1:
            exc.add(p)
    assert exc == exceptional_moduli(n)

    delta = [a[k] - b[k] for k in range(n)]
    gamma = [C[k] + delta[k] * S[H] for k in range(n)]
    rows = list(range(H + 1, L))

    for p in range(n // 2 + 1, n + 1):
        if not is_prime(p):
            continue
        r = n - p
        s = p - 1 - r
        m, M = min(r, s), max(r, s)
        assert m <= H

        for k in range(p):
            assert vp(t[k], p) == 2 * (k > r) + 2 * (k > s)

        for k in range(m + 1):
            assert (t[k] - term(r, k)) % p == 0
            assert (term(r, k) - term(s, k)) % p == 0
        for k in range(m + 1, p):
            assert t[k] % p == 0
        for j in range(m + 1):
            assert (t[p + j] - 4 * t[j]) % p == 0

        central = (r == s == H)
        assert vp(t[H + 1], p) == (4 if central else 2)

        # Corrected coefficient-determinant content.
        dets = []
        for ii, k in enumerate(rows):
            for ell in rows[ii + 1:]:
                D = delta[ell] * gamma[k] - delta[k] * gamma[ell]
                dets.append(D)
        mu = min(vp(D, p) for D in dets)
        assert mu == (4 if M <= H + 1 else 2)

        if apery(r) % p == 0:
            assert all(S[k] % p == 0 for k in range(m, p))
            assert all(C[k] % p == 0 for k in range(m, p - 1))
            if m > 0:
                assert S[m - 1] % p != 0
                assert C[m - 1] % p != 0
                assert Delta[m] % p != 0
            assert Delta[m + 1] % p != 0
            assert vp(a[m], p) == (4 if r == s else 2)
