## 10. Dependency-free exact audit code

The following uses only the Python standard library. It computes shell values, the candidate ideal, nuisance factors, consecutive Newton carriers, and searches for failures of D_Q=R_Q while verifying the universal identity \gcd(D_Q,P)=R_Q.

```python
from __future__ import annotations

from dataclasses import dataclass
from math import comb, gcd, prod
from typing import Iterable


def C(n: int, k: int) -> int:
    """Binomial coefficient with the zero-outside-support convention."""
    if k < 0 or k > n:
        return 0
    return comb(n, k)


def is_prime(n: int) -> bool:
    if n < 2:
        return False
    if n % 2 == 0:
        return n == 2
    p = 3
    while p * p <= n:
        if n % p == 0:
            return False
        p += 2
    return True


def prime_list(lo: int, hi: int) -> list[int]:
    return [q for q in range(lo, hi + 1) if is_prime(q)]


def shell(M: int, d: int) -> int:
    """Exact C_M(d) for the Q5694 Laurent polynomial."""
    if M < 0 or d <= 0:
        raise ValueError("require M >= 0 and d >= 1")
    a = M // d
    total = 0
    for t in range(M + 1):
        A = sum(C(M, M - t + d * alpha)
                for alpha in range(-a, a + 1))
        B = sum(C(2 * M - t, M - t + d * beta)
                for beta in range(-a, a + 1))
        total += C(M, t) * A * B * B
    return total


def gcd_many(values: Iterable[int]) -> int:
    g = 0
    for value in values:
        g = gcd(g, abs(value))
    return g


def factor_trial(n: int) -> dict[int, int]:
    n = abs(n)
    out: dict[int, int] = {}
    p = 2
    while p * p <= n:
        while n % p == 0:
            out[p] = out.get(p, 0) + 1
            n //= p
        p = 3 if p == 2 else p + 2
    if n > 1:
        out[n] = out.get(n, 0) + 1
    return out


@dataclass(frozen=True)
class BlockAudit:
    n: int
    a: int
    M: int
    Q: tuple[int, ...]
    values: dict[int, int]
    P: int
    D: int
    target_product: int

    @property
    def nuisance(self) -> int:
        return self.D // self.target_product


def audit_block(
    n: int,
    a: int,
    q_lo: int,
    q_hi: int,
    *,
    interior_only: bool = True,
) -> BlockAudit:
    M = n - a
    Q: list[int] = []
    for q in prime_list(q_lo, q_hi):
        r = n - a * q
        if interior_only and not (1 <= r <= q - 2):
            continue
        if M // (q - 1) != a:
            continue
        Q.append(q)

    if not Q:
        raise ValueError("empty candidate block")

    values = {q: shell(M, q - 1) for q in Q}
    P = prod(Q)
    generators = [(P // q) * values[q] for q in Q]
    D = gcd_many(generators)
    target_product = prod(q for q in Q if values[q] % q == 0)

    assert gcd(D, P) == target_product

    return BlockAudit(
        n=n,
        a=a,
        M=M,
        Q=tuple(Q),
        values=values,
        P=P,
        D=D,
        target_product=target_product,
    )


def newton_weights(d0: int, d1: int) -> list[int]:
    """Weights for interpolation on d0,...,d1 evaluated at -1."""
    if d1 < d0:
        raise ValueError("require d1 >= d0")
    L = d1 - d0
    return [
        (-1) ** i
        * C(d0 + i, i)
        * C(d0 + L + 1, L - i)
        for i in range(L + 1)
    ]


def primitive_newton_carrier(M: int, d0: int, d1: int) -> int:
    weights = newton_weights(d0, d1)
    return sum(
        weights[i] * shell(M, d0 + i)
        for i in range(d1 - d0 + 1)
    )


def covering_newton_carrier(
    M: int,
    d0: int,
    d1: int,
    Q: Iterable[int],
) -> int:
    """
    Multiply a Newton carrier by every candidate prime whose node q-1
    is omitted from the stencil.
    """
    Q = tuple(Q)
    L = d1 - d0
    for q in Q:
        if d0 <= q - 1 <= d1 and q <= L:
            raise ValueError(
                f"CRT weight lemma requires q > stencil length; q={q}, L={L}"
            )
    omitted = prod(q for q in Q if not (d0 <= q - 1 <= d1))
    return omitted * primitive_newton_carrier(M, d0, d1)


def audit_long_family(
    M: int,
    Q: Iterable[int],
    stencils: Iterable[tuple[int, int]],
) -> tuple[list[int], int]:
    Q = tuple(Q)
    carriers = [
        covering_newton_carrier(M, d0, d1, Q)
        for d0, d1 in stencils
    ]
    return carriers, gcd_many(carriers)


def search_D_not_target(
    n_max: int,
    *,
    min_candidates: int = 2,
) -> list[BlockAudit]:
    failures: list[BlockAudit] = []
    for n in range(3, n_max + 1):
        a = 1
        lo = n // 2 + 1
        hi = n - 1
        Q = [
            q for q in prime_list(lo, hi)
            if 1 <= n - q <= q - 2
            and (n - 1) // (q - 1) == 1
        ]
        if len(Q) < min_candidates:
            continue
        audit = audit_block(n, 1, min(Q), max(Q))
        if audit.D != audit.target_product:
            failures.append(audit)
    return failures


def regression() -> None:
    # Exact interior two-node counterexample from Section 7.1.
    a8 = audit_block(8, 1, 5, 7)
    assert a8.Q == (5, 7)
    assert a8.values[5] == 2_479_411_520
    assert a8.values[7] == 712_499_254
    assert a8.P == 35
    assert a8.D == 10
    assert a8.target_product == 5
    assert gcd(a8.D, a8.P) == 5

    # Reported Q5700 block.
    a321 = audit_block(321, 1, 162, 211)
    assert a321.target_product == 179 * 193 * 211
    assert gcd(a321.D, a321.P) == a321.target_product

    # The user's computation says the nuisance is one in this block.
    assert a321.D == 7_289_417

    print("n=8:", a8)
    print("n=321 targets:", factor_trial(a321.target_product))
    print("n=321 D:", a321.D)


if __name__ == "__main__":
    regression()

    # Example family of long stencils containing every n=321 candidate node.
    # Adjust this list to reproduce the precise overlapping stencils used in
    # the original computation.
    Q321 = prime_list(162, 211)
    stencils = [(161, 210 + j) for j in range(8)]
    carriers, carrier_gcd = audit_long_family(320, Q321, stencils)
    print("digit lengths:", [len(str(abs(x))) for x in carriers])
    print("long-family gcd:", carrier_gcd)

    # Find further exact failures of D = target product.
    failures = search_D_not_target(100, min_candidates=2)
    for item in failures[:20]:
        print(
            "counterexample",
            item.n,
            item.Q,
            "D=", item.D,
            "R=", item.target_product,
            "nuisance=", item.nuisance,
        )
```
