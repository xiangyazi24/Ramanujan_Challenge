def finite_difference(values, start, order):
    return ZZ(sum(
        (-1) ** (order - i) * binomial(order, i) * values[start + i]
        for i in range(order + 1)
    ))


def boundary(moment_start, length, surplus):
    """F_s(D,L) = Delta^L C_(D+L+s)(D)."""
    D, L, s = ZZ(moment_start), ZZ(length), ZZ(surplus)
    assert L >= 1
    assert D > L + s                    # strict first cell
    M = D + L + s
    values = shell_batch(M, range(D, D + L + 1))
    return finite_difference(values, D, L)


def boundary_coefficients(D, L, s):
    """Independent coefficient-by-coefficient expansion of (2.1)."""
    D, L, s = ZZ(D), ZZ(L), ZZ(s)
    assert L >= 1 and D > L + s
    M = D + L + s
    total = ZZ(0)
    for kappa in polytope_points(1):
        for i in range(L + 1):
            d = D + i
            total += (
                (-1) ** (L - i)
                * binomial(L, i)
                * coefficient(
                    M,
                    d * kappa[0],
                    d * kappa[1],
                    d * kappa[2],
                )
            )
    return total


def from_zero_surplus(D, L, s):
    return ZZ(sum(
        (-1) ** j * binomial(s, j)
        * boundary(D + s - j, L + j, 0)
        for j in range(s + 1)

    coefficients = [K(operator[i]) for i in range(operator.order() + 1)]
    poly_den = lcm(value.denominator() for value in coefficients)
    polynomials = [R(poly_den * value) for value in coefficients]
