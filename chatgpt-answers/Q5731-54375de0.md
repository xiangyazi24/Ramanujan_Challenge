def first_cell_shell_mod_p2(M: int, p: int) -> int:
    d = p - 1
    assert p <= M <= 2 * p - 3
    s = M - p
    modulus = p * p
    row = binomial_row_mod(M, modulus)

    z0 = comb(2 * M, M)
    zp = comb(2 * M, M + d)
    zm = comb(2 * M, s + 1)
    out = 0

    for t in range(M + 1):
        X = row[t]
        if t >= d:
            X += row[t - d]
        if t <= s + 1:
            X += row[s + 1 - t]
        X %= modulus

        Z = (z0 + zp + zm) % modulus
        out = (out + row[t] * X * Z * Z) % modulus

        if t == M:
            break
        N = 2 * M - t
        k0 = M - t
        kp = M - t + d
        km = s + 1 - t
        z0 = z0 * k0 // N
        zp = zp * kp // N
        zm = 0 if km <= 0 else zm * km // N
    return out
