def mul(P,Q):

def lambda_power(M):

def shell(P,d):
    return sum(v for (a,b,c),v in P.items()
               if a%d==0 and b%d==0 and c%d==0)


def carrier(values,D,N):
    return sum((-1)**i*comb(D+i,i)*comb(D+N+1,N-i)*values[i]
               for i in range(N+1))


def packet(P,K,d):
    return sum(v for e,v in P.items()
               if all(abs(x)<=K*d for x in e))


def interval_radical(targets):

    return r
