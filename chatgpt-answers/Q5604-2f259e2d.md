Thus the certificate is nondegenerate at `p=13` and recovers `xi=4`.

# 12. Reproducible exact-arithmetic audit

The following script performs no fitting.  It:

1. verifies the stuffle and reversal reductions at exact prime-power
   precision;
2. verifies the direct/reflected `C_4` and `C_5` block formulas for `m=1,2`;
3. reconstructs the four endpoint vectors from the universal polynomials;
4. checks the constant Bézout identity prime by prime.

```python
#!/usr/bin/env python3
from __future__ import annotations

from fractions import Fraction
from math import comb, isqrt


def primes_at_most(limit: int) -> list[int]:
    sieve = bytearray(b"\x01") * (limit + 1)
    sieve[:2] = b"\x00\x00"
    for q in range(2, isqrt(limit) + 1):
        if sieve[q]:
            sieve[q * q : limit + 1 : q] = b"\x00" * (
                (limit - q * q) // q + 1
            )
    return [q for q in range(2, limit + 1) if sieve[q]]


def mh(comp: tuple[int, ...], p: int, modulus: int) -> int:
    """Strict H(comp) over 1<=k_1<...<k_d<=p-1."""
    dp = [0] * (len(comp) + 1)
    dp[0] = 1
    for k in range(1, p):
        for d in range(len(comp), 0, -1):
            dp[d] = (
                dp[d]
                + dp[d - 1] * pow(k, -comp[d - 1], modulus)
            ) % modulus
    return dp[-1]


def fmod(value: Fraction, modulus: int) -> int:
    return (
        value.numerator
        * pow(value.denominator, -1, modulus)
        % modulus
    )


def t(m: int, j: int) -> int:
    return comb(m, j) ** 2 * comb(m + j, j) ** 2


def q7(c: int) -> Fraction:
    return Fraction(-c**5 * (c + 3), 6)


def phi(c: int) -> Fraction:
    return sum((q7(u) for u in range(1, c)), Fraction(0))


def anchor(m: int, j: int) -> Fraction:
    return 2 * (phi(m + j) - 2 * phi(j) - phi(m - j))


def block_vector(
    m: int, j: int, reflected: bool
) -> tuple[Fraction, Fraction, Fraction, Fraction]:
    if not reflected:
        a = m - j
        return (
            Fraction(5 * a**4 - 6 * a**2 * m**2 + 4 * a * m**2 * j)
            + Fraction(4 * m**4, 3),
            Fraction(3 * m**3 * (4 * j - m)),
            Fraction(4 * m**3 * (m**2 + 3 * m * j - 6 * j**2)),
            Fraction(-2 * m**3 * (m - 3 * j) * (m - 6 * j)),
        )
    c = m + j
    return (
        Fraction(5 * c**4 - 6 * c**2 * m**2 - 4 * c * m**2 * j)
        + Fraction(4 * m**4, 3),
        Fraction(-3 * m**3 * (m + 4 * j)),
        Fraction(4 * m**3 * (-m**2 + 3 * m * j + 6 * j**2)),
        Fraction(2 * m**3 * (m + 3 * j) * (m + 6 * j)),
    )


def endpoint_vector(m: int, reflected: bool) -> tuple[Fraction, ...]:
    out = [Fraction(0) for _ in range(4)]
    anchor_limit = m if not reflected else m - 1
    for j in range(anchor_limit + 1):
        base = t(m, j) if not reflected else t(m - 1, j)
        out[0] += base * anchor(m, j)
    for j in range(m):
        base = (
            (m - j) ** 2 * t(m, j)
            if not reflected
            else (m + j) ** 2 * t(m - 1, j)
        )
        vec = block_vector(m, j, reflected)
        for i in range(4):
            out[i] += base * vec[i]
    return tuple(out)


def raw_block_sums(
    p: int, m: int, j: int, reflected: bool
) -> tuple[int, int]:
    """Return the C4 sum mod p^2 and C5 sum mod p."""
    mod2 = p * p
    h2 = h3 = h4 = h5 = 0
    total4 = 0
    total5 = 0
    a = m - j
    c = m + j

    for s in range(1, p):
        h2 = (h2 + pow(s, -2, mod2)) % mod2
        h3 = (h3 + pow(s, -3, mod2)) % mod2
        h4 = (h4 + pow(s, -4, mod2)) % mod2
        h5 = (h5 + pow(s, -5, mod2)) % mod2

        i1 = pow(s, -1, mod2)
        i2 = i1 * i1 % mod2
        i3 = i2 * i1 % mod2
        i4 = i2 * i2 % mod2
        i5 = i4 * i1 % mod2
        i6 = i3 * i3 % mod2

        if not reflected:
            c4 = (
                5 * a**4 * i6
                - 6 * a**2 * m**2 * h2 * i4
                + 8 * a * m**2 * j * h3 * i3
                + 2 * m**4 * h2 * h2 * i2
                - m**2 * (m**2 + 6 * j**2) * h4 * i2
            )
            c5 = (
                6 * a**5 * pow(s, -7, p)
                - 8 * a**3 * m**2 * (h2 % p) * pow(s, -5, p)
                + 12 * a**2 * m**2 * j * (h3 % p) * pow(s, -4, p)
                - 2 * a * m**2 * (m**2 + 6 * j**2)
                  * (h4 % p) * pow(s, -3, p)
                + 4 * m**2 * j * (m**2 + 2 * j**2)
                  * (h5 % p) * pow(s, -2, p)
                + 4 * a * m**4 * (h2 % p) ** 2 * pow(s, -3, p)
                - 8 * m**4 * j * (h2 % p) * (h3 % p)
                  * pow(s, -2, p)
            )
        else:
            c4 = (
                5 * c**4 * i6
                - 6 * c**2 * m**2 * h2 * i4
                - 8 * c * m**2 * j * h3 * i3
                + 2 * m**4 * h2 * h2 * i2
                - m**2 * (m**2 + 6 * j**2) * h4 * i2
            )
            c5 = (
                -6 * c**5 * pow(s, -7, p)
                + 8 * c**3 * m**2 * (h2 % p) * pow(s, -5, p)
                + 12 * c**2 * m**2 * j * (h3 % p) * pow(s, -4, p)
                + 2 * c * m**2 * (m**2 + 6 * j**2)
                  * (h4 % p) * pow(s, -3, p)
                + 4 * m**2 * j * (m**2 + 2 * j**2)
                  * (h5 % p) * pow(s, -2, p)
                - 4 * c * m**4 * (h2 % p) ** 2 * pow(s, -3, p)
                - 8 * m**4 * j * (h2 % p) * (h3 % p)
                  * pow(s, -2, p)
            )

        total4 = (total4 + c4) % mod2
        total5 = (total5 + c5) % p

    return total4, total5


def check_prime(p: int) -> tuple[int, int, int, int, tuple[int, ...]]:
    mod2 = p * p
    s2 = mh((2,), p, mod2)
    s3 = mh((3,), p, mod2)
    s4 = mh((4,), p, mod2)
    s6 = mh((6,), p, mod2)
    h22 = mh((2, 2), p, mod2)
    h24 = mh((2, 4), p, mod2)
    h42 = mh((4, 2), p, mod2)
    h33 = mh((3, 3), p, mod2)
    h222 = mh((2, 2, 2), p, mod2)

    assert s2 % p == s4 % p == s6 % p == 0
    assert s3 == 0
    assert h24 % p == 0
    assert (s2 * s4 - h24 - h42 - s6) % mod2 == 0
    assert (s3 * s3 - 2 * h33 - s6) % mod2 == 0
    assert (s2 * h22 - 3 * h222 - h42 - h24) % mod2 == 0

    xi = (s6 // p) % p
    eta = (h24 // p) % p
    aa = mh((2, 2, 3), p, p)
    bb = mh((2, 5), p, p)

    assert mh((3, 2, 2), p, p) == (-aa) % p
    assert mh((2, 3, 2), p, p) == 0
    assert mh((5, 2), p, p) == (-bb) % p
    assert mh((3, 4), p, p) == (2 * aa + bb) % p
    assert mh((4, 3), p, p) == (-2 * aa - bb) % p

    for m in (1, 2):
        for j in range(m):
            for reflected in (False, True):
                raw4, raw5 = raw_block_sums(p, m, j, reflected)
                assert raw4 % p == 0
                vec = block_vector(m, j, reflected)
                expected4 = (
                    fmod(vec[0], p) * xi
                    + fmod(vec[1], p) * eta
                ) % p
                expected5 = (
                    fmod(vec[2], p) * aa
                    + fmod(vec[3], p) * bb
                ) % p
                assert raw4 // p % p == expected4
                assert raw5 == expected5

    delta = endpoint_vector(1, True)
    pi = endpoint_vector(1, False)
    mu = endpoint_vector(2, True)
    nu = endpoint_vector(2, False)

    expected = (
        (Fraction(1, 3), Fraction(-3), Fraction(-4), Fraction(2)),
        (Fraction(-5), Fraction(-3), Fraction(4), Fraction(-2)),
        (Fraction(16940, 3), Fraction(-5376), Fraction(8704), Fraction(23296)),
        (Fraction(-61964, 3), Fraction(1536), Fraction(5120), Fraction(-2560)),
    )
    assert (delta, pi, mu, nu) == expected

    coords = (xi, eta, aa, bb)
    rows = tuple(
        sum(fmod(row[i], p) * coords[i] for i in range(4)) % p
        for row in expected
    )
    d, pp, mm, nn = rows
    assert (nn + 896 * d - 384 * pp + 18436 * xi) % p == 0
    return xi, eta, aa, bb, rows


def main() -> None:
    checked = 0
    for p in primes_at_most(43):
        if p < 11:
            continue
        result = check_prime(p)
        checked += 1
        if p in (11, 13):
            xi, eta, aa, bb, rows = result
            print(
                f"p={p} xi={xi} eta={eta} A={aa} B={bb} "
                f"delta,pi,mu,nu={rows}"
            )
    print(f"checked_primes={checked}")
    print("endpoint_vectors=4/4")
    print("bezout_checks=all")
    print("failures=0")


if __name__ == "__main__":
    main()
```

The prime loop is only an audit.  The proof is the carry decomposition,
logarithmic expansion, stuffle/reversal reduction, and exact coefficient
algebra in Sections 2–8.
