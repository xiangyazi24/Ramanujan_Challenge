We work in the K-gauge with
$P(X)=34X^3+51X^2+27X+5$, $K_0=1$, $K_1=P(X)$, and
$K_{m+1}(X)=P(X+m)K_m(X)-(X+m)^6K_{m-1}(X)$.
Let
$N_h(r)=K_{h-1}(r+1)$ and $A_h=\prod_{j=1}^h(X+j)^3$.

\begin{proposition}[Endpoint law]\label{svl:endpoint}
For every $m\geq0$,
\[
K_m(-m)=(-1)^m(m!)^3b_m.
\]
\end{proposition}

\begin{proof}
Substituting $X=-m$ in the recurrence gives exactly the cleared-denominator
Apery recurrence; induction from the initial values proves the identity.
\end{proof}

\begin{proposition}[Pole values]\label{svl:pole}
For $1\leq j\leq h$,
\[
N_h(-j)=(-1)^{j-1}((j-1)!)^3((h-j)!)^3b_{j-1}b_{h-j}.
\]
Consequently $\gcd(N_h,A_h)=1$ over $\mathbb Q$.
\end{proposition}

\begin{proof}
The tridiagonal determinant representation splits at the cut position $j$.
Applying \ref{svl:endpoint} to the two resulting blocks gives the formula.
Since all Apery numbers are positive, no factor $X+j$ divides $N_h$.
\end{proof}

\begin{proposition}[Adjacent resultant]\label{svl:adjres}
With the resultant convention
$\operatorname{Res}(f,g)=\operatorname{lc}(g)^{\deg f}\prod_{g(\beta)=0}f(\beta)$,
we have
\[
\operatorname{Res}(K_m,K_{m+1})=
(-1)^{m(m+1)/2}\prod_{j=1}^m((j!)^3b_j)^6.
\]
\end{proposition}

\begin{proof}
Reduce $K_{m+1}$ modulo $K_m$ using the recurrence. Multiplicativity of
resultants and the endpoint formula \ref{svl:endpoint} give the product.
\end{proof}

\begin{remark}
If $p\mid b_j$, then $p$ divides
$\operatorname{Res}(K_m,K_k)$ for every $j\leq m<k$.
This is cut-edge pollution: raw resultant counts are non-diagnostic; see the
axis-strip and balanced-core reduction.
\end{remark}

\begin{proposition}[Renewal identity]\label{svl:renewal}
For $g\geq1$,
\[
K_{m+g+1}(X)=K_{m+1}(X)K_g(X+m+1)
-(X+m+1)^6K_m(X)K_{g-1}(X+m+2).
\]
\end{proposition}

The delta-gauge law
\[

\]
is the same identity after clearing $A$-denominators. The dictionary is
$A_h(x)\delta_h(x)=N_h(x)=K_{h-1}(x+1)$: one identity, two gauges.
\end{remark}

\begin{proposition}[Renewal resultant factorization]\label{svl:renres}
\[
\operatorname{Res}(K_m,K_{m+g+1})
=(-1)^{mg}\operatorname{Res}(K_m,K_{m+1})
\operatorname{Res}(K_m(X),K_g(X+m+1)).
\]
\end{proposition}

\begin{proof}
Apply the renewal identity and resultant multiplicativity. The sign is the
Sylvester parity sign $9mg\equiv mg\pmod2$.
The displayed sign follows from the fixed resultant ordering convention.
\end{proof}

\begin{proposition}[Linear factors]\label{svl:linfactor}
For $1\leq j\leq m$,
\[
\operatorname{Res}(K_m,X+j)=(-1)^mK_m(-j)
=(-1)^{m+j-1}((j-1)!)^3((m+1-j)!)^3b_{j-1}b_{m+1-j}.

\end{proposition}
