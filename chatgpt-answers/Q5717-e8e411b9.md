#!/usr/bin/env python3
from __future__ import annotations

import argparse
from functools import lru_cache
from math import comb, gcd, isqrt, prod

from sympy import factorint, isprime


def C(n: int, k: int) -> int:
    return comb(n, k) if 0 <= k <= n else 0


def primes_upto(n: int) -> list[int]:
    if n < 2:
        return []
    sieve = bytearray(b"\x01") * (n + 1)
    sieve[:2] = b"\x00\x00"
    for p in range(2, isqrt(n) + 1):
        if sieve[p]:
            sieve[p * p:n + 1:p] = b"\x00" * (((n - p * p) // p) + 1)
    return [p for p in range(2, n + 1) if sieve[p]]


def apery_mod(r: int, p: int) -> int:
    return sum((C(r, k) * C(r + k, k)) ** 2 for k in range(r + 1)) % p


@lru_cache(maxsize=None)
def shell(M: int, d: int) -> int:
    """Exact C_M(d), using the live one-sum Laurent-shell formula."""
    if d <= 0:
        raise ValueError((M, d))
    a = M // d
    total = 0
    for t in range(M + 1):
        X = sum(C(M, M - t + d * u) for u in range(-a, a + 1))
        Z = sum(C(2 * M - t, M - t + d * v) for v in range(-a, a + 1))
        total += C(M, t) * X * Z * Z
    return total


def weight(d: int, L: int, i: int) -> int:
    return (-1) ** i * C(d + i, i) * C(d + L + 1, L - i)


def carrier(M: int, d: int, L: int) -> int:
    return sum(weight(d, L, i) * shell(M, d + i) for i in range(L + 1))


def delta(M: int, d: int, k: int) -> int:
    return sum((-1) ** (k - i) * C(k, i) * shell(M, d + i)
               for i in range(k + 1))


def gcd_many(values) -> int:
    g = 0
    for x in values:
        g = gcd(g, abs(x))
    return g


def valuation(x: int, p: int) -> int:
    x = abs(x)
    e = 0
    while x and x % p == 0:
        x //= p
        e += 1
    return e


def factor_dict(x: int) -> dict[int, int]:
    x = abs(x)
    if x <= 1:
        return {}
    raw = factorint(x, limit=10**7, use_ecm=True)
    out = {int(p): int(e) for p, e in raw.items()}
    for p in out:
        if not isprime(p):
            raise RuntimeError(f"unfactored composite cofactor {p} in {x}")
    return dict(sorted(out.items()))


def fmt_factor(x: int) -> str:
    if x == 0:
        return "0"
    if abs(x) == 1:
        return str(abs(x))
    return " * ".join(
        str(p) if e == 1 else f"{p}^{e}"
        for p, e in factor_dict(x).items()
    )


def a1_targets(n: int) -> list[int]:
    """Targets q in the first quotient cell n/2 < q < n."""
    ans = []
    for q in primes_upto(n):
        if 2 * q <= n:
            continue
        r = n - q
        if 1 <= r <= q - 2 and apery_mod(r, q) == 0:
            ans.append(q)
    return ans


def canonical_core(n: int) -> tuple[int, int, list[int]] | None:
    """Core used by the canonical first-cell covering placement."""
    M = n - 1
    targets = a1_targets(n)
    if len(targets) < 2:
        return None
    D = M // 2 + 1
    N = max(targets) - D
    assert N >= 1
    assert all(D <= q - 1 <= D + N - 1 for q in targets)
    return D, N, targets


def remove_target_powers(K: int, targets: list[int]) -> int:
    x = abs(K)
    for q in targets:
        while x % q == 0:
            x //= q
    return x


def analyze_core(n: int, D: int, N: int, targets: list[int], m: int) -> dict:
    M = n - 1
    L = N + m - 2
    d0 = D - m + 1
    if d0 <= 0:
        raise ValueError((n, D, N, m))

    gs = [carrier(M, d0 + t, L) for t in range(m)]
    K = gcd_many(gs)
    R = prod(targets)
    assert K % R == 0, (n, D, N, m, targets)

    Bs = [C(D + N + t, L) for t in range(m - 1)]
    Hs = [(-1) ** (L + 1) * delta(M, d0 + t, L + 1)
          for t in range(m - 1)]
    for t in range(m - 1):
        assert gs[t] - gs[t + 1] == Bs[t] * Hs[t]

    euclid = gcd_many([gs[0]] + [B * H for B, H in zip(Bs, Hs)])
    assert euclid == K

    F = gcd_many(Bs)
    nuisance = K // R
    outside = remove_target_powers(K, targets)
    target_excess = nuisance // outside

    nf = factor_dict(nuisance)
    classes = {}
    for p, e in nf.items():
        if p in targets:
            cls = "target-excess"
        elif F % p == 0:
            cls = "common-prefactor"
        else:
            forced = [t for t, B in enumerate(Bs) if B % p != 0]
            assert all(Hs[t] % p == 0 for t in forced)
            cls = f"high-difference t={forced}"
        classes[p] = (e, cls, valuation(F, p))

    return {
        "n": n, "M": M, "D": D, "N": N, "m": m,
        "L": L, "d0": d0, "targets": targets,
        "K": K, "R": R, "nuisance": nuisance,
        "outside": outside, "target_excess": target_excess,
        "F": F, "classes": classes,
        "carrier_digits": [len(str(abs(g))) for g in gs],
    }


def print_record(r: dict) -> None:
    print(
        f"n={r['n']} D={r['D']} N={r['N']} m={r['m']} "
        f"L={r['L']} d0={r['d0']} targets={r['targets']}"
    )
    print("  nuisance      =", fmt_factor(r["nuisance"]))
    print("  outside       =", fmt_factor(r["outside"]))
    print("  target excess =", fmt_factor(r["target_excess"]))
    print("  prefactor gcd =", fmt_factor(r["F"]))
    print("  classes       =", r["classes"])
    print("  carrier digits=", r["carrier_digits"])


EXPECTED_M2 = {
    200: 47,
    272: 1,
    300: 1,
    321: 43,
    755: 275,
}


def scan_named(max_m: int) -> None:
    for n, expected in EXPECTED_M2.items():
        spec = canonical_core(n)
        assert spec is not None, n
        D, N, targets = spec
        print("\nNAMED", n, "canonical", (D, N), targets)
        first_clean = None
        for m in range(2, max_m + 1):
            rec = analyze_core(n, D, N, targets, m)
            print_record(rec)
            if m == 2:
                assert rec["nuisance"] == expected, (n, rec["nuisance"], expected)
            if rec["nuisance"] == 1 and first_clean is None:
                first_clean = m
        print("  least tested clean m =", first_clean)

    # Shifted n=321 placement recorded in Sections 55-56.
    n = 321
    D, N, targets = canonical_core(n)
    shifted = analyze_core(n, D + 1, N - 1, targets, 2)
    print("\nSHIFTED n=321")
    print_record(shifted)
    assert shifted["nuisance"] == 1


def estimated_work(n: int, D: int, N: int, max_m: int) -> int:
    # Number of t-summands times number of distinct shell nodes, a stable
    # target-blind proxy for exact big-integer work.
    M = n - 1
    nodes = N + 2 * max_m - 3
    return (M + 1) * nodes


def scan_all(limit: int, max_m: int, work_budget: int) -> None:
    records = []
    skipped = []
    used = 0
    for n in range(3, limit + 1):
        spec = canonical_core(n)
        if spec is None:
            continue
        D, N, targets = spec
        cost = estimated_work(n, D, N, max_m)
        if used + cost > work_budget:
            skipped.append((n, D, N, targets, cost))
            continue
        used += cost
        first_clean = None
        rows = []
        for m in range(2, max_m + 1):
            rec = analyze_core(n, D, N, targets, m)
            rows.append((m, factor_dict(rec["nuisance"]), rec["classes"]))
            if rec["nuisance"] == 1 and first_clean is None:
                first_clean = m
        records.append((n, D, N, targets, first_clean, rows))

    print("\nALL EXACT RECORDS", len(records), "work", used)
    for row in records:
        print(row)
    print("\nSKIPPED", len(skipped))
    for row in skipped:
        print(row)


def main() -> None:
    ap = argparse.ArgumentParser()
    ap.add_argument("--max-m", type=int, default=8)
    ap.add_argument("--limit", type=int, default=2000)
    ap.add_argument("--work-budget", type=int, default=20_000_000)
    ap.add_argument("--named-only", action="store_true")
    args = ap.parse_args()

    scan_named(args.max_m)
    if not args.named_only:
        scan_all(args.limit, args.max_m, args.work_budget)


if __name__ == "__main__":
    main()
