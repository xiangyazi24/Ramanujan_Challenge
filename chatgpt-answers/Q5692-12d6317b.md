from math import comb


def C(n, k):
    return comb(n, k) if 0 <= k <= n else 0


def coeff(M, A, B, D, modulus=None):
    out = 0
    for k in range(M + 1):
        term = (
            C(M, k)
            * C(M, A + M - k)
            * C(2 * M - k, B + M - k)
            * C(2 * M - k, D + M - k)
        )
        out += term
        if modulus:
            out %= modulus
    return out if modulus is None else out % modulus


def sigma(kappa):
    u, v, w = kappa
    return max(0, u) - min(0, u, v, w)


def alias_set(a):
    return [
        (u, v, w)
        for u in range(-a, a + 1)
        for v in range(-a, a + 1)
        for w in range(-a, a + 1)
        if v <= a and w <= a and sigma((u, v, w)) <= a
    ]


def shell(M, d, a, modulus=None):
    total = 0
    for u, v, w in alias_set(a):
        total += coeff(M, u * d, v * d, w * d, modulus)
        if modulus:
            total %= modulus
    return total if modulus is None else total % modulus

for i, p in enumerate(hits):

        mod = (p * q) ** 2
        X = shell(M, p - 1, 1, mod)
        Y = shell(M, q - 1, 1, mod)

        assert cross % (p * q) == 0
