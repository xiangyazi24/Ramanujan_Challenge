The following standard-library script verifies the exact support identity, the divisor q_(n,d) | D_n, the prime-index lower bound behind Theorem 4.1, the parity-section bounds, and the 2^n,6^n warning. It does not assume the desired pointwise theorem.

from math import comb, gcd, lcm, log


def primes_upto(n: int) -> list[int]:
    if n < 2:
        return []
    sieve = bytearray(b"\x01") * (n + 1)
    sieve[0:2] = b"\x00\x00"
    for p in range(2, int(n**0.5) + 1):
        if sieve[p]:
            sieve[p*p:n+1:p] = b"\x00" * (((n - p*p) // p) + 1)
    return [p for p in range(2, n + 1) if sieve[p]]


def apery_table(N: int) -> list[int]:
    b = [0] * (N + 1)
    b[0] = 1
    if N >= 1:
        b[1] = 5
    for n in range(1, N):
        P = 34*n**3 + 51*n**2 + 27*n + 5
        num = P*b[n] - n**3*b[n-1]
        den = (n + 1)**3
        assert num % den == 0
        b[n+1] = num // den
    return b


def D(N: int) -> int:
    z = 1
    for k in range(1, N + 1):
        z = lcm(z, k)
    return z


def B(n: int, d: int) -> int:
    return comb(n, n // d)


def reduced_den(bn: int, carrier: int) -> int:
    return carrier // gcd(bn, carrier)


def rad_from_primes(x: int, ps: list[int]) -> int:
    z = 1
    for p in ps:
        if x % p == 0:
            z *= p
    return z


N = 1200
b = apery_table(N + 1)
ps = primes_upto(N + 1)

# Exact top-half support; n=9 is the unique midpoint mismatch.
for n in range(10, N + 1):
    Cn = comb(n, n // 2)
    interval = [p for p in ps if n/2 < p <= n]
    lhs = rad_from_primes(gcd(b[n], Cn), interval)
    rhs = rad_from_primes(b[n], interval)
    assert lhs == rhs

# Every binomial carrier and every reduced denominator divide D_n.
for d in range(2, 10):
    for n in range(0, N + 1):
        carrier = B(n, d)
        q = reduced_den(b[n], carrier)
        assert D(n) % carrier == 0
        assert D(n) % q == 0

# Prime-index lower bound for the general fixed-d theorem.
for d in range(2, 10):
    for p in ps:
        if p > max(d, 5) and p <= N:
            assert b[p] % p == 5 % p
            assert B(p, d) % p == 0
            assert reduced_den(b[p], B(p, d)) % p == 0

# Prefix LCM bounds and the exact divisibility sandwich.
for d in range(2, 8):
    Q = 1
    prime_product = 1
    for n in range(N + 1):
        Q = lcm(Q, reduced_den(b[n], B(n, d)))
    for p in ps:
        if p <= N and p > max(d, 5):
            prime_product *= p
    assert Q % prime_product == 0
    assert D(N) % Q == 0
    print("d", d, "logQ/N", log(Q) / N,
          "logD/N", log(D(N)) / N)

# Central parity sections.
M = N // 2
Q0 = Q1 = 1
P0 = P1 = 1
for m in range(M + 1):
    Q0 = lcm(Q0, reduced_den(b[2*m], comb(2*m, m)))
    Q1 = lcm(Q1, reduced_den(b[2*m+1], comb(2*m+1, m)))
for p in ps:
    if 7 <= p <= 2*M - 1:
        P0 *= p
    if 7 <= p <= 2*M + 1:
        P1 *= p
assert Q0 % P0 == 0 and D(2*M) % Q0 == 0
assert Q1 % P1 == 0 and D(2*M + 1) % Q1 == 0
print("even", log(Q0) / (2*M), "odd", log(Q1) / (2*M + 1))
