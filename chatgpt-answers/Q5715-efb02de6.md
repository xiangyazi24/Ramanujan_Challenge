```python
#!/usr/bin/env python3
from __future__ import annotations
from argparse import ArgumentParser
from functools import lru_cache
from math import comb, gcd, isqrt

try:
    from sympy import factorint, isprime
except Exception:
    factorint = None
    isprime = None


def C(n, k):
    return comb(n, k) if 0 <= k <= n else 0


def primes_upto(n):
    if n < 2:
        return []
    s = bytearray(b"\x01") * (n + 1)
    s[0:2] = b"\x00\x00"
    for p in range(2, isqrt(n) + 1):
        if s[p]:
            s[p*p:n+1:p] = b"\x00" * (((n-p*p)//p)+1)
    return [p for p in range(2, n + 1) if s[p]]


@lru_cache(None)
def apery_mod(m, p):
    return sum((C(m,k)*C(m+k,k))**2 for k in range(m+1)) % p


@lru_cache(None)
def coeff_mod(m, u, v, w, p):
    return sum(
        C(m,t)*C(m,t-u)*C(2*m-t,m-v)*C(2*m-t,m-w)
        for t in range(m+1)
    ) % p


@lru_cache(None)
def shell_exact(M, d):
    a = M // d
    out = 0
    for t in range(M + 1):
        xp = sum(C(M, M-t+d*u) for u in range(-a, a+1))
        yp = sum(C(2*M-t, M-t+d*v) for v in range(-a, a+1))
        out += C(M,t) * xp * yp * yp
    return out


@lru_cache(None)
def shell_mod(M, d, p):
    a = M // d
    out = 0
    for t in range(M + 1):
        xp = sum(C(M, M-t+d*u) for u in range(-a, a+1)) % p
        yp = sum(C(2*M-t, M-t+d*v) for v in range(-a, a+1)) % p
        out = (out + C(M,t)*xp*yp*yp) % p
    return out


def weight(d, L, j):
    return (-1)**j * C(d+j,j) * C(d+L+1,L-j)


def carrier_exact(M, d, L):
    return sum(weight(d,L,j)*shell_exact(M,d+j) for j in range(L+1))


def carrier_mod(M, d, L, p):
    return sum((weight(d,L,j) % p)*shell_mod(M,d+j,p)
               for j in range(L+1)) % p


def polytope_points(a):
    return [
        (x,y,z)
        for x in range(-a,a+1)
        for y in range(-a,a+1)
        for z in range(-a,a+1)
        if x-y <= a and x-z <= a
    ]


def ghost_pairing_mod(M, rho, k):
    alpha, s = divmod(M, rho)
    d = k*rho - 1
    b = M // d
    return sum(
        coeff_mod(alpha,k*x,k*y,k*z,rho)
        * coeff_mod(s,-x,-y,-z,rho)
        for x,y,z in polytope_points(b)
    ) % rho


def unique_multiple(p, lo, hi):
    k = (lo + p - 1) // p
    m = k*p
    return (k,m) if m <= hi else None


def target_primes(n, a, D, N):
    out = set()
    for q in primes_upto(D+N+1):
        if D+1 <= q <= D+N:
            r = n-a*q
            if 1 <= r <= q-2 and apery_mod(r,q) == 0:
                out.add(q)
    return out


def classify_prime(n, M, a, D, N, p, targets):
    if p <= N:
        return "small<=N"
    if D+1 <= p <= D+N:
        return "candidate-target" if p in targets else "candidate-NONTARGET"
    Kdiv = C(D+N,N) % p == 0
    ga = unique_multiple(p,D,D+N)
    gb = unique_multiple(p,D+1,D+N+1)
    if ga and gb and ga[1] == gb[1]:
        return f"common-ghost k={ga[0]} prefactor={Kdiv}"
    if ga or gb:
        g = ga if ga else gb
        return f"edge-ghost k={g[0]} prefactor={Kdiv}"
    return f"generic-packet prefactor={Kdiv}"


def partial_factor(x, bound=10**6):
    x = abs(x)
    fac = {}
    for p in primes_upto(min(bound,isqrt(x)+1)):
        if p*p > x:
            break
        if x % p == 0:
            e = 0
            while x % p == 0:
                x //= p; e += 1
            fac[p] = e
    if x > 1 and isprime is not None and isprime(x):
        fac[int(x)] = fac.get(int(x),0) + 1
        x = 1
    elif x > 1 and factorint is not None and x.bit_length() <= 512:
        ff = factorint(x, limit=bound, use_ecm=True)
        rem = 1
        for p,e in ff.items():
            p,e = int(p),int(e)
            if isprime(p):
                fac[p] = fac.get(p,0) + e
            else:
                rem *= p**e
        x = rem
    return dict(sorted(fac.items())), x


def audit_block(n, a, D, N, exact, rho_max):
    M = n-a
    assert all(M//d == a for d in range(D-1,D+N+1))
    targets = target_primes(n,a,D,N)
    print(f"BLOCK n={n} a={a} M={M} I=[{D},{D+N-1}] N={N}")
    print(" targets=",sorted(targets))

    known = set()
    if exact:
        A = carrier_exact(M,D-1,N)
        B = carrier_exact(M,D,N)
        DI = gcd(abs(A),abs(B))
        fac, rem = partial_factor(DI)
        print(" digits(A,B,D)=",len(str(abs(A))),len(str(abs(B))),len(str(DI)))
        print(" partial factors=",fac)
        print(" remaining cofactor digits=",0 if rem == 1 else len(str(rem)))
        known.update(fac)
        for p in fac:
            print("  ",p,classify_prime(n,M,a,D,N,p,targets))
        assert not any(
            D+1 <= p <= D+N and p not in targets for p in fac
        )

    for p in primes_upto(rho_max):
        if p in known:
            continue
        if carrier_mod(M,D-1,N,p) == 0 and carrier_mod(M,D,N,p) == 0:
            label = classify_prime(n,M,a,D,N,p,targets)
            print(" modular divisor",p,label)
            if p > N and "common-ghost" in label:
                k,m = unique_multiple(p,D+1,D+N)
                assert shell_mod(M,m-1,p) == 0
                alpha,s = divmod(M,p)
                b = M//(m-1)
                if p > s+b:
                    assert ghost_pairing_mod(M,p,k) == 0


def cell_bounds(n,a):
    M = n-a
    lo = M//(a+1)+1
    hi = M//a
    lo = max(lo,isqrt(n))
    return lo,hi


def blocks_for_cell(n,a):
    lo,hi = cell_bounds(n,a)
    length = hi-lo+1
    if length < 2:
        return []
    h = max(2,round(n**(1/3)))
    scales = sorted({2,3,4,5,8,h,min(length,2*h),length})
    out = []
    for N in scales:
        if N > length:
            continue
        step = max(1,N//2)
        starts = list(range(lo,hi-N+2,step))
        if starts[-1] != hi-N+1:
            starts.append(hi-N+1)
        out.extend((D,N) for D in starts)
    return sorted(set(out))


def main():
    ap = ArgumentParser()
    ap.add_argument("--limit",type=int,default=2000)
    ap.add_argument("--rho-max",type=int,default=10000)
    ap.add_argument("--exact-M",type=int,default=400)
    args = ap.parse_args()

    named = {200,272,300,321,755}
    for n in range(20,args.limit+1):
        for a in range(1,isqrt(n)+2):
            for D,N in blocks_for_cell(n,a):
                exact = n in named or (n-a <= args.exact_M and N <= 16)
                audit_block(n,a,D,N,exact,args.rho_max)


if __name__ == "__main__":
    main()
```
