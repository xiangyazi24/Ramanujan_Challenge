```python
from functools import lru_cache
from math import comb, gcd


def C(n, k):
    return comb(n, k) if 0 <= k <= n else 0


@lru_cache(None)
def shell(M, d):
    a = M // d
    total = 0
    for u in range(M + 1):
        X = sum(C(M, M-u+d*j) for j in range(-a, a+1))
        Z = sum(C(2*M-u, M-u+d*j) for j in range(-a, a+1))
        total += C(M, u) * X * Z * Z
    return total


def delta(M, d, r):
    return sum((-1)**(r-i) * C(r, i) * shell(M, d+i)
               for i in range(r+1))


def G(M, d, L):
    return sum((-1)**i * C(d+i, i) * C(d+L+1, L-i)
               * shell(M, d+i) for i in range(L+1))


def H(M, d, L, s, t):
    return G(M, d-s, L+s+t)


def rectangle_gcd(M, d, L, S, T):
    out = 0
    for s in range(S+1):
        for t in range(T+1):
            out = gcd(out, abs(H(M, d, L, s, t)))
    return out
```
