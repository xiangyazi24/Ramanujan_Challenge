```python
from ore_algebra import OreAlgebra

Par.<M,d,L> = PolynomialRing(QQ)
K = Par.fraction_field()
R.<t,i> = PolynomialRing(K)
Frac = R.fraction_field()
OA = OreAlgebra(R, names=("St", "Si"))
St, Si = OA.gens()


def lower_shift_ratio(N, k, step):
    if step == 1:
        return Frac((N-k)/(k+1))
    if step == -1:
        return Frac(k/(N-k+1))
    return Frac(1)


def annihilator_from_ratio(S, ratio):
    ratio = Frac(ratio)
    return ratio.denominator()*S - ratio.numerator()


def orientation_ideal(u, v, w):
    kx = M-t+(d+i)*u
    ky = M-t+(d+i)*v
    kz = M-t+(d+i)*w

    ri = Frac(
        -(d+i+1)*(L-i)/((i+1)*(d+i+2))
        * lower_shift_ratio(M, kx, u)
        * lower_shift_ratio(2*M-t, ky, v)
        * lower_shift_ratio(2*M-t, kz, w)
    )

    rt = Frac(
        (M-t)/(t+1)
        * kx/(M-kx+1)
        * ky*kz/(2*M-t)^2
    )

    ideal = OA.ideal([
        annihilator_from_ratio(Si, ri),
        annihilator_from_ratio(St, rt),
    ])
    return ideal


for u in (-1,0,1):
    for v in (-1,0,1):
        for w in (-1,0,1):
            # The five impossible first-cell orientations are skipped by
            # their support test; all remaining summands are proper
            # hypergeometric in (t,i).
            I = orientation_ideal(u,v,w)
            telescopers, certificates = I.ct(
                Si - 1,
                certificates=True,
                early_termination=True,
                iteration_limit=16,
            )
            for P, C in zip(telescopers, certificates):
                assert OA(P) - (Si - 1)*C in I
```
