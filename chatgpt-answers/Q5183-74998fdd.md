# 7. Exact Sage verification

The following code verifies the constant-term identity and the critical-value resultant.

```python
from sage.all import *

# ------------------------------------------------------------
# 1. The defining sum
# ------------------------------------------------------------
def b_sum(n):
    return sum(
        binomial(n, k)^2
        * binomial(n+k, n)
        * binomial(n+2*k, n)
        for k in range(n+1)
    )

# ------------------------------------------------------------
# 2. Laurent-period model
# ------------------------------------------------------------
L.<x,y,z> = LaurentPolynomialRing(QQ)
P = ((1+y)*(1+z)*(1+x)*(x + (1+y)*(1+z)^2))/(x*y*z)

def CT(f):
    return QQ(f.dict().get((0,0,0), 0))

for n in range(8):
    assert CT(P^n) == b_sum(n)

print([CT(P^n) for n in range(6)])
# [1, 7, 163, 5191, 191251, 7665757]

# ------------------------------------------------------------
# 3. Eliminate the positive-critical-point parameter q
# ------------------------------------------------------------
R.<q,lam> = PolynomialRing(QQ, 2)
F = q*(q+1)^2*(q-2)^2 - 4
G = lam*(q-2)^2 - q*(q+1)
res = factor(F.resultant(G, q))
print(res)

chi = 4*lam^3 - 220*lam^2 + 8*lam - 1
assert (R(res) % chi) == 0

# Numerical characteristic values
print(chi.univariate_polynomial().roots(CC, multiplicities=False))
```
