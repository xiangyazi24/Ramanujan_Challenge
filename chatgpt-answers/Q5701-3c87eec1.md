## 12. Dependency-free exact audit

The following standard-library script checks the coefficient formula, all 22 quotient coordinates, the corridor identities, the (h=2,4) hidden contents, the prime-power formula for (g_h), and the three (n=321) pairs. It does not expand (A^{321}) as a three-dimensional dictionary.

```python
#!/usr/bin/env python3
from itertools import product
from math import comb, gcd

O = (0, 0, 0)


def C(n, k):
    return comb(n, k) if 0 <= k <= n else 0


def in_P(u):
    x, y, z = u
    return (
        -1 <= x <= 1
        and -1 <= y <= 1
        and -1 <= z <= 1
        and x - y <= 1
        and x - z <= 1
    )


S = [u for u in product(range(-1, 2), repeat=3) if in_P(u)]
assert len(S) == 22


def coeff(m, u):
    """[X^u] Lambda^m, exact one-fold formula."""
    a, b, c = u
    return sum(
        C(m, k)
        * C(m, a + m - k)
        * C(2 * m - k, b + m - k)
        * C(2 * m - k, c + m - k)
        for k in range(m + 1)
    )


LAM = {u: coeff(1, u) for u in S}
assert LAM[O] == 5
assert sum(LAM.values()) == 40


def scale(q, u):
    return tuple(q * x for x in u)


def neg(u):
    return tuple(-x for x in u)


def b(m):
    return coeff(m, O)


def E(q, t, n, mu):
    num = coeff(n, scale(q, mu)) - b(t) * LAM[mu]
    assert num % q == 0
    return num // q


def corridor_pair(n, p, ell):
    h = ell - p
    r = n - p
    s = n - ell
    assert r == s + h and 0 <= r < p and 0 <= s < ell

    ep = {mu: E(p, r, n, mu) for mu in S}
    el = {mu: E(ell, s, n, mu) for mu in S}

    # Pure-cross target checks for the supplied n=321 examples.
    assert b(r) % p == 0
    assert b(s) % ell == 0

    for mu in S:
        lam = LAM[mu]
        pn = coeff(n, scale(p, mu))
        ln = coeff(n, scale(ell, mu))

        # Off-grid quotient coordinates.
        unum = pn - lam * coeff(s, scale(-h, mu))
        vnum = ln - lam * coeff(r, scale(h, mu))
        assert unum % ell == 0
        assert vnum % p == 0
        u = unum // ell
        v = vnum // p

        assert p * ep[mu] - ell * u == lam * (
            coeff(s, scale(-h, mu)) - b(r)
        )
        assert p * v - ell * el[mu] == lam * (
            b(s) - coeff(r, scale(h, mu))
        )

    assert p * ep[O] - ell * el[O] == 5 * (b(s) - b(r))

    # One explicit unsaturated p*ell determinant carrier.
    mu = (1, 1, 1)
    nu = (1, 1, 0)
    D = (
        coeff(n, scale(p, mu)) * coeff(n, scale(ell, nu))
        - coeff(n, scale(p, nu)) * coeff(n, scale(ell, mu))
    )
    assert D % (p * ell) == 0

    return {
        "pair": (p, ell),
        "h": h,
        "r": r,
        "s": s,
        "max_E_bits": max(
            max(abs(x).bit_length() for x in ep.values()),
            max(abs(x).bit_length() for x in el.values()),
        ),
        "det_bits": abs(D).bit_length(),
    }


def hidden_edge_gcd(h):
    g = 0
    for j in range(1, h):
        g = gcd(g, comb(h, j))
    return g


# First nontrivial odd-prime gaps.
assert hidden_edge_gcd(2) == 2
assert hidden_edge_gcd(4) == 2
assert coeff(2, (2, 2, 1)) == 2
assert coeff(4, (4, 4, 3)) == 4
assert coeff(4, (4, 4, 2)) == 6

# n=321 pair Fitting contents.
assert hidden_edge_gcd(14) == 1
assert hidden_edge_gcd(18) == 1
assert hidden_edge_gcd(32) == 2

for pair in ((179, 193), (193, 211), (179, 211)):
    print(corridor_pair(321, *pair))

print("Q5701 audit passed")
```
