#!/usr/bin/env python3
"""Exact audit of the denominator-free Apéry precision-p^8 target law."""

from __future__ import annotations

import argparse
from collections import Counter
from fractions import Fraction
from math import comb, gcd, isqrt

import sympy as sp

ORDER = 9  # coefficients X^0,...,X^8


def primes_at_most(limit: int) -> list[int]:
    sieve = bytearray(b"\x01") * (limit + 1)
    sieve[:2] = b"\x00\x00"
    for p in range(2, isqrt(limit) + 1):
        if sieve[p]:
            sieve[p * p : limit + 1 : p] = b"\x00" * (
                (limit - p * p) // p + 1
            )
    return [p for p in range(2, limit + 1) if sieve[p]]


def P(n: int) -> int:
    return 34 * n**3 + 51 * n**2 + 27 * n + 5


def apery_numbers(limit: int) -> list[int]:
    A = [1, 5]
    for n in range(1, limit):
        numerator = P(n) * A[n] - n**3 * A[n - 1]
        denominator = (n + 1) ** 3
        assert numerator % denominator == 0
        A.append(numerator // denominator)
    return A


def zero_poly() -> list[Fraction]:
    return [Fraction(0) for _ in range(ORDER)]


def padded(values: list[int | Fraction]) -> list[Fraction]:
    out = zero_poly()
    for i, value in enumerate(values):
        out[i] = Fraction(value)
    return out


def add(a: list[Fraction], b: list[Fraction]) -> list[Fraction]:
    return [a[i] + b[i] for i in range(ORDER)]


def scale(a: list[Fraction], scalar: int) -> list[Fraction]:
    return [scalar * coefficient for coefficient in a]


def multiply(a: list[Fraction], b: list[Fraction]) -> list[Fraction]:
    out = zero_poly()
    for i, ai in enumerate(a):
        for j, bj in enumerate(b):
            if i + j >= ORDER:
                break
            out[i + j] += ai * bj
    return out


def inverse_shift_cube(shift: int) -> list[Fraction]:
    return [
        Fraction((-1) ** d * comb(d + 2, 2), shift ** (d + 3))
        for d in range(ORDER)
    ]


def shifted_fundamental_solutions(
    limit: int,
) -> tuple[list[list[Fraction]], list[list[Fraction]]]:
    zero = zero_poly()
    one = zero.copy()
    one[0] = Fraction(1)

    P_at_X = padded([5, 27, 51, 34])
    U = [one, multiply(P_at_X, inverse_shift_cube(1))]
    V = [zero, one]

    for n in range(1, limit):
        P_shift = padded(
            [
                P(n),
                102 * n**2 + 102 * n + 27,
                102 * n + 51,
                34,
            ]
        )
        cube_shift = padded([n**3, 3 * n**2, 3 * n, 1])
        denominator_inverse = inverse_shift_cube(n + 1)

        for sequence in (U, V):
            numerator = add(
                multiply(P_shift, sequence[n]),
                scale(multiply(cube_shift, sequence[n - 1]), -1),
            )
            sequence.append(multiply(numerator, denominator_inverse))

    return U, V


def fraction_mod(value: Fraction, modulus: int) -> int:
    assert gcd(value.denominator, modulus) == 1
    return (
        value.numerator
        * pow(value.denominator, -1, modulus)
        % modulus
    )


def evaluate_mod(
    polynomial: list[Fraction], argument: int, modulus: int
) -> int:
    return sum(
        fraction_mod(coefficient, modulus)
        * pow(argument, degree, modulus)
        for degree, coefficient in enumerate(polynomial)
    ) % modulus


def divide_residue(numerator: int, p: int, modulus_before: int) -> int:
    residue = numerator % modulus_before
    assert residue % p == 0
    return residue // p


def assert_p_integral(value: Fraction, p: int) -> None:
    # Fraction has already been reduced. A p in the denominator would mean
    # a negative p-adic valuation.
    assert value.denominator % p != 0


def symbolic_checks() -> None:
    x, H, F, e = sp.symbols("x H F e")
    a = sp.Integer(166144)
    c = sp.Integer(33296)

    D = x * (1 - H / 5) - e / 5
    Z = x * (1 - sp.Rational(336, 25) * H - F / 5) + a * e / 25
    residual = sp.factor((a + c * H + F) * D + 5 * Z - (a + 5) * x)

    assert sp.expand(
        residual + (c * H + F) * (H * x + e) / 5
    ) == 0
    assert 336 * 584 - 5 * 6016 == 166144
    assert (166144 + 336) // 5 == 33296
    assert 166144 + 5 == 166149
    assert sp.factorint(166149) == {3: 2, 18461: 1}


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser()
    parser.add_argument("--prime-limit", type=int, default=1000)
    return parser.parse_args()


def main() -> None:
    args = parse_args()
    primes = [p for p in primes_at_most(args.prime_limit) if p >= 11]
    if not primes:
        raise SystemExit("--prime-limit must be at least 11")

    maximum_prime = max(primes)
    A = apery_numbers(2 * maximum_prime)
    U, V = shifted_fundamental_solutions(maximum_prime - 1)
    J = [multiply(V[j], inverse_shift_cube(1)) for j in range(maximum_prime)]

    symbolic_checks()
    checks: Counter[str] = Counter()

    for p in primes:
        p2, p3 = p**2, p**3
        p5, p6, p7 = p**5, p**6, p**7
        p8, p9 = p**8, p**9

        delta = A[p - 1] - 1
        endpoint_H = A[p] - 5 + 7 * delta
        assert delta % p3 == 0
        assert endpoint_H % p5 == 0

        E = (
            Fraction(A[2 * p] - 73 + 824 * delta)
            - Fraction(752, 5) * endpoint_H
        )
        F = (
            Fraction(A[2 * p - 1] - 5 - 8 * delta)
            - Fraction(336, 5) * endpoint_H
        )

        w = E / p6
        f6 = F / p6
        bar_v = (769 * F + 103 * E) / p7
        assert_p_integral(w, p)
        assert_p_integral(f6, p)
        assert_p_integral(bar_v, p)
        checks["endpoint_integrality"] += 1

        delta0 = (delta // p3) % p
        assert fraction_mod(w, p) == fraction_mod(
            Fraction(24 * 769, 5) * delta0**2, p
        )
        assert fraction_mod(f6, p) == fraction_mod(
            -Fraction(24 * 103, 5) * delta0**2, p
        )
        checks["endpoint_first_digit"] += 1

        if p != 769:
            v = (F + Fraction(103, 769) * E) / p7
            assert_p_integral(v, p)
            assert v == bar_v / 769
            assert F == -Fraction(103, 769) * p6 * w + p7 * v
            checks["v_integrality"] += 1

        # Exact Casoratian and universal reflected J formula.
        c = [J[j][0] for j in range(p)]
        for value in c:
            assert_p_integral(value, p)

        assert (
            A[p - 2] * c[p - 1] - A[p - 1] * c[p - 2]
            == Fraction(1, (p - 1) ** 3)
        )
        checks["casoratian"] += 1

        for j in range(p):
            reflected_residual = (
                c[p - 1 - j] - c[j] - c[p - 1] * A[j]
            )
            assert fraction_mod(reflected_residual, p) == 0
            checks["universal_J_reflection"] += 1

        for r in range(p):
            if A[r] % p != 0:
                continue

            s = p - 1 - r
            assert A[s] % p == 0
            assert fraction_mod(J[s][0] - J[r][0], p) == 0
            checks["target_J_reflection"] += 1

            upper = p + r
            assert A[upper] % p == 0
            x_exact = A[upper] // p
            x = x_exact % p8

            U_r = evaluate_mod(U[r], p, p9)
            J_r = evaluate_mod(J[r][:6], p, p9)
            U_s = evaluate_mod(U[s], -2 * p, p9)
            J_s = evaluate_mod(J[s][:6], -2 * p, p9)

            # The retained U arrays already contain only degrees 0,...,8.
            assert U_r % p == 0
            assert U_s % p == 0

            forward = (
                A[p] * U_r - p3 * A[p - 1] * J_r
            ) % p9
            reflected = (
                A[2 * p - 1] * U_s
                + 8 * p3 * A[2 * p] * J_s
            ) % p9
            assert forward == A[upper] % p9
            assert reflected == A[upper] % p9
            checks["exact_shifted_decomposition"] += 2

            direct_numerator = (
                (5 - 7 * delta) * U_r
                - p3 * (1 + delta) * J_r
            )
            reflected_numerator = (
                (5 + 8 * delta) * U_s
                + 8 * p3 * (73 - 824 * delta) * J_s
            )
            D8 = divide_residue(direct_numerator, p, p9)
            Z8 = divide_residue(reflected_numerator, p, p9)

            q = U_r // p
            q_reflected = U_s // p
            predicted_q = fraction_mod(
                Fraction(x, 5) + Fraction(p2, 5) * J[r][0], p3
            )
            predicted_q_reflected = fraction_mod(
                Fraction(x, 5) - Fraction(584 * p2, 5) * J[s][0],
                p3,
            )
            assert q % p3 == predicted_q
            assert q_reflected % p3 == predicted_q_reflected
            checks["quotient_auxiliary_rows"] += 2

            predicted_D8 = fraction_mod(
                Fraction(x) * (1 - Fraction(endpoint_H, 5))
                - Fraction(p2 * endpoint_H, 5) * J[r][0],
                p8,
            )
            predicted_Z8 = fraction_mod(
                Fraction(x)
                * (
                    1
                    - Fraction(336 * endpoint_H, 25)
                    - F / 5
                )
                + Fraction(166144 * p2 * endpoint_H, 25) * J[s][0],
                p8,
            )
            assert D8 == predicted_D8
            assert Z8 == predicted_Z8
            checks["direct_row"] += 1
            checks["reflected_row"] += 1

            K = Fraction(166144 + 33296 * endpoint_H) + F
            clean_left = (
                fraction_mod(K, p8) * D8 + 5 * Z8
            ) % p8
            assert clean_left == 166149 * x % p8
            checks["clean_fixed_law"] += 1

            integer_K = (
                5 * A[2 * p - 1]
                + 830695
                - 40 * delta
                + 166144 * endpoint_H
            )
            assert Fraction(integer_K, 5) == K
            assert (
                integer_K * D8 + 25 * Z8
            ) % p8 == 830745 * x % p8
            checks["integer_fixed_law"] += 1

            if p != 769:
                v = (F + Fraction(103, 769) * E) / p7
                user_K = (
                    Fraction(769)
                    * (166144 + 33296 * endpoint_H + p7 * v)
                    - 103 * p6 * w
                )
                assert user_K == 769 * K
                assert (
                    fraction_mod(user_K, p8) * D8 + 3845 * Z8
                ) % p8 == (769 * 166149 * x) % p8
                checks["user_wv_law"] += 1

    print(f"prime_limit={args.prime_limit}")
    print("symbolic_checks=PASS")
    for name in sorted(checks):
        print(f"{name}={checks[name]}")
    print("validity_range=p>=11")
    print("wv_normalization_exception=769")
    print("fixed_inversion_exception=18461")
    print("failures=0")


if __name__ == "__main__":
    main()
