#!/usr/bin/env python3
from __future__ import annotations

from dataclasses import dataclass
from math import comb, gcd, isqrt, log, prod
from typing import Iterable


def C(n: int, k: int) -> int:
    return comb(n, k) if 0 <= k <= n else 0


def primes_upto(limit: int) -> list[int]:
    mark = bytearray(b"\x01") * (limit + 1)
    if limit >= 0:
        mark[0] = 0
    if limit >= 1:
        mark[1] = 0
    for p in range(2, isqrt(limit) + 1):
        if mark[p]:
            mark[p * p : limit + 1 : p] = b"\x00" * (
                (limit - p * p) // p + 1
            )
    return [p for p in range(2, limit + 1) if mark[p]]


def apery_coefficient(m: int) -> int:
    return 34 * m**3 + 51 * m**2 + 27 * m + 5


def apery_mod(r: int, p: int) -> int:
    """Return b_r modulo p for 0 <= r < p."""
    if r == 0:
        return 1 % p
    previous, current = 1 % p, 5 % p
    for m in range(1, r):
        numerator = (
            apery_coefficient(m) * current - m**3 * previous
        ) % p
        denominator = (m + 1) ** 3 % p
        following = numerator * pow(denominator, -1, p) % p
        previous, current = current, following
    return current


def shell_exact(M: int, d: int) -> int:
    r"""Return C_M(d)=sum_{v in d Z^3} [X^v] Lambda^M exactly.

    The exact coefficient formula is

      c_M(u,v,w)=sum_t binom(M,t) binom(M,t+u)
                    binom(M+t,t+v) binom(M+t,t+w).

    Summing u,v,w over dZ factorizes the inner three sums.
    """
    assert M >= 0 and d >= 1
    a = M // d
    total = 0
    for t in range(M + 1):
        x_packet = sum(
            C(M, t + d * u) for u in range(-a, a + 1)
        )
        yz_packet = sum(
            C(M + t, t + d * v) for v in range(-a, a + 1)
        )
        total += C(M, t) * x_packet * yz_packet * yz_packet
    return total


def valuation(value: int, p: int) -> int:
    assert p >= 2 and value != 0
    value = abs(value)
    out = 0
    while value % p == 0:
        value //= p
        out += 1
    return out


def log_int(value: int) -> float:
    """Accurate natural logarithm without converting a huge int to float."""
    value = abs(value)
    if value == 0:
        return float("-inf")
    bits = value.bit_length()
    if bits <= 53:
        return log(value)
    shift = bits - 53
    mantissa = value >> shift
    return log(mantissa) + shift * log(2.0)


def interlaced(values: list[int]) -> list[int]:
    """Endpoints first; often kills a nuisance gcd early."""
    out: list[int] = []
    lo, hi = 0, len(values) - 1
    while lo <= hi:
        out.append(values[lo])
        lo += 1
        if lo <= hi:
            out.append(values[hi])
            hi -= 1
    return out


@dataclass
class BlockResult:
    n: int
    a: int
    lo: int
    hi: int
    primes: tuple[int, ...]
    targets: tuple[int, ...]
    nuisance: int
    D: int
    common_shell_gcd: int | None
    outside_common: int | None
    target_excess: dict[int, int]
    exact_shells_used: int

    @property
    def H(self) -> float:
        return self.n ** (1.0 / 3.0)

    @property
    def nuisance_log_over_H(self) -> float:
        return 0.0 if self.nuisance == 1 else log_int(self.nuisance) / self.H


def candidate_primes(
    n: int,
    a: int,
    lo: int,
    hi: int,
    prime_table: Iterable[int],
) -> list[int]:
    out = []
    for q in prime_table:
        if q < lo:
            continue
        if q > hi:
            break
        if n // q != a:
            continue
        r = n - a * q
        if 1 <= r <= q - 2:
            out.append(q)
    return out


def analyze_block(
    n: int,
    a: int,
    lo: int,
    hi: int,
    prime_table: list[int],
    *,
    force_full: bool = False,
) -> BlockResult | None:
    qs = candidate_primes(n, a, lo, hi, prime_table)
    if len(qs) < 2:
        return None

    M = n - a
    target_list = [
        q for q in qs if apery_mod(n - a * q, q) == 0
    ]
    target_set = set(target_list)
    P = prod(qs)
    T = prod(target_list)

    nuisance = 0
    shell_values: dict[int, int] = {}
    used = 0

    for q in interlaced(qs):
        xq = shell_exact(M, q - 1)
        used += 1
        # Fixed-moment torus theorem regression.
        assert xq % q == apery_mod(n - a * q, q)
        shell_values[q] = xq

        scaled = (P // q) * xq
        assert scaled % T == 0
        nuisance = gcd(nuisance, scaled // T)

        if nuisance == 1 and not force_full:
            return BlockResult(
                n=n,
                a=a,
                lo=lo,
                hi=hi,
                primes=tuple(qs),
                targets=tuple(target_list),
                nuisance=1,
                D=T,
                common_shell_gcd=None,
                outside_common=None,
                target_excess={},
                exact_shells_used=used,
            )

    # If force_full was requested, fill values skipped by interlacing only
    # in case the ordering routine is later changed.
    for q in qs:
        if q not in shell_values:
            shell_values[q] = shell_exact(M, q - 1)
            used += 1

    G = 0
    for q in qs:
        G = gcd(G, shell_values[q])

    outside = G
    for p in qs:
        while outside % p == 0:
            outside //= p

    excess: dict[int, int] = {}
    excess_product = 1
    for p in target_list:
        alpha = valuation(shell_values[p], p)
        beta = min(
            valuation(shell_values[q], p)
            for q in qs
            if q != p
        )
        exponent = min(alpha - 1, beta)
        if exponent:
            excess[p] = exponent
            excess_product *= p**exponent

    classified_nuisance = outside * excess_product
    assert nuisance == classified_nuisance
    D = T * nuisance

    # Direct definition regression.
    direct_D = 0
    for q in qs:
        direct_D = gcd(direct_D, (P // q) * shell_values[q])
    assert direct_D == D

    return BlockResult(
        n=n,
        a=a,
        lo=lo,
        hi=hi,
        primes=tuple(qs),
        targets=tuple(target_list),
        nuisance=nuisance,
        D=D,
        common_shell_gcd=G,
        outside_common=outside,
        target_excess=excess,
        exact_shells_used=used,
    )


def quotient_slice_bounds(n: int, a: int) -> tuple[int, int]:
    # Filtering in candidate_primes handles exact endpoints/interior residues.
    return n // (a + 1) + 1, n // a


def subblocks(
    lo: int,
    hi: int,
    widths: Iterable[int],
    stride_divisor: int = 2,
) -> list[tuple[int, int]]:
    seen = {(lo, hi)}
    for width in widths:
        width = max(2, width)
        stride = max(1, width // stride_divisor)
        start = lo
        while start <= hi:
            end = min(hi, start + width - 1)
            seen.add((start, end))
            if end == hi:
                break
            start += stride
    return sorted(seen)


def scan(
    max_n: int,
    max_a: int,
    *,
    start_n: int = 20,
) -> list[BlockResult]:
    ps = primes_upto(max_n)
    records: list[BlockResult] = []
    maximum: BlockResult | None = None

    for n in range(start_n, max_n + 1):
        H = max(2, round(n ** (1.0 / 3.0)))
        widths = (H, H * H)

        for a in range(1, min(max_a, n - 1) + 1):
            lo, hi = quotient_slice_bounds(n, a)
            if lo > hi:
                continue

            for left, right in subblocks(lo, hi, widths):
                result = analyze_block(n, a, left, right, ps)
                if result is None:
                    continue
                records.append(result)

                if maximum is None or (
                    result.nuisance_log_over_H
                    > maximum.nuisance_log_over_H
                ):
                    maximum = result
                    print("NEW MAX", maximum)

    print("blocks", len(records))
    print("nontrivial nuisance", sum(r.nuisance > 1 for r in records))
    print("higher target valuations", sum(bool(r.target_excess) for r in records))
    print("maximum", maximum)
    return records


def regression_321() -> None:
    ps = primes_upto(321)
    result = analyze_block(
        321, 1, 162, 211, ps, force_full=True
    )
    assert result is not None
    assert result.targets == (179, 193, 211)
    assert result.nuisance == 1
    assert result.D == 179 * 193 * 211
    print("PASS n=321", result)


if __name__ == "__main__":
    regression_321()
