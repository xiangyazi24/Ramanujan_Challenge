# 13. Dependency-free verification code

The following standard-library program verifies the exact fixed-point identity and the Newton carrier. It performs no heuristic density experiment and needs no factorization package.

```python
#!/usr/bin/env python3
from math import comb


def apery_upto(limit: int) -> list[int]:
    """Exact Apéry numbers from the order-two recurrence."""
    if limit < 0:
        return []
    values = [1]
    if limit == 0:
        return values
    values.append(5)
    for n in range(1, limit):
        middle = (2 * n + 1) * (17 * n * n + 17 * n + 5)
        numerator = middle * values[n] - n**3 * values[n - 1]
        denominator = (n + 1) ** 3
        assert numerator % denominator == 0
        values.append(numerator // denominator)
    return values


def prime_flags(limit: int) -> bytearray:
    flags = bytearray(b"\x01") * (limit + 1)
    if limit >= 0:
        flags[0] = 0
    if limit >= 1:
        flags[1] = 0
    p = 2
    while p * p <= limit:
        if flags[p]:
            flags[p * p : limit + 1 : p] = b"\x00" * (
                (limit - p * p) // p + 1
            )
        p += 1
    return flags


def forward_difference_heads(values: list[int]) -> list[int]:
    row = values[:]
    heads: list[int] = []
    while row:
        heads.append(row[0])
        row = [row[i + 1] - row[i] for i in range(len(row) - 1)]
    return heads


def newton_carrier(moment: int, prefix: int, apery: list[int]) -> int:
    heads = forward_difference_heads(apery[: prefix + 1])
    return sum(heads[j] * comb(moment, j) for j in range(prefix + 1))


def audit(limit: int = 2000) -> None:
    apery = apery_upto(limit)
    prime = prime_flags(limit)
    identity_checks = 0
    newton_checks = 0

    for moment in range(24, limit + 1):
        H = max(1, round(moment ** (1 / 3)))
        prefix = max(0, moment // 4 - 2 * H)

        direct = sorted(
            moment - s
            for s in range(prefix + 1)
            if prime[moment - s] and apery[s] % (moment - s) == 0
        )
        interval = sorted(
            p
            for p in range(moment - prefix, moment + 1)
            if prime[p] and apery[moment] % p == 0
        )
        assert direct == interval
        identity_checks += len(interval)

        carrier = newton_carrier(moment, prefix, apery)
        for s in range(prefix + 1):
            p = moment - s
            if not prime[p]:
                continue
            assert carrier % p == apery[s] % p
            assert apery[moment] % p == 5 * apery[s] % p
            newton_checks += 1

    print("FIXED_POINT_TARGET_CHECKS", identity_checks)
    print("NEWTON_CONGRUENCE_CHECKS", newton_checks)
    print("PASS")


if __name__ == "__main__":
    audit()
```
